from __future__ import annotations
import argparse, json, time

def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--family", choices=["hi_nerv", "snerv", "pr95"], required=True)
    ap.add_argument("--pairs", type=int, default=4)
    ap.add_argument("--steps", type=int, default=4)
    ap.add_argument("--out", required=True)
    args = ap.parse_args()
    row = {
        "schema": "nerv_crux_trace_row.v1",
        "family": args.family,
        "status": "stub",
        "todo": [
            "init", "live_forward", "eval_roundtrip", "scorer_outputs",
            "loss_terms", "grad_norm_by_group", "optimizer_update",
            "fakequant", "archive_build", "parseback", "replay_score"
        ],
        "captured_at_unix": time.time(),
    }
    with open(args.out, "w") as f:
        f.write(json.dumps(row, sort_keys=True) + "\n")
    return 0

if __name__ == "__main__":
    raise SystemExit(main())
