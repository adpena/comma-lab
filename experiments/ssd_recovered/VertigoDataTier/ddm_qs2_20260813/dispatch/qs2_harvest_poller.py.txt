#!/usr/bin/env python3
"""Detached harvest poller for the qs2 R2 dual-axis dispatch (qs1 poller pattern,
parameterized). Polls FunctionCall.get, writes the remote result JSON + ledger
outcome + .done/.failed markers. Reaper-safe SSD path; launched via
tools/launch_detached_process.py ONLY."""
import json, sys, time, pathlib

CALL_ID = "fc-01KZYQJE57P4FV9PS0SXMA2BXM"
OUT = pathlib.Path("/Volumes/VertigoDataTier/pact/ddm_qs2_20260813/dispatch/ddm_qs2_dual_axis_20260813_r2")
RESULT = OUT / "QS2_T4_REMOTE_RESULT.json"
DEADLINE_S = 3 * 3600
POLL_S = 60

sys.path.insert(0, "/Users/adpena/Projects/pact/src")

def mark(status, rc, harvest):
    from tac.deploy.modal.call_id_ledger import update_call_id_outcome
    update_call_id_outcome(call_id=CALL_ID, status=status, rc=rc, agent="MAIN",
                           harvest_result=harvest)

def main():
    import modal
    t0 = time.time()
    while time.time() - t0 < DEADLINE_S:
        try:
            r = modal.functions.FunctionCall.from_id(CALL_ID).get(timeout=5)
        except TimeoutError:
            time.sleep(POLL_S)
            continue
        except Exception as e:  # terminal remote failure
            (OUT / "poller.failed").write_text(f"{type(e).__name__}: {e}\n")
            mark("failed", 1, {"error": str(e)[:500]})
            return 1
        RESULT.write_text(json.dumps(r, indent=2, default=str))
        mark("completed", 0, {"result_path": str(RESULT),
                              "net_delta_S_projection": -4.3749179e-6})
        (OUT / "poller.done").write_text("ok\n")
        return 0
    (OUT / "poller.failed").write_text("deadline 3h exceeded\n")
    mark("failed", 124, {"error": "poller deadline 3h"})
    return 1

if __name__ == "__main__":
    sys.exit(main())
