def _sha(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


# --------------------------------------------------------------------------- #
# Quantization ladder (each returns dequantized float64 + a storage-floor note)
# --------------------------------------------------------------------------- #
def quant_identity(arr: np.ndarray) -> np.ndarray:
    return arr.astype(np.float64, copy=True)


def quant_fp32(arr: np.ndarray) -> np.ndarray:
    return arr.astype(np.float32).astype(np.float64)


def quant_fp16(arr: np.ndarray) -> np.ndarray:
    return arr.astype(np.float16).astype(np.float64)


def _uint_levels(arr: np.ndarray, *, bits: int, per: str) -> np.ndarray:
    """Quantize to 2**bits levels with min/max scale, dequantize to float64.

    per: 'global' | 'frame' | 'frame_channel'  (scale granularity).
    """
    levels = (1 << bits) - 1
    a = arr.astype(np.float64, copy=False)
    if per == "global":
        lo = float(a.min())
        hi = float(a.max())
        span = max(hi - lo, 1e-12)
        codes = np.rint((a - lo) / span * levels)
        return (codes / levels * span + lo).astype(np.float64)
    if per == "frame":
        # axis 0 is the frame index (N,3,H,W)
        lo = a.min(axis=(1, 2, 3), keepdims=True)
        hi = a.max(axis=(1, 2, 3), keepdims=True)
    elif per == "frame_channel":
        lo = a.min(axis=(2, 3), keepdims=True)
        hi = a.max(axis=(2, 3), keepdims=True)
    else:
        raise ValueError(per)
    span = np.maximum(hi - lo, 1e-12)
    codes = np.rint((a - lo) / span * levels)
    return (codes / levels * span + lo).astype(np.float64)


def quant_uint8_frame(arr):
    return _uint_levels(arr, bits=8, per="frame")


def quant_uint8_frame_channel(arr):
    return _uint_levels(arr, bits=8, per="frame_channel")


def quant_uint8_global(arr):
    return _uint_levels(arr, bits=8, per="global")


def quant_uint6_frame(arr):
    return _uint_levels(arr, bits=6, per="frame")


def quant_uint4_frame(arr):
    return _uint_levels(arr, bits=4, per="frame")

# Re-encode the decoder_payload with a mutated skip_high (float64 manifest path)

def reencode_decoder_payload(decoder_payload: bytes, new_skip_high: np.ndarray) -> bytes:
    from tac.substrates.snerv_inverse_steg_carrier import archive as A

    header, comp = A._unpack_subpacket(
        decoder_payload,
        magic=A.SNERV_DECODER_MAGIC,
        schema=A.DECODER_PAYLOAD_OFFICIAL_MFU_HFR_TUB_SCHEMA,
    )
    raw = lzma.decompress(comp)
    tensors = A._unpack_tensor_manifest(raw, header.get("tensor_manifest") or [])
    tensors[SKIP_HIGH_NAME] = np.ascontiguousarray(new_skip_high, dtype=np.float64)
    manifest, new_raw = A._pack_tensor_manifest(tensors)
    new_comp = lzma.compress(
        new_raw,
        format=lzma.FORMAT_XZ,
        preset=A.DECODER_PAYLOAD_OFFICIAL_MFU_HFR_TUB_LZMA_PRESET,
    )
    new_header = dict(header)
    new_header["tensor_manifest"] = manifest
    new_header["tensor_count"] = len(manifest)
    new_header["raw_tensor_bytes"] = len(new_raw)
    new_header["raw_tensor_sha256"] = A._sha256(new_raw)
    new_header["compressed_bytes"] = len(new_comp)
    new_header["compressed_sha256"] = A._sha256(new_comp)
    return A._pack_subpacket(A.SNERV_DECODER_MAGIC, new_header, new_comp)


def rebuild_packet(orig_packet: bytes, new_decoder_payload: bytes) -> bytes:
    from tac.substrates.snerv_inverse_steg_carrier import archive as A

    decoded = A.unpack_snerv_archive(orig_packet)
    sections = dict(decoded.sections)
    sections["decoder_payload"] = new_decoder_payload
    return A.pack_snerv_archive_snar2(
        metadata_payload=sections["metadata_payload"],
        lf_payload=sections["lf_payload"],
        decoder_payload=sections["decoder_payload"],
        step_map_packet=sections["step_map_packet"],
        metadata=decoded.metadata,
    ).packet


# --------------------------------------------------------------------------- #
# Carrier LF-codec storage-floor (lossless intN + brotli/lzma on integer codes)
# --------------------------------------------------------------------------- #
def carrier_lf_codec_storage_bytes(arr: np.ndarray, *, bits: int, per: str):
    """Quantize skip_high to int codes, encode via the carrier's lossless LF v2
    codec, return (code_payload_bytes, scale_sidecar_bytes, dequant_float64).

    This is the carrier's OWN bandwise quantizer + entropy coder (path-A
    machinery the G1b verdict said is UNUSED on skip_high). It measures the true
    entropy-coded storage of the quantization; the dequant_float64 is what the
    receiver would reconstruct.
    """
    from tac.substrates.snerv_inverse_steg_carrier.lf_payload_codec import (
        encode_lf_quant_payload_v2_with_report,
    )

    levels = (1 << bits) - 1
    a = arr.astype(np.float64, copy=False)
    if per == "frame":
        lo = a.min(axis=(1, 2, 3), keepdims=True)
        hi = a.max(axis=(1, 2, 3), keepdims=True)
    elif per == "frame_channel":
        lo = a.min(axis=(2, 3), keepdims=True)
        hi = a.max(axis=(2, 3), keepdims=True)
    else:
        raise ValueError(per)
    span = np.maximum(hi - lo, 1e-12)
    codes = np.rint((a - lo) / span * levels).astype(np.int64)
    dequant = (codes.astype(np.float64) / levels * span + lo).astype(np.float64)
    # One signed-int plane per frame. The carrier LF codec requires 2-D integer
    # planes (_validate_lf_plane), so flatten each frame's (3,H,W) codes into a
    # 2-D (3, H*W) plane (channel-major rows, spatial columns) — lossless re the
    # stored byte count, which is order-invariant for the entropy coder.
    planes = [
        codes[i].reshape(codes.shape[1], -1) for i in range(codes.shape[0])
    ]
    packet, report = encode_lf_quant_payload_v2_with_report(
        planes, mode="portfolio_auto", wrapper="portfolio_auto"
    )
    # Scale sidecar: lo+span as fp32, lzma-compressed.
    scale_raw = np.concatenate(
        [lo.astype(np.float32).reshape(-1), span.astype(np.float32).reshape(-1)]
    ).tobytes()
    scale_blob = lzma.compress(scale_raw, preset=6)
    return len(packet), len(scale_blob), dequant

def score_packet(packet: bytes, n_pairs: int, source_pairs, posenet, segnet):
    from tac.substrates.snerv_inverse_steg_carrier.archive import (
        decode_snerv_archive_frames,
    )
    from tac.substrates.snerv_inverse_steg_carrier.inflate import _resize_nchw_bilinear
    from tac.analysis.inverse_steganalysis_linf_vs_l2_gate import (
        measure_pairs_d_seg_d_pose_batched,
    )
    import torch

    t0 = time.time()
    frames = decode_snerv_archive_frames(packet)
    decode_s = time.time() - t0
    arr = np.asarray(frames, dtype=np.float32)
    if arr.ndim == 5 and arr.shape[1] == 2:
        pairs_arr = arr
    elif arr.ndim == 4 and arr.shape[0] % 2 == 0:
        pairs_arr = arr.reshape(arr.shape[0] // 2, 2, *arr.shape[1:])
    else:
        raise RuntimeError(f"unexpected decoded frame shape {arr.shape}")
    cam_h, cam_w = int(source_pairs.shape[-2]), int(source_pairs.shape[-1])
    n = min(n_pairs, int(pairs_arr.shape[0]))
    sel = pairs_arr[:n]
    flat = sel.reshape(-1, 3, sel.shape[-2], sel.shape[-1])
    resized = _resize_nchw_bilinear(flat, out_hw=(cam_h, cam_w))
    u8 = np.rint(np.clip(resized, 0.0, 255.0)).astype(np.uint8)
    recon_np = u8.astype(np.float32).reshape(n, 2, 3, cam_h, cam_w)
    recon = torch.from_numpy(recon_np).to(source_pairs)
    if tuple(recon.shape) != tuple(source_pairs[:n].shape):
        raise RuntimeError(
            f"shape mismatch: recon {tuple(recon.shape)} vs source {tuple(source_pairs[:n].shape)}"
        )
    t1 = time.time()
    dsegs, dposes = measure_pairs_d_seg_d_pose_batched(
        posenet, segnet, source_pairs[:n], recon, batch_pairs=8
    )
    d_seg = float(np.mean(dsegs))
    d_pose = float(np.mean(dposes))
    return {
        "decode_seconds": decode_s,
        "score_seconds": time.time() - t1,
        "n_pairs_scored": n,
        "d_seg": d_seg,
        "d_pose": d_pose,
        "pose_score_term": float(np.sqrt(10.0 * max(d_pose, 0.0))),
        "per_pair_d_seg_first8": [float(v) for v in dsegs[:8]],
        "per_pair_d_pose_first8": [float(v) for v in dposes[:8]],
    }

def _zip_stored_bytes(packet: bytes) -> int:
    import io

    buf = io.BytesIO()
    with zipfile.ZipFile(buf, "w", zipfile.ZIP_STORED) as zf:
        zf.writestr("x", packet)
    return len(buf.getvalue())


def _weights_lzma_bytes(tensors) -> int:
    """LZMA-preset6 bytes of the float64 manifest of EVERYTHING except skip_high."""
    from tac.substrates.snerv_inverse_steg_carrier import archive as A

    sub = {k: v for k, v in tensors.items() if k != SKIP_HIGH_NAME}
    _manifest, raw = A._pack_tensor_manifest(sub)
    return len(lzma.compress(raw, format=lzma.FORMAT_XZ, preset=6))


def _best_rung(rungs):
    passing = [r for r in rungs if r["guard_pass"]]
    if not passing:
        return None
    best = min(passing, key=lambda r: r["advisory_total_packet_rate"])
    return {
        "rung_id": best["rung_id"],
        "advisory_total_packet_rate": best["advisory_total_packet_rate"],
        "packet_bytes": best["packet_bytes"],
        "d_seg": best["d_seg"],
        "pose_score_term": best["pose_score_term"],
    }


def _binding_gap(out):
    best = out.get("best_rung")
    if not best:
        return {"note": "no passing rung"}
    nonrate = 100.0 * best["d_seg"] + best["pose_score_term"]
    rate = CONTEST_BYTE_PRICE * best["packet_bytes"]
    return {
        "best_total": best["advisory_total_packet_rate"],
        "best_nonrate_floor": nonrate,
        "best_rate_term": rate,
        "frontier_target": 0.19199,
        "gap_vs_frontier": best["advisory_total_packet_rate"] - 0.19199,
        "nonrate_floor_vs_frontier": nonrate - 0.19199,
        "binding": "rate" if rate > nonrate else "nonrate",
        "note": (
            "non-rate floor 0.389 (seg 0.247 + pose 0.142) is ~2x PR101-class "
            "0.193 even at rate->0; rate must drop AND seg/pose must descend."
        ),
    }
