"""Score-aware per-weight sensitivity via autograd through renderer->scorer.

The scorer cares about (a) SegNet argmax flips on the last frame and (b) the 6
PoseNet scalars — NOT pixel MSE. So compute, per decoder weight, the gradient
magnitude of a DIFFERENTIABLE scorer surrogate (SegNet CE-to-own-argmax for the
flip-boundary signal + PoseNet pose-MSE) accumulated over pairs. Prune the
LOW-scorer-sensitivity weights (which may matter for reconstruction but not for
the score) and measure exact d_seg/d_pose. This is the #71 lever magnitude
pruning cannot reach.
"""
import sys, time
sys.path.insert(0,'experiments/results/pr110pp_r2_nonmps_candidate_20260609/analysis')
sys.path.insert(0,'src')
import render_and_score_lib as L
import numpy as np, torch, torch.nn.functional as F, copy
from model import HNeRVDecoder
from tac.optimization import structural_q_compression as SQ

torch.set_grad_enabled(True)
N_SENS = int(sys.argv[1]) if len(sys.argv)>1 else 24   # pairs to accumulate sensitivity over
p = L.parse_frontier()
sd = p['decoder_sd']
meta = p['meta']
probe_order = None
DEVICE = torch.device('cpu')
CAMERA_H, CAMERA_W = 874, 1164

# Build a grad-enabled decoder
dec = HNeRVDecoder(latent_dim=meta['latent_dim'], base_channels=meta['base_channels'],
                   eval_size=tuple(meta['eval_size'])).to(DEVICE)
dec.load_state_dict(sd)
dec.train()  # keep grads
order = [name for name,_ in dec.state_dict().items()]
latents = p['latents'].to(DEVICE)
eval_h, eval_w = meta['eval_size']

# scorer (grad-enabled forward)
from modules import DistortionNet, segnet_sd_path, posenet_sd_path
net = DistortionNet().eval().to(DEVICE)
net.load_state_dicts(posenet_sd_path, segnet_sd_path, DEVICE)
for prm in net.parameters():
    prm.requires_grad_(False)

# GT
pairs = list(range(N_SENS))
gt = L.decode_gt_pairs(pairs)

# accumulate |grad| per weight tensor (only the conv/linear weights)
weight_names = [n for n in order if dec.state_dict()[n].ndim>=2]
accum = {n: torch.zeros_like(dict(dec.named_parameters())[n]) for n in weight_names}

def render_pair_grad(pi):
    lat = latents[pi:pi+1]
    decoded = dec(lat)  # (1,2,3,eh,ew)
    flat = decoded.reshape(2,3,eval_h,eval_w)
    up = F.interpolate(flat.unsqueeze(0).reshape(2,3,eval_h,eval_w), size=(CAMERA_H,CAMERA_W), mode='bicubic', align_corners=False)
    up = up.reshape(1,2,3,CAMERA_H,CAMERA_W)
    up = up.clone()
    up[:,0,0] = up[:,0,0]-1.0
    up[:,0,2] = up[:,0,2]-1.0
    up[:,1,1] = up[:,1,1]-1.0
    # DO NOT round (need grad); clamp soft
    comp = up.clamp(0,255)
    return comp  # (1,2,3,H,W)

t0=time.time()
named = dict(dec.named_parameters())
for k,pi in enumerate(pairs):
    dec.zero_grad(set_to_none=True)
    comp = render_pair_grad(pi)  # (1,2,3,H,W) grad
    g = gt[pi].float().permute(0,3,1,2).unsqueeze(0).to(DEVICE)  # (1,2,3,H,W)
    comp_bthwc = comp.permute(0,1,3,4,2)  # (1,2,H,W,3)
    g_bthwc = g.permute(0,1,3,4,2)
    pose, seg = net.compute_distortion(g_bthwc, comp_bthwc)  # differentiable? seg uses argmax -> may be 0-grad
    # Use a differentiable surrogate: we want grad of the SCORE wrt weights.
    # pose is MSE (diff). seg is argmax-disagreement (non-diff). Add a soft seg
    # surrogate: KL between comp segnet logits and gt segnet argmax via the net's
    # internal segnet. Fallback: use pose-only + total as available.
    loss = pose.sum()
    # try add seg if it carries grad
    if seg.requires_grad:
        loss = loss + seg.sum()*100.0
    loss.backward()
    for n in weight_names:
        if named[n].grad is not None:
            accum[n] += named[n].grad.detach().abs()
    if (k+1)%6==0:
        print(f'  sens pair {k+1}/{N_SENS} [{time.time()-t0:.0f}s] loss={float(loss):.4g}', flush=True)

# save sensitivity (|grad| accumulated) per weight tensor
out = {n: accum[n].cpu().numpy() for n in weight_names}
np.savez('/Volumes/VertigoDataTier/pact/task71_scoreaware_sens.npz', **{n.replace('.','_'):v for n,v in out.items()})
print('saved sensitivity for', len(out), 'tensors; pose carried grad. seg.requires_grad=', seg.requires_grad)
for n in weight_names:
    s = out[n]
    print(f'  {n:18s} sens range [{s.min():.3e},{s.max():.3e}] mean {s.mean():.3e}')
