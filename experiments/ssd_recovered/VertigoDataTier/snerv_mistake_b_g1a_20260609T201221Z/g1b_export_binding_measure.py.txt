    return hashlib.sha256(b).hexdigest()

def run_path_a_advisory() -> tuple[dict, bytes]:

    t0 = time.time()
    advisory = run_snerv_advisory(
        n_pairs=600,

        target_bits_per_coeff=1.0,

        upstream_dir=str(REPO_ROOT / "upstream"),

        snerv_temporal_mode="delta",

        "lf_payload_bytes": int(advisory.lf_payload_bytes),

        "metadata_bytes": int(payload.get("metadata_bytes", 0)),

            payload.get("receiver_archive_header_bytes", 0)

        "archive_bytes_total_l2": int(advisory.archive_bytes_total_l2),
        "rate_term_l2": float(advisory.rate_term_l2),

        "z8_disease_detail_store_frac": float(advisory.z8_disease_detail_store_frac),

        "pr101_frontier_rate": float(advisory.pr101_frontier_rate),

# 3. INDEPENDENT INFLATE CONFIRM — build archive.zip, decode it, re-score.
# ---------------------------------------------------------------------------
def independent_inflate_confirm(packet: bytes) -> dict:
    from tac.substrates.snerv_inverse_steg_carrier.archive import (
        decode_snerv_archive_frames,
    )
    from tac.analysis.inverse_steganalysis_linf_vs_l2_gate import (
        measure_pairs_d_seg_d_pose_batched,
    )

    # Build the byte-closed archive.zip exactly as the contest member-name expects.
    pkg_dir = G1B_DIR / "snerv_archive_bound_package"
    pkg_dir.mkdir(parents=True, exist_ok=True)
    bin_path = pkg_dir / "0.bin"
    bin_path.write_bytes(packet)
    archive_zip = pkg_dir / "archive.zip"
    with zipfile.ZipFile(archive_zip, "w", zipfile.ZIP_STORED) as zf:
        # The carrier uses member name "x" (the byte-save). Mirror that.
        zf.writestr("x", packet)
    archive_bytes = archive_zip.stat().st_size
    archive_sha = _sha(archive_zip.read_bytes())

    # Decode the archive packet via the real (scorer-free) inflate path.
    t0 = time.time()
    frames = decode_snerv_archive_frames(packet)  # numpy receiver path
    decode_seconds = time.time() - t0

    # Re-score the inflated frames vs the SAME source pairs through real DistortionNet.
    # Real upstream scorers + real pair decode (the same source the advisory used).
    from tac.analysis.score_exact_saliency import (
        decode_real_pairs,
        load_score_exact_scorers,
    )

    confirm: dict = {
        "archive_zip_path": str(archive_zip),
        "archive_zip_bytes": int(archive_bytes),
        "archive_zip_sha256": archive_sha,
        "archive_member_name": "x",
        "decode_seconds": decode_seconds,
        "inflated_frames_shape": list(np.asarray(frames).shape),
    }
    try:
        posenet, segnet = load_score_exact_scorers(
            upstream_dir=str(REPO_ROOT / "upstream"), device="cpu"
        )
        n_total = int(np.asarray(frames).shape[0])
        # N>=48 subset per the task spec; the advisory's internal 600-pair
        # measurement is the primary archive-surface number — this is the
        # independent belt-and-suspenders confirm through the inflate path.
        n = min(48, n_total)
        source_pairs = decode_real_pairs(
            str(REPO_ROOT / "upstream/videos/0.mkv"), n, pair_stride=1, start_pair=0
        )
        recon = torch.from_numpy(np.asarray(frames)[:n]).to(source_pairs)
        dsegs, dposes = measure_pairs_d_seg_d_pose_batched(
            posenet, segnet, source_pairs, recon, batch_pairs=8
        )
        d_seg = float(np.mean(dsegs))
        d_pose = float(np.mean(dposes))
        score = 100.0 * d_seg + float(np.sqrt(10.0 * max(d_pose, 0.0))) + (
            CONTEST_BYTE_PRICE * archive_bytes
        )
        confirm.update(
            {
                "independent_inflate_d_seg": d_seg,
                "independent_inflate_d_pose": d_pose,
                "independent_inflate_score_advisory": score,
                "independent_inflate_n_pairs_scored": n,
                "independent_inflate_scorer": "real_upstream_segnet_posenet_compute_distortion",
            }
        )
    except Exception as exc:  # noqa: BLE001
        confirm["independent_inflate_rescore_error"] = f"{type(exc).__name__}: {exc}"
    return confirm


# ---------------------------------------------------------------------------
# 2b. PATH-B TRAINED PACKET (if the run's export tail landed it): the trained
#     official-conv MLX receiver state bound into a SNAR packet
#     (allocation_mode=official_mfu_hfr_tub_trained_mlx_receiver_payload,
#     mlx_native_train_export.py:8107-8144). Mechanically byte-closed; the
#     official-source-forward AUTHORITY claim remains honestly blocked.
# ---------------------------------------------------------------------------
def measure_path_b_trained_packet() -> dict | None:
    from tac.substrates.snerv_inverse_steg_carrier.archive import (
        decode_snerv_archive_frames,
        unpack_snerv_archive,
    )
    from tac.analysis.inverse_steganalysis_linf_vs_l2_gate import (
        measure_pairs_d_seg_d_pose_batched,
    )
    from tac.analysis.score_exact_saliency import (
        decode_real_pairs,
        load_score_exact_scorers,
    )

    candidates = sorted(
        RUN_DIR.rglob("*.snar"), key=lambda p: p.stat().st_mtime, reverse=True
    )
    if not candidates:
        return None
    packet_path = candidates[0]
    packet = packet_path.read_bytes()
    decoded = unpack_snerv_archive(packet)
    meta = dict(decoded.metadata)
    section_bytes = {
        name: len(blob) for name, blob in decoded.sections.items()
    }
    out: dict = {
        "packet_path": str(packet_path),
        "packet_bytes_total": len(packet),
        "packet_sha256": _sha(packet),
        "archive_schema": decoded.schema,
        "allocation_mode": meta.get("allocation_mode"),
        "hf_decoder_fit_mode": meta.get("hf_decoder_fit_mode"),
        "n_pairs": meta.get("n_pairs"),
        "official_skip_high_mode": (meta.get("model_size") or {}).get(
            "official_skip_high_mode"
        )
        if isinstance(meta.get("model_size"), dict)
        else meta.get("official_skip_high_mode"),
        "section_bytes": section_bytes,
        "rate_term": CONTEST_BYTE_PRICE * len(packet),
        "trained_state_packet": (
            meta.get("allocation_mode")
            == "official_mfu_hfr_tub_trained_mlx_receiver_payload"
        ),
    }
    # Byte-closed archive.zip wrap (single member "x", STORED) for the rate term.
    pkg_dir = G1B_DIR / "path_b_trained_archive"
    pkg_dir.mkdir(parents=True, exist_ok=True)
    archive_zip = pkg_dir / "archive.zip"
    with zipfile.ZipFile(archive_zip, "w", zipfile.ZIP_STORED) as zf:
        zf.writestr("x", packet)
    out["archive_zip_path"] = str(archive_zip)
    out["archive_zip_bytes"] = int(archive_zip.stat().st_size)
    out["archive_zip_sha256"] = _sha(archive_zip.read_bytes())
    out["rate_term_archive_zip"] = CONTEST_BYTE_PRICE * out["archive_zip_bytes"]

    # Inflate (real scorer-free receiver path) + score N>=48 pairs vs source.
    t0 = time.time()
    frames = decode_snerv_archive_frames(packet)
    out["decode_seconds"] = time.time() - t0
    arr = np.asarray(frames)
    out["inflated_frames_shape"] = list(arr.shape)
    posenet, segnet = load_score_exact_scorers(
        upstream_dir=str(REPO_ROOT / "upstream"), device="cpu"
    )
    n = min(48, int(arr.shape[0]))
    source_pairs = decode_real_pairs(
        str(REPO_ROOT / "upstream/videos/0.mkv"), n, pair_stride=1, start_pair=0
    )
    recon = torch.from_numpy(arr[:n]).to(source_pairs)
    dsegs, dposes = measure_pairs_d_seg_d_pose_batched(
        posenet, segnet, source_pairs, recon, batch_pairs=8
    )
    d_seg = float(np.mean(dsegs))
    d_pose = float(np.mean(dposes))
    out.update(
        {
            "archive_surface_d_seg": d_seg,
            "archive_surface_d_pose": d_pose,
            "archive_surface_n_pairs_scored": n,
            "archive_surface_scorer": "real_upstream_segnet_posenet_compute_distortion",
            "advisory_score_with_packet_rate": (
                100.0 * d_seg
                + float(np.sqrt(10.0 * max(d_pose, 0.0)))
                + CONTEST_BYTE_PRICE * len(packet)
            ),
        }
    )
    return out


OUT_PATH = RUN_DIR / "snerv_g1b_export_binding_verdict.v1.json"


def _flush() -> None:
    OUT["utc_last_flush"] = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    OUT_PATH.write_text(json.dumps(OUT, indent=2, sort_keys=True, default=str))


def main() -> int:
    OUT["path_b_export_probe"] = probe_path_b_export_blocker()
    print("[g1b] path-B export probe done; blocked =", not OUT["path_b_export_probe"]["export_bound"], flush=True)
    _flush()

    try:
        trained_packet = measure_path_b_trained_packet()
    except Exception as exc:  # noqa: BLE001
        trained_packet = {"error": f"{type(exc).__name__}: {exc}"}
    OUT["path_b_trained_packet"] = trained_packet
    print("[g1b] path-B trained packet:", (trained_packet or {}).get("packet_bytes_total", "NOT FOUND"), flush=True)
    _flush()

    try:
        adv, packet = run_path_a_advisory()
        OUT["path_a_advisory"] = adv
        OUT["path_a_advisory"]["receiver_archive_packet_bytes"] = len(packet)
        OUT["path_a_advisory"]["receiver_archive_packet_sha256"] = _sha(packet)
        print(
            "[g1b] path-A advisory done; archive_bytes_total_linf =",
            adv["byte_decomposition"]["archive_bytes_total_linf"],
            "d_seg =",
            adv["archive_surface_distortion"]["d_seg_mean_linf"],
            flush=True,
        )
        _flush()

        OUT["independent_inflate_confirm"] = independent_inflate_confirm(packet)
        print(
            "[g1b] independent inflate confirm done; d_seg =",
            OUT["independent_inflate_confirm"].get("independent_inflate_d_seg"),
            flush=True,
        )
    except Exception as exc:  # noqa: BLE001
        OUT["path_a_advisory_error"] = f"{type(exc).__name__}: {exc}"
        print("[g1b] path-A advisory FAILED:", OUT["path_a_advisory_error"], flush=True)

    _flush()
    print("[g1b] WROTE", OUT_PATH, flush=True)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
