#!/usr/bin/env python3
"""Retain a scorer-free byte price for the DDM BG1 4x8 bilinear gate.

This is an idempotent design-price materializer, not a training or scorer run.
Every raw/coded/archive payload it creates is retained beside a manifest.
"""

from __future__ import annotations

import brotli
import hashlib
import io
import json
import math
import os
from pathlib import Path
import shutil
import zipfile

import numpy as np


ROOT = Path("/Volumes/VertigoDataTier/pact/ddm_bg1_20260814/retained/pricing")
BASE_ARCHIVE = Path(
    "/Volumes/APDataStore/pact/submittable_custody_mirror_20260811/"
    "cp135_packet/adapted_runtime/archive.zip"
)
BASE_SHA256 = "6eb1a3b79cb167e03372339e07e93cae13b6ba3114a9eb917288bb038622edb6"
BASE_BYTES = 186_252
SURROGATE_MODULE = Path(
    "/Volumes/VertigoDataTier/pact/pr135_joint_solve_20260810/edge_conditioned/"
    "ddm_ec1_20260814/design/retained/oriented/"
    "seeded_nonzero_capacity_reference.br"
)
SURROGATE_MODULE_SHA256 = "616ee8d94bf14e5399e56e5009d4e98b5e10f965c3888073d57fe97d4e03b83e"
SURROGATE_ARCHIVE = Path(
    "/Volumes/VertigoDataTier/pact/pr135_joint_solve_20260810/edge_conditioned/"
    "ddm_ec1_20260814/design/retained/oriented/"
    "seeded_nonzero_capacity_reference.archive.zip"
)
SURROGATE_ARCHIVE_SHA256 = "aac8552c0718d96945f824c82769c52a692d015b1ef65e0c2ff27e59ab6cd008"
SURROGATE_ARCHIVE_BYTES = 187_959
MAGIC = b"EC1L\x01"
SEED = 20_260_814
PIXELS = 600 * 512 * 384
RATE_DENOMINATOR = 37_545_489


def sha256_bytes(value: bytes) -> str:
    return hashlib.sha256(value).hexdigest()


def record(path: Path) -> dict[str, object]:
    value = path.read_bytes()
    return {"path": str(path), "bytes": len(value), "sha256": sha256_bytes(value)}


def atomic_bytes(path: Path, value: bytes) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    temporary.write_bytes(value)
    os.replace(temporary, path)


def atomic_json(path: Path, value: object) -> None:
    atomic_bytes(path, (json.dumps(value, indent=2, sort_keys=True) + "\n").encode())


def require(path: Path, expected_sha256: str, expected_bytes: int | None = None) -> bytes:
    value = path.read_bytes()
    if sha256_bytes(value) != expected_sha256:
        raise RuntimeError(f"input SHA-256 mismatch: {path}")
    if expected_bytes is not None and len(value) != expected_bytes:
        raise RuntimeError(f"input byte count mismatch: {path}")
    return value


def parse_raw(raw: bytes) -> tuple[dict[str, object], list[tuple[dict[str, object], bytes]]]:
    if not raw.startswith(MAGIC):
        raise RuntimeError("surrogate module magic differs")
    offset = len(MAGIC)
    header_size = int.from_bytes(raw[offset : offset + 4], "little")
    offset += 4
    header = json.loads(raw[offset : offset + header_size])
    offset += header_size
    tensors: list[tuple[dict[str, object], bytes]] = []
    for metadata in header["tensors"]:
        size = int(metadata["bytes"])
        tensors.append((dict(metadata), raw[offset : offset + size]))
        offset += size
    if offset != len(raw):
        raise RuntimeError("surrogate module has trailing or missing bytes")
    return header, tensors


def deterministic_archive(module: bytes) -> bytes:
    with zipfile.ZipFile(BASE_ARCHIVE) as source:
        if source.namelist() != ["p"]:
            raise RuntimeError("CP135 base archive grammar differs")
        payload = source.read("p")
    stream = io.BytesIO()
    with zipfile.ZipFile(stream, "w", compression=zipfile.ZIP_STORED) as archive:
        for name, value in (("p", payload), ("ec1_latent.br", module)):
            info = zipfile.ZipInfo(name, date_time=(1980, 1, 1, 0, 0, 0))
            info.compress_type = zipfile.ZIP_STORED
            info.create_system = 3
            info.external_attr = 0o100644 << 16
            archive.writestr(info, value)
    return stream.getvalue()


def build_variant(
    label: str,
    original_header: dict[str, object],
    original_tensors: list[tuple[dict[str, object], bytes]],
    weight: np.ndarray,
    bias: np.ndarray,
) -> dict[str, object]:
    additions = [
        (
            {
                "name": "frame_gate.bias",
                "shape": [4],
                "dtype": "float16",
                "scale": None,
                "bytes": 8,
            },
            np.asarray(bias, dtype="<f2").tobytes(order="C"),
        ),
        (
            {
                "name": "frame_gate.weight",
                "shape": [4, 8],
                "dtype": "int8",
                "scale": 1.0 / 127.0,
                "bytes": 32,
            },
            np.asarray(weight, dtype=np.int8).tobytes(order="C"),
        ),
    ]
    tensors = sorted(original_tensors + additions, key=lambda item: str(item[0]["name"]))
    header = dict(original_header)
    header["schema"] = "ddm_ec1_latent_conditioner.v2"
    header["tensors"] = [metadata for metadata, _ in tensors]
    encoded_header = json.dumps(header, sort_keys=True, separators=(",", ":")).encode()
    raw = MAGIC + len(encoded_header).to_bytes(4, "little") + encoded_header + b"".join(
        payload for _, payload in tensors
    )
    coded = brotli.compress(raw, quality=11)
    repeat = brotli.compress(raw, quality=11)
    archive = deterministic_archive(coded)
    archive_repeat = deterministic_archive(coded)
    variant_root = ROOT / label
    paths = {
        "raw": variant_root / "ec1_latent_v2.raw",
        "coded": variant_root / "ec1_latent_v2.br",
        "coded_repeat": variant_root / "ec1_latent_v2.repeat.br",
        "archive": variant_root / "archive.zip",
        "archive_repeat": variant_root / "archive.repeat.zip",
    }
    for key, value in (
        ("raw", raw),
        ("coded", coded),
        ("coded_repeat", repeat),
        ("archive", archive),
        ("archive_repeat", archive_repeat),
    ):
        atomic_bytes(paths[key], value)
    if coded != repeat or archive != archive_repeat:
        raise RuntimeError(f"determinism failure for {label}")
    parsed_header, parsed_tensors = parse_raw(brotli.decompress(coded))
    if parsed_header != header or parsed_tensors != tensors:
        raise RuntimeError(f"scorer-free raw parse-back failure for {label}")
    archive_delta = len(archive) - SURROGATE_ARCHIVE_BYTES
    score_per_flip = 100.0 / PIXELS
    score_per_byte = 25.0 / RATE_DENOMINATOR
    break_even = math.ceil(archive_delta * score_per_byte / score_per_flip)
    nameable = math.ceil((archive_delta * score_per_byte + 1e-5) / score_per_flip)
    result = {
        "label": label,
        "classification": "SCORER_FREE_DESIGN_PRICE_SURROGATE",
        "is_candidate": False,
        "receiver_validated": False,
        "source_surrogate": {
            "module": record(SURROGATE_MODULE),
            "archive": record(SURROGATE_ARCHIVE),
        },
        "logical_gate_parameter_count": 36,
        "gate_tensor_bytes_before_header_or_brotli": 40,
        "gate_form": "h_prime=h*(1+tanh(W_g@z+b_g)); W_g:[4,8], b_g:[4]",
        "injection_site": "after EC1 depthwise GELU and before its 96-channel 1x1 head",
        "artifacts": {key: record(path) for key, path in paths.items()},
        "module_delta_bytes_vs_oriented_design_surrogate": len(coded) - len(require(
            SURROGATE_MODULE, SURROGATE_MODULE_SHA256
        )),
        "archive_delta_bytes_vs_oriented_design_surrogate": archive_delta,
        "archive_delta_bytes_vs_cp135": len(archive) - BASE_BYTES,
        "zero_pose_break_even_fixed_flips_for_incremental_gate_price": break_even,
        "zero_pose_fixed_flips_for_delta_score_at_most_minus_1e_5": nameable,
        "deterministic_repeat_equal": True,
        "raw_parseback_equal": True,
    }
    atomic_json(ROOT / f"STAGE_{label}.json", result)
    return result


def main() -> None:
    ROOT.mkdir(parents=True, exist_ok=True)
    manifest_path = ROOT / "PRICE_MANIFEST.json"
    if manifest_path.is_file():
        prior_manifest = json.loads(manifest_path.read_text())
        free = int(prior_manifest["storage_preflight"]["free_bytes_before"])
    else:
        free = shutil.disk_usage(ROOT).free
    if free < 10 * 1024 * 1024:
        raise RuntimeError("storage preflight failed: need at least 10 MiB free")
    require(BASE_ARCHIVE, BASE_SHA256, BASE_BYTES)
    module = require(SURROGATE_MODULE, SURROGATE_MODULE_SHA256)
    require(SURROGATE_ARCHIVE, SURROGATE_ARCHIVE_SHA256, SURROGATE_ARCHIVE_BYTES)
    header, tensors = parse_raw(brotli.decompress(module))
    rng = np.random.default_rng(SEED)
    identity = build_variant(
        "identity_gate",
        header,
        tensors,
        np.zeros((4, 8), dtype=np.int8),
        np.zeros(4, dtype="<f2"),
    )
    capacity = build_variant(
        "seeded_nonzero_capacity_reference",
        header,
        tensors,
        rng.integers(-127, 128, size=(4, 8), dtype=np.int16).astype(np.int8),
        rng.uniform(-0.25, 0.25, size=4).astype("<f2"),
    )
    manifest = {
        "schema": "ddm_bg1_bilinear_gate_price.v1",
        "seed": SEED,
        "classification": "SCORER_FREE_DESIGN_PRICE_ONLY",
        "is_candidate": False,
        "modal_or_metal_used": False,
        "scorer_used": False,
        "receiver_validated": False,
        "storage_preflight": {"free_bytes_before": free, "minimum_free_bytes": 10 * 1024 * 1024},
        "inputs": {
            "base_archive": record(BASE_ARCHIVE),
            "oriented_design_surrogate_module": record(SURROGATE_MODULE),
            "oriented_design_surrogate_archive": record(SURROGATE_ARCHIVE),
        },
        "variants": [identity, capacity],
        "limits": [
            "The trained EC2 module bytes were not locally available, so these are inline-schema deltas on the retained EC1 oriented seeded capacity-reference surrogate.",
            "The current v1 receiver rejects v2; this measures serialization and whole-container price only, not receiver validity.",
            "No endpoint field was read and no mechanism/admission verdict follows from this manifest.",
        ],
        "reproducibility": {
            "argv": [str(Path(__file__).resolve())],
            "script": record(Path(__file__).resolve()),
            "idempotent": True,
            "per_variant_stage_manifests": [
                str(ROOT / "STAGE_identity_gate.json"),
                str(ROOT / "STAGE_seeded_nonzero_capacity_reference.json"),
            ],
        },
    }
    atomic_json(manifest_path, manifest)
    print(json.dumps(manifest, indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
