        raise RuntimeError(f"unexpected PK2 runner SHA-256 before import: {actual_sha256}")

    if spec is None or spec.loader is None:

def utcnow() -> str:
    return datetime.now(UTC).isoformat().replace("+00:00", "Z")


def sha256_bytes(payload: bytes) -> str:
    return hashlib.sha256(payload).hexdigest()


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1 << 20), b""):
            digest.update(chunk)
    return digest.hexdigest()


def array_sha256(value: np.ndarray) -> str:
    return sha256_bytes(np.ascontiguousarray(value).tobytes())


def canonical_json(value: Any) -> str:
    return json.dumps(value, sort_keys=True, separators=(",", ":"))


def atomic_write_bytes(path: Path, payload: bytes) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_suffix(path.suffix + ".tmp")
    temporary.write_bytes(payload)
    os.replace(temporary, path)


def atomic_write_text(path: Path, text: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_suffix(path.suffix + ".tmp")
    temporary.write_text(text, encoding="utf-8")
    os.replace(temporary, path)


def atomic_json(path: Path, value: Any) -> None:
    atomic_write_text(path, json.dumps(value, indent=2, sort_keys=True) + "\n")


def validate_out_dir(path: Path) -> Path:
    resolved = path.resolve()
    allowed = Path("/Volumes/VertigoDataTier/pact")
    if resolved == allowed or allowed not in resolved.parents:
        raise ValueError("PK3 evidence must use an arm-specific VertigoDataTier directory")
    if resolved == Path("/tmp") or Path("/tmp") in resolved.parents:
        raise ValueError("PK3 persisted evidence may not use /tmp")
    return resolved


def storage_preflight(path: Path, required_free_bytes: int) -> dict[str, Any]:
    path.mkdir(parents=True, exist_ok=True)
    stats = os.statvfs(path)
    free_bytes = int(stats.f_bavail * stats.f_frsize)
    result = {
        "path": str(path),
        "free_bytes": free_bytes,
        "required_free_bytes": int(required_free_bytes),
        "passed": free_bytes >= required_free_bytes,
        "measured_at_utc": utcnow(),
    }
    if not result["passed"]:
        raise RuntimeError(f"storage preflight failed at {path}: {free_bytes} < {required_free_bytes}")
    return result


def copy_pinned_receiver_inputs(retained_root: Path, archive_path: Path) -> dict[str, dict[str, Any]]:
    source_root = retained_root / "inputs" / f"pr130_runtime_{EXPECTED_INTAKE_COMMIT[:12]}"
    copied: dict[str, dict[str, Any]] = {}
    for name, expected_sha256 in EXPECTED_RUNTIME_SHA256.items():
        source = pk2.PR130_CODE / name
        payload = source.read_bytes()
        actual_sha256 = sha256_bytes(payload)
        if actual_sha256 != expected_sha256:
            raise RuntimeError(f"unexpected pinned receiver source SHA-256 for {source}: {actual_sha256}")
        destination = source_root / name
        if destination.exists() and sha256_file(destination) != expected_sha256:
            raise RuntimeError(f"refusing drifted existing receiver copy at {destination}")
        if not destination.exists():
            atomic_write_bytes(destination, payload)
        if sha256_file(destination) != expected_sha256:
            raise RuntimeError(f"copied receiver source drift at {destination}")
        copied[name] = {
            "intake_path": str(source),
            "copied_path": str(destination),
            "bytes": destination.stat().st_size,
            "sha256": expected_sha256,
        }
    archive_payload = archive_path.read_bytes()
    archive_sha256 = sha256_bytes(archive_payload)
    if archive_sha256 != EXPECTED_ARCHIVE_SHA256:
        raise RuntimeError(f"unexpected baseline archive SHA-256: {archive_sha256}")
    archive_copy = retained_root / "inputs" / "archive.zip"
    if archive_copy.exists() and sha256_file(archive_copy) != archive_sha256:
        raise RuntimeError(f"refusing drifted existing archive copy at {archive_copy}")
    if not archive_copy.exists():
        atomic_write_bytes(archive_copy, archive_payload)
    copied["archive.zip"] = {
        "intake_path": str(archive_path),
        "copied_path": str(archive_copy),
        "bytes": archive_copy.stat().st_size,
        "sha256": archive_sha256,
    }
    runner_source = Path(__file__).resolve()
    runner_payload = runner_source.read_bytes()
    runner_sha256 = sha256_bytes(runner_payload)
    if runner_sha256 != RUNNER_SOURCE_SHA256_AT_IMPORT:
        raise RuntimeError("PK3 runner source changed after module import")
    runner_copy = retained_root / "inputs" / "pk3_runner.py"
    if runner_copy.exists() and sha256_file(runner_copy) != runner_sha256:
        raise RuntimeError(f"refusing drifted existing PK3 runner copy at {runner_copy}")
    if not runner_copy.exists():

        "intake_path": str(runner_source),

    )

def load_copied_receiver_sources(retained_root: Path) -> tuple[Any, Any]:
    source_root = retained_root / "inputs" / f"pr130_runtime_{EXPECTED_INTAKE_COMMIT[:12]}"
    if str(source_root) not in sys.path:
        sys.path.insert(0, str(source_root))

    codec_path = source_root / "carrier_codec.py"
    codec_spec = importlib.util.spec_from_file_location("carrier_codec", codec_path)
    if codec_spec is None or codec_spec.loader is None:
        raise RuntimeError(f"cannot load copied carrier codec from {codec_path}")
    codec = importlib.util.module_from_spec(codec_spec)
    sys.modules[codec_spec.name] = codec
    codec_spec.loader.exec_module(codec)

    inflate_path = source_root / "inflate.py"
    inflate_spec = importlib.util.spec_from_file_location("ddm_pk3_copied_inflate", inflate_path)
    if inflate_spec is None or inflate_spec.loader is None:
        raise RuntimeError(f"cannot load copied inflate runtime from {inflate_path}")
    inflate = importlib.util.module_from_spec(inflate_spec)
    sys.modules[inflate_spec.name] = inflate
    inflate_spec.loader.exec_module(inflate)
    loaded_paths = {
        "carrier_codec.py": Path(codec.__file__).resolve(),
        "inflate.py": Path(inflate.__file__).resolve(),
    }
    for module_name in ("hpac_integer", "hpac_integer_sparse", "integer_model_io"):
        module = sys.modules.get(module_name)
        if module is None or not getattr(module, "__file__", None):
            raise RuntimeError(f"copied inflate did not load required module {module_name}")
        loaded_paths[f"{module_name}.py"] = Path(module.__file__).resolve()
    for name, loaded_path in loaded_paths.items():
        expected_path = (source_root / name).resolve()
        if loaded_path != expected_path:
            raise RuntimeError(f"receiver module escaped copied runtime: {loaded_path} != {expected_path}")
        if sha256_file(loaded_path) != EXPECTED_RUNTIME_SHA256[name]:
            raise RuntimeError(f"loaded copied receiver source drift at {loaded_path}")
    return codec, inflate


@dataclass(frozen=True)
class GaugeState:
    candidate_id: str
    sequence: tuple[dict[str, Any], ...]
    basis_codes: np.ndarray
    basis_scales: np.ndarray
    coefficient_codes: np.ndarray
    coefficient_scales: np.ndarray

    @property
    def basis(self) -> np.ndarray:
        return self.basis_codes.astype(np.float32) * self.basis_scales[:, None, None, None]

    @property
    def coefficients(self) -> np.ndarray:
        return self.coefficient_codes.astype(np.float32) * self.coefficient_scales[None]


def normalized_basis_tensor(raw_basis: np.ndarray, inflate: Any) -> torch.Tensor:
    if tuple(raw_basis.shape) != pk2.BASIS_SHAPE:
        raise ValueError(f"raw basis shape must be {pk2.BASIS_SHAPE}")
    with torch.inference_mode():
        tensor = torch.from_numpy(np.ascontiguousarray(raw_basis, dtype=np.float32))
        normalized = inflate.normalized_basis(tensor).detach().cpu().contiguous()
    expected_shape = (pk2.DIM, 3, int(inflate.EVAL_H), int(inflate.EVAL_W))
    if tuple(normalized.shape) != expected_shape or not torch.isfinite(normalized).all():
        raise RuntimeError("pinned receiver produced an invalid normalized basis")
    return normalized


def centered_bicubic_rms(raw_basis: np.ndarray, inflate: Any) -> np.ndarray:
    if tuple(raw_basis.shape) != pk2.BASIS_SHAPE:
        raise ValueError(f"raw basis shape must be {pk2.BASIS_SHAPE}")
    with torch.inference_mode():
        tensor = torch.from_numpy(np.ascontiguousarray(raw_basis, dtype=np.float32))
        upsampled = F.interpolate(
            tensor,
            size=(int(inflate.EVAL_H), int(inflate.EVAL_W)),
            mode="bicubic",
            align_corners=False,
        )
        centered = upsampled - upsampled.mean(dim=(1, 2, 3), keepdim=True)
        rms = centered.square().mean(dim=(1, 2, 3)).sqrt().clamp_min(1e-5)
    result = rms.detach().cpu().numpy().astype(np.float64)
    if result.shape != (pk2.DIM,) or not np.isfinite(result).all():
        raise RuntimeError("pinned receiver produced invalid basis RMS values")
    return result


def receiver_product_mse_contraction(
    basis_a: np.ndarray,
    coefficients_a: np.ndarray,
    basis_b: np.ndarray,
    coefficients_b: np.ndarray,
    inflate: Any,
    *,
    normalized_basis_b: torch.Tensor | None = None,
) -> float:
    normalized_a = normalized_basis_tensor(basis_a, inflate)
    normalized_b = normalized_basis_tensor(basis_b, inflate) if normalized_basis_b is None else normalized_basis_b
    a_basis = np.ascontiguousarray(normalized_a.numpy().reshape(pk2.DIM, -1), dtype=np.float64)
    b_basis = np.ascontiguousarray(normalized_b.numpy().reshape(pk2.DIM, -1), dtype=np.float64)
    a_coeff = np.ascontiguousarray(coefficients_a, dtype=np.float64)
    b_coeff = np.ascontiguousarray(coefficients_b, dtype=np.float64)
    if a_coeff.shape != (pk2.N, pk2.DIM) or b_coeff.shape != (pk2.N, pk2.DIM):
        raise ValueError("coefficient arrays must cover the full 600x12 population")
    a_coeff_gram = np.einsum("ni,nj->ij", a_coeff, a_coeff, optimize=False)
    b_coeff_gram = np.einsum("ni,nj->ij", b_coeff, b_coeff, optimize=False)
    cross_coeff_gram = np.einsum("ni,nj->ij", b_coeff, a_coeff, optimize=False)
    a_basis_gram = np.einsum("ip,jp->ij", a_basis, a_basis, optimize=False)
    b_basis_gram = np.einsum("ip,jp->ij", b_basis, b_basis, optimize=False)
    cross_basis_gram = np.einsum("ip,jp->ij", a_basis, b_basis, optimize=False)
    a_self = np.einsum("ij,ji->", a_coeff_gram, a_basis_gram, optimize=False)
    b_self = np.einsum("ij,ji->", b_coeff_gram, b_basis_gram, optimize=False)
    cross = np.einsum("ij,ji->", cross_coeff_gram, cross_basis_gram, optimize=False)
    if not np.isfinite([a_self, b_self, cross]).all():
        raise RuntimeError("receiver-product contraction produced a non-finite term")
    sse = float(a_self + b_self - 2.0 * cross) / pk2.DIM
    scale = max(abs(float(a_self)), abs(float(b_self)), abs(float(cross)), 1.0) / pk2.DIM
    tolerance = 1e-11 * scale
    if sse < -tolerance:
        raise RuntimeError(f"receiver-product contraction produced negative SSE {sse}")
    sse = max(0.0, sse)
    return sse / (pk2.N * a_basis.shape[1])


def receiver_product_mse_explicit(
    basis_a: np.ndarray,
    coefficients_a: np.ndarray,
    basis_b: np.ndarray,
    coefficients_b: np.ndarray,
    inflate: Any,
    *,
    batch_size: int = 8,
) -> float:
    if batch_size < 1:
        raise ValueError("explicit MSE batch size must be positive")
    normalized_a = normalized_basis_tensor(basis_a, inflate)
    normalized_b = normalized_basis_tensor(basis_b, inflate)
    coefficients_a_tensor = torch.from_numpy(np.ascontiguousarray(coefficients_a, dtype=np.float32))
    coefficients_b_tensor = torch.from_numpy(np.ascontiguousarray(coefficients_b, dtype=np.float32))
    sse = 0.0
    count = 0
    with torch.inference_mode():
        for start in range(0, pk2.N, batch_size):
            end = min(start + batch_size, pk2.N)
            product_a = torch.einsum("bk,kchw->bchw", coefficients_a_tensor[start:end], normalized_a) / math.sqrt(
                pk2.DIM
            )
            product_b = torch.einsum("bk,kchw->bchw", coefficients_b_tensor[start:end], normalized_b) / math.sqrt(
                pk2.DIM
            )
            delta = product_a - product_b
            sse += float(delta.double().square().sum().item())
            count += int(delta.numel())
    if count != pk2.N * 3 * int(inflate.EVAL_H) * int(inflate.EVAL_W):
        raise RuntimeError("explicit receiver-product MSE did not cover the full population")
    return sse / count


def sequence_is_exact_receiver_gauge(sequence: tuple[dict[str, Any], ...]) -> bool:
    return all(operation.get("kind") in EXACT_RECEIVER_GAUGE_KINDS for operation in sequence)


def baseline_state(bundle: Any, codec: Any) -> GaugeState:
    decoded = codec.decode_compact_carrier(
        bundle.carrier,
        basis_count=math.prod(pk2.BASIS_SHAPE),
        frames=pk2.N,
        dimensions=pk2.DIM,
    )
    return GaugeState(
        candidate_id="source_cpr1",
        sequence=(),
        basis_codes=decoded[1].reshape(pk2.BASIS_SHAPE).astype(np.int32),
        basis_scales=decoded[0].astype("<f4"),
        coefficient_codes=pk2.decode_absolute_codes(decoded[3]),
        coefficient_scales=decoded[2].astype("<f4"),
    )


def state_from_carrier(
    carrier_path: Path,

    decoded = codec.decode_compact_carrier(

def signed_state(parent: GaugeState, signs: np.ndarray) -> GaugeState:
    signs = np.asarray(signs, dtype=np.int32)
    if signs.shape != (pk2.DIM,) or not np.isin(signs, (-1, 1)).all():
        raise ValueError("sign gauge must contain exactly twelve +/-1 entries")
    operation = {"kind": "exact_sign", "signs": signs.tolist()}
    return GaugeState(
        candidate_id=parent.candidate_id,
        sequence=(*parent.sequence, operation),
        basis_codes=parent.basis_codes * signs[:, None, None, None],
        basis_scales=parent.basis_scales.copy(),
        coefficient_codes=parent.coefficient_codes * signs[None],
        coefficient_scales=parent.coefficient_scales.copy(),
    )


def permuted_state(parent: GaugeState, permutation: np.ndarray) -> GaugeState:
    permutation = np.asarray(permutation, dtype=np.int32)
    if permutation.shape != (pk2.DIM,) or sorted(permutation.tolist()) != list(range(pk2.DIM)):
        raise ValueError("permutation gauge must be a bijection over twelve dimensions")
    operation = {"kind": "exact_permutation", "permutation": permutation.tolist()}
    return GaugeState(
        candidate_id=parent.candidate_id,
        sequence=(*parent.sequence, operation),
        basis_codes=parent.basis_codes[permutation],
        basis_scales=parent.basis_scales[permutation],
        coefficient_codes=parent.coefficient_codes[:, permutation],
        coefficient_scales=parent.coefficient_scales[permutation],
    )


def scale_state(parent: GaugeState, dimension: int, exponent: int) -> GaugeState:
    if dimension not in range(pk2.DIM) or exponent == 0:
        raise ValueError("invalid scale-gauge coordinate")
    factors = np.ones(pk2.DIM, dtype=np.float64)
    factors[dimension] = math.ldexp(1.0, exponent)
    return radial_scale_state(parent, factors, label=f"dimension_{dimension}_exp_{exponent}")


def radial_scale_state(
    parent: GaugeState,
    factors: np.ndarray,
    *,
    label: str,
) -> GaugeState:
    factors = np.asarray(factors, dtype=np.float64)
    if factors.shape != (pk2.DIM,) or not np.isfinite(factors).all() or np.any(factors <= 0.0):
        raise ValueError("receiver-null radial factors must be twelve finite positive values")
    basis_scales = (parent.basis_scales.astype(np.float64) * factors).astype("<f4")
    if not np.isfinite(basis_scales).all() or np.any(basis_scales <= 0.0):
        raise ValueError("radial gauge produced an invalid basis scale")
    operation = {
        "kind": "decoder_dead_radial_scale",
        "label": label,
        "factors": factors.tolist(),
    }
    return GaugeState(
        candidate_id=parent.candidate_id,
        sequence=(*parent.sequence, operation),
        basis_codes=parent.basis_codes.copy(),
        basis_scales=basis_scales,
        coefficient_codes=parent.coefficient_codes.copy(),
        coefficient_scales=parent.coefficient_scales.copy(),
    )


def dc_shifted_state(parent: GaugeState, shifts: np.ndarray) -> GaugeState:
    shifts = np.asarray(shifts, dtype=np.int32)
    if shifts.shape != (pk2.DIM,) or not np.any(shifts):
        raise ValueError("DC shift gauge must contain twelve integers and change at least one row")
    basis_codes = parent.basis_codes + shifts[:, None, None, None]
    if basis_codes.min() < -16 or basis_codes.max() > 15:
        raise ValueError("DC shift gauge exceeds the signed-int5 code range")
    operation = {"kind": "exact_dc_shift", "shifts": shifts.tolist()}
    return GaugeState(
        candidate_id=parent.candidate_id,
        sequence=(*parent.sequence, operation),
        basis_codes=basis_codes.astype(np.int32),
        basis_scales=parent.basis_scales.copy(),
        coefficient_codes=parent.coefficient_codes.copy(),
        coefficient_scales=parent.coefficient_scales.copy(),
    )


def transformed_state(
    parent: GaugeState,
    matrix: np.ndarray,
    operation: dict[str, Any],
    inflate: Any,
) -> GaugeState:
    matrix = np.asarray(matrix, dtype=np.float64)
    if matrix.shape != (pk2.DIM, pk2.DIM):
        raise ValueError("gauge matrix must be 12x12")
    condition = float(np.linalg.cond(matrix))
    if not math.isfinite(condition) or condition > 4.0:
        raise ValueError(f"gauge matrix is ill-conditioned: {condition}")
    inverse = np.linalg.inv(matrix)
    parent_rms = centered_bicubic_rms(parent.basis, inflate)
    canonical_basis = parent.basis.astype(np.float64) / parent_rms[:, None, None, None]
    basis = np.einsum("ij,jchw->ichw", matrix, canonical_basis, optimize=False)
    mixed_rms = centered_bicubic_rms(basis.astype(np.float32), inflate)
    if np.any(mixed_rms <= 1e-5):
        raise ValueError("normalization-aware gauge produced a degenerate basis row")
    coefficients = (
        np.einsum(
            "nd,dk->nk",
            parent.coefficients.astype(np.float64),
            inverse,
            optimize=False,
        )
        * mixed_rms[None]
    )
    basis_restored, basis_codes, basis_scales = pk2.quantize_basis(basis, 5)
    coeff_restored, coefficient_codes, coefficient_scales = pk2.quantize_coefficients(coefficients)
    if basis_restored.shape != pk2.BASIS_SHAPE or coeff_restored.shape != (
        pk2.N,
        pk2.DIM,
    ):
        raise RuntimeError("quantized gauge produced the wrong full-population shape")
    operation = {
        **operation,
        "condition_number": condition,
        "matrix_sha256": array_sha256(matrix),
        "gauge_law": "A'=H*diag(r(A)^-1)*A; C'=C*H^-1*diag(r(A'))",
        "parent_centered_bicubic_rms": parent_rms.tolist(),
        "prequant_mixed_centered_bicubic_rms": mixed_rms.tolist(),
    }
    return GaugeState(
        candidate_id=parent.candidate_id,
        sequence=(*parent.sequence, operation),
        basis_codes=basis_codes,
        basis_scales=basis_scales,
        coefficient_codes=coefficient_codes,
        coefficient_scales=coefficient_scales,
    )


def givens_matrix(first: int, second: int, angle: float) -> np.ndarray:
    if not 0 <= first < second < pk2.DIM or angle == 0.0:
        raise ValueError("invalid Givens coordinate")
    matrix = np.eye(pk2.DIM, dtype=np.float64)
    cosine = math.cos(angle)
    sine = math.sin(angle)
    matrix[first, first] = cosine
    matrix[first, second] = -sine
    matrix[second, first] = sine
    matrix[second, second] = cosine
    return matrix


def shear_matrix(source: int, destination: int, amount: float) -> np.ndarray:
    if source == destination or source not in range(pk2.DIM) or destination not in range(pk2.DIM):
        raise ValueError("invalid shear coordinate")
    if amount == 0.0:
        raise ValueError("shear amount must be nonzero")
    matrix = np.eye(pk2.DIM, dtype=np.float64)
    matrix[source, destination] = amount
    return matrix


def candidate_identity(stage: str, parent: GaugeState, state: GaugeState) -> str:
    payload = {
        "receiver_metric_version": RECEIVER_METRIC_VERSION,
        "transform_law_version": TRANSFORM_LAW_VERSION,
        "baseline_archive_sha256": EXPECTED_ARCHIVE_SHA256,
        "runtime_sha256": EXPECTED_RUNTIME_SHA256,
        "pk2_runner_sha256": EXPECTED_PK2_RUNNER_SHA256,
        "pk3_runner_sha256": RUNNER_SOURCE_SHA256_AT_IMPORT,
        "stage": stage,
        "parent_candidate_id": parent.candidate_id,
        "basis_codes_sha256": array_sha256(state.basis_codes),
        "basis_scales_sha256": array_sha256(state.basis_scales),
        "coefficient_codes_sha256": array_sha256(state.coefficient_codes),
        "coefficient_scales_sha256": array_sha256(state.coefficient_scales),
        "last_operation": state.sequence[-1] if state.sequence else {"kind": "identity"},
    }
    return f"{stage}_{sha256_bytes(canonical_json(payload).encode())[:20]}"


def verify_retained_candidate(
    receipt_path: Path,
    expected_identity: dict[str, Any],
) -> dict[str, Any] | None:
    if not receipt_path.is_file():
        return None
    receipt = json.loads(receipt_path.read_text(encoding="utf-8"))
    if receipt.get("schema") != "ddm_pk3_rate_aware_gauge_candidate.v3":
        raise RuntimeError(f"refusing stale candidate receipt schema at {receipt_path}")
    if receipt.get("identity") != expected_identity:
        raise RuntimeError(f"candidate identity drift at {receipt_path}")
    for key in ("carrier", "archive"):
        item = receipt["retained_payloads"][key]
        path = Path(item["path"])
        if not path.is_file() or path.stat().st_size != item["bytes"] or sha256_file(path) != item["sha256"]:
            raise RuntimeError(f"retained {key} drift at {path}")
    return receipt


def materialize_candidate(
    *,
    stage: str,
    parent: GaugeState,
    state: GaugeState,
    bundle: Any,
    codec: Any,
    inflate: Any,
    baseline: GaugeState,
    baseline_normalized_basis: torch.Tensor,
    retained_root: Path,
) -> dict[str, Any]:
    candidate_id = candidate_identity(stage, parent, state)
    candidate_dir = retained_root / "candidates" / candidate_id
    identity = {
        "candidate_id": candidate_id,
        "receiver_metric_version": RECEIVER_METRIC_VERSION,
        "transform_law_version": TRANSFORM_LAW_VERSION,
        "baseline_archive_sha256": EXPECTED_ARCHIVE_SHA256,
        "runtime_sha256": EXPECTED_RUNTIME_SHA256,
        "pk2_runner_sha256": EXPECTED_PK2_RUNNER_SHA256,
        "pk3_runner_sha256": RUNNER_SOURCE_SHA256_AT_IMPORT,
        "stage": stage,
        "parent_candidate_id": parent.candidate_id,
        "sequence": list(state.sequence),
        "basis_codes_sha256": array_sha256(state.basis_codes),
        "basis_scales_sha256": array_sha256(state.basis_scales),
        "coefficient_codes_sha256": array_sha256(state.coefficient_codes),
        "coefficient_scales_sha256": array_sha256(state.coefficient_scales),
    }
    receipt_path = candidate_dir / "candidate.json"
    prior = verify_retained_candidate(receipt_path, identity)
    if prior is not None:
        return prior

    carrier = codec.encode_compact_carrier(
        state.basis_scales,
        state.basis_codes.reshape(-1),
        state.coefficient_scales,
        pk2.encode_absolute_codes(state.coefficient_codes),
    )
    carrier_path = candidate_dir / "carrier.cpr1"
    atomic_write_bytes(carrier_path, carrier)
    archive = pk2.replace_carrier(bundle, carrier)
    archive_path = candidate_dir / "archive.zip"
    atomic_write_bytes(archive_path, archive)

    parsed = pk2.extract_bundle(archive_path)
    if (
        parsed.carrier != carrier
        or parsed.semantic != bundle.semantic
        or parsed.hpac != bundle.hpac
        or parsed.tokens != bundle.tokens
    ):
        raise RuntimeError(f"{candidate_id}: full archive parse-back mismatch")
    decoded = codec.decode_compact_carrier(
        parsed.carrier,
        basis_count=math.prod(pk2.BASIS_SHAPE),
        frames=pk2.N,
        dimensions=pk2.DIM,
    )
    decoded_basis_codes = decoded[1].reshape(pk2.BASIS_SHAPE).astype(np.int32)
    decoded_coefficient_codes = pk2.decode_absolute_codes(decoded[3])
    if not np.array_equal(decoded_basis_codes, state.basis_codes):
        raise RuntimeError(f"{candidate_id}: Huffman parse-back changed basis codes")
    if not np.array_equal(decoded_coefficient_codes, state.coefficient_codes):
        raise RuntimeError(f"{candidate_id}: Rice parse-back changed coefficient codes")
    semantic_pose = np.asarray([len(bundle.semantic), len(carrier)], dtype="<u4").tobytes() + bundle.semantic + carrier
    _, inflate_basis, inflate_coefficients = inflate.unpack_semantic_pose(semantic_pose)
    decoded_basis = decoded_basis_codes.astype(np.float32) * decoded[0][:, None, None, None]
    decoded_coefficients = decoded_coefficient_codes.astype(np.float32) * decoded[2][None]
    if not np.array_equal(inflate_basis.numpy(), decoded_basis):
        raise RuntimeError(f"{candidate_id}: real inflate parser changed basis")
    if not np.array_equal(inflate_coefficients.numpy(), decoded_coefficients):
        raise RuntimeError(f"{candidate_id}: real inflate parser changed coefficients")
    _, basis_bit_count, coefficient_bit_count = codec.HEADER.unpack_from(carrier)
    raw_stored_factor_mse = pk2.carrier_product_mse(
        decoded_basis,
        decoded_coefficients,
        baseline.basis,
        baseline.coefficients,
    )
    if sequence_is_exact_receiver_gauge(state.sequence):
        receiver_mse = 0.0
        receiver_mse_method = "derived_exact_receiver_gauge_screen_pending_explicit_validation"
    else:
        receiver_mse = receiver_product_mse_contraction(
            decoded_basis,
            decoded_coefficients,
            baseline.basis,
            baseline.coefficients,
            inflate,
            normalized_basis_b=baseline_normalized_basis,
        )
        receiver_mse_method = "full_population_float64_gram_contraction_of_pinned_fp32_normalized_basis"
    receipt = {
        "schema": "ddm_pk3_rate_aware_gauge_candidate.v3",
        "created_at_utc": utcnow(),
        "axis": AXIS,
        "score_claim": SCORE_CLAIM,
        "identity": identity,
        "population": {
            "basis_shape": list(pk2.BASIS_SHAPE),
            "basis_values": int(math.prod(pk2.BASIS_SHAPE)),
            "coefficient_shape": [pk2.N, pk2.DIM],
            "coefficient_values": pk2.N * pk2.DIM,
            "selection_mode": "full_population_no_slice",
        },
        "real_coder": {
            "magic": carrier[:4].decode("ascii"),
            "basis_coder": "pinned canonical Huffman",
            "basis_bit_count": int(basis_bit_count),
            "coefficient_coder": "pinned per-column optimal-k Rice",
            "coefficient_bit_count": int(coefficient_bit_count),
            "receiver_parse_back": True,
            "real_inflate_unpack": True,
        },
        "receiver_carrier_product_mse": float(receiver_mse),
        "receiver_carrier_product_mse_method": receiver_mse_method,
        "receiver_carrier_product_mse_explicit": None,
        "receiver_carrier_product_mse_explicit_population_values": None,
        "raw_stored_factor_product_mse_non_authority": float(raw_stored_factor_mse),
        "mse_bar": MSE_BAR,
        "mse_bar_strict_screen_pass": bool(receiver_mse < MSE_BAR),
        "mse_bar_strict_explicit_pass": None,
        "retained_payloads": {
            "carrier": {
                "path": str(carrier_path),
                "bytes": carrier_path.stat().st_size,
                "sha256": sha256_file(carrier_path),
            },
            "archive": {
                "path": str(archive_path),
                "bytes": archive_path.stat().st_size,
                "sha256": sha256_file(archive_path),
            },
        },
        "decoded_arrays": {
            "basis_sha256": array_sha256(decoded_basis),
            "coefficients_sha256": array_sha256(decoded_coefficients),
        },
    }
    atomic_json(receipt_path, receipt)
    return receipt


def explicitly_validate_candidate(
    receipt: dict[str, Any],
    codec: Any,
    inflate: Any,
    baseline: GaugeState,
) -> dict[str, Any]:
    carrier_path = Path(receipt["retained_payloads"]["carrier"]["path"])
    state = state_from_carrier(
        carrier_path,
        codec,
        receipt["identity"]["candidate_id"],
        tuple(receipt["identity"]["sequence"]),
    )
    explicit_mse = receiver_product_mse_explicit(
        state.basis,
        state.coefficients,
        baseline.basis,
        baseline.coefficients,
        inflate,
    )
    screen_mse = float(receipt["receiver_carrier_product_mse"])
    if "gram_contraction" in receipt["receiver_carrier_product_mse_method"]:
        tolerance = max(1e-12, 2e-6 * max(explicit_mse, screen_mse, 1e-12))
        if abs(explicit_mse - screen_mse) > tolerance:
            raise RuntimeError(
                "explicit receiver-product MSE disagrees with full-population contraction: "
                f"{explicit_mse} vs {screen_mse}"
            )
    receipt = {
        **receipt,
        "receiver_carrier_product_mse_explicit": float(explicit_mse),
        "receiver_carrier_product_mse_explicit_population_values": (
            pk2.N * 3 * int(inflate.EVAL_H) * int(inflate.EVAL_W)
        ),
        "mse_bar_strict_explicit_pass": bool(explicit_mse < MSE_BAR),
        "explicit_validation": {
            "method": "chunked_direct_fp32_C_at_normalized_basis_div_sqrt12_float64_SSE",
            "selection_mode": "full_population_no_slice",
            "basis_shape_after_normalization": [pk2.DIM, 3, int(inflate.EVAL_H), int(inflate.EVAL_W)],
            "coefficient_shape": [pk2.N, pk2.DIM],
            "validated_at_utc": utcnow(),
        },
    }
    receipt_path = carrier_path.parent / "candidate.json"
    atomic_json(receipt_path, receipt)
    return receipt


def receipt_row(receipt: dict[str, Any], baseline_archive_bytes: int) -> dict[str, Any]:
    archive_bytes = int(receipt["retained_payloads"]["archive"]["bytes"])
    explicit_mse = receipt.get("receiver_carrier_product_mse_explicit")
    return {
        "candidate_id": receipt["identity"]["candidate_id"],
        "stage": receipt["identity"]["stage"],
        "parent_candidate_id": receipt["identity"]["parent_candidate_id"],
        "last_operation": receipt["identity"]["sequence"][-1]
        if receipt["identity"]["sequence"]
        else {"kind": "identity"},
        "archive_bytes": archive_bytes,
        "archive_bytes_saved": baseline_archive_bytes - archive_bytes,
        "archive_sha256": receipt["retained_payloads"]["archive"]["sha256"],
        "archive_path": receipt["retained_payloads"]["archive"]["path"],
        "carrier_bytes": int(receipt["retained_payloads"]["carrier"]["bytes"]),
        "carrier_sha256": receipt["retained_payloads"]["carrier"]["sha256"],
        "carrier_path": receipt["retained_payloads"]["carrier"]["path"],
        "basis_bit_count": int(receipt["real_coder"]["basis_bit_count"]),
        "coefficient_bit_count": int(receipt["real_coder"]["coefficient_bit_count"]),
        "receiver_carrier_product_mse": float(receipt["receiver_carrier_product_mse"]),
        "receiver_carrier_product_mse_method": receipt["receiver_carrier_product_mse_method"],
        "receiver_carrier_product_mse_explicit": (float(explicit_mse) if explicit_mse is not None else None),
        "raw_stored_factor_product_mse_non_authority": float(receipt["raw_stored_factor_product_mse_non_authority"]),
        "mse_bar_strict_screen_pass": bool(receipt["mse_bar_strict_screen_pass"]),
        "mse_bar_strict_explicit_pass": receipt.get("mse_bar_strict_explicit_pass"),
        "explicitly_validated": explicit_mse is not None,
        "trigger_pass": bool(
            baseline_archive_bytes - archive_bytes >= SAVING_BAR_BYTES
            and explicit_mse is not None
            and float(explicit_mse) < MSE_BAR
        ),
    }


def select_best(rows: list[dict[str, Any]]) -> dict[str, Any]:
    admissible = [row for row in rows if row["mse_bar_strict_screen_pass"]]
    if not admissible:
        raise RuntimeError("gauge stage produced no MSE-admissible candidate")
    return min(
        admissible,
        key=lambda row: (
            row["archive_bytes"],
            row["receiver_carrier_product_mse"],
            row["candidate_id"],
        ),
    )


def load_selected_state(
    row: dict[str, Any],
    codec: Any,
    sequence: tuple[dict[str, Any], ...],
) -> GaugeState:
    return state_from_carrier(
        Path(row["carrier_path"]),
        codec,
        row["candidate_id"],
        sequence,
    )


def stage_checkpoint(
    out: Path,
    stage: str,
    rows: list[dict[str, Any]],
    selected: dict[str, Any],
) -> None:
    atomic_json(
        out / "checkpoints" / f"stage_{stage}.json",
        {
            "schema": "ddm_pk3_rate_aware_gauge_stage.v1",
            "stage": stage,
            "completed_at_utc": utcnow(),
            "candidate_count": len(rows),
            "selected": selected,
        },
    )


def run(args: argparse.Namespace) -> int:
    out = validate_out_dir(args.out_dir)
    retained = out / "retained"
    preflight = storage_preflight(out, args.required_free_bytes)
    archive_sha = sha256_file(args.archive)
    if archive_sha != EXPECTED_ARCHIVE_SHA256:
        raise RuntimeError(f"unexpected baseline archive SHA-256: {archive_sha}")
    intake_commit = subprocess.check_output(["git", "-C", str(pk2.PR130), "rev-parse", "HEAD"], text=True).strip()
    if intake_commit != EXPECTED_INTAKE_COMMIT:
        raise RuntimeError(f"unexpected PR130 intake commit: {intake_commit}")
    pk2_commit = subprocess.check_output(
        ["git", "log", "-1", "--format=%H", "--", str(PK2_PATH.relative_to(REPO))],
        cwd=REPO,
        text=True,
    ).strip()
    if pk2_commit != EXPECTED_PK2_COMMIT:
        raise RuntimeError(f"unexpected PK2 reuse commit: {pk2_commit}")
    copied_inputs = copy_pinned_receiver_inputs(retained, args.archive)
    codec, inflate = load_copied_receiver_sources(retained)
    copied_archive = Path(copied_inputs["archive.zip"]["copied_path"])
    bundle = pk2.extract_bundle(copied_archive)
    baseline = baseline_state(bundle, codec)
    baseline_normalized_basis = normalized_basis_tensor(baseline.basis, inflate)
    baseline_archive_bytes = len(bundle.archive_blob)
    all_rows: list[dict[str, Any]] = []

    control_receipt = materialize_candidate(
        stage="control",
        parent=baseline,
        state=baseline,
        bundle=bundle,
        codec=codec,
        inflate=inflate,
        baseline=baseline,
        baseline_normalized_basis=baseline_normalized_basis,
        retained_root=retained,
    )
    control_receipt = explicitly_validate_candidate(control_receipt, codec, inflate, baseline)
    control_row = receipt_row(control_receipt, baseline_archive_bytes)
    all_rows.append(control_row)
    if control_row["archive_sha256"] != EXPECTED_ARCHIVE_SHA256:
        raise RuntimeError("identity gauge did not reproduce the exact source archive")
    if (
        control_row["carrier_bytes"] != 23_054
        or control_row["basis_bit_count"] != 104_135
        or control_row["coefficient_bit_count"] != 79_076
        or control_row["receiver_carrier_product_mse_explicit"] != 0.0
    ):
        raise RuntimeError("identity control failed frozen carrier anatomy or receiver-MSE sentinels")
    current = GaugeState(
        candidate_id=control_row["candidate_id"],
        sequence=(),
        basis_codes=baseline.basis_codes,
        basis_scales=baseline.basis_scales,
        coefficient_codes=baseline.coefficient_codes,
        coefficient_scales=baseline.coefficient_scales,
    )

    def state_key(state: GaugeState) -> tuple[str, str, str, str]:
        return (
            array_sha256(state.basis_codes),
            array_sha256(state.basis_scales),
            array_sha256(state.coefficient_codes),
            array_sha256(state.coefficient_scales),
        )

    seen_states = {state_key(baseline)}

    def materialize_unique(stage: str, parent: GaugeState, state: GaugeState) -> dict[str, Any] | None:
        key = state_key(state)
        if key in seen_states:
            return None
        seen_states.add(key)
        receipt = materialize_candidate(
            stage=stage,
            parent=parent,
            state=state,
            bundle=bundle,
            codec=codec,
            inflate=inflate,
            baseline=baseline,
            baseline_normalized_basis=baseline_normalized_basis,
            retained_root=retained,
        )
        row = receipt_row(receipt, baseline_archive_bytes)
        all_rows.append(row)
        if len(all_rows) > args.max_candidates:
            raise RuntimeError(f"declared candidate bank exceeded {args.max_candidates} rows")
        return row

    def advance_stage(
        stage: str,
        parent: GaugeState,
        parent_row: dict[str, Any],
        rows: list[dict[str, Any]],
    ) -> tuple[GaugeState, dict[str, Any]]:
        selected = select_best([parent_row, *rows])
        stage_checkpoint(out, stage, rows, selected)
        if selected["candidate_id"] == parent_row["candidate_id"]:
            return parent, parent_row
        selected_receipt = json.loads(
            (Path(selected["carrier_path"]).parent / "candidate.json").read_text(encoding="utf-8")
        )
        return (
            load_selected_state(
                selected,
                codec,
                tuple(selected_receipt["identity"]["sequence"]),
            ),
            selected,
        )

    current_row = control_row

    # E1: every sign coordinate, then the full interaction cube over the six
    # strongest singleton dimensions plus two global adversarial anchors.
    parent = current
    sign_rows: list[dict[str, Any]] = []
    singleton_sign_rows: list[tuple[int, dict[str, Any]]] = []
    for dimension in range(pk2.DIM):
        signs = np.ones(pk2.DIM, dtype=np.int32)
        signs[dimension] = -1
        row = materialize_unique("E1_exact_sign", parent, signed_state(parent, signs))
        if row is not None:
            sign_rows.append(row)
            singleton_sign_rows.append((dimension, row))
    top_sign_dimensions = [
        dimension
        for dimension, _ in sorted(
            singleton_sign_rows,
            key=lambda item: (
                item[1]["archive_bytes"],
                item[1]["candidate_id"],
            ),
        )[:6]
    ]
    for mask in range(1, 1 << len(top_sign_dimensions)):
        signs = np.ones(pk2.DIM, dtype=np.int32)
        for bit, dimension in enumerate(top_sign_dimensions):
            if mask & (1 << bit):
                signs[dimension] = -1
        row = materialize_unique("E1_exact_sign", parent, signed_state(parent, signs))
        if row is not None:
            sign_rows.append(row)
    for dimensions in (
        tuple(range(pk2.DIM)),
        tuple(dimension for dimension in range(pk2.DIM) if dimension not in top_sign_dimensions),
    ):
        signs = np.ones(pk2.DIM, dtype=np.int32)
        signs[list(dimensions)] = -1
        row = materialize_unique("E1_exact_sign", parent, signed_state(parent, signs))
        if row is not None:
            sign_rows.append(row)
    current, current_row = advance_stage("E1_exact_sign", parent, current_row, sign_rows)

    # E2: decoder-null rowwise DC shifts.  Each row is shifted by one constant
    # across all channels and spatial positions, which the receiver mean removes.
    parent = current
    dc_rows: list[dict[str, Any]] = []
    singleton_dc_rows: list[tuple[int, int, dict[str, Any]]] = []
    for dimension in range(pk2.DIM):
        lower = -16 - int(parent.basis_codes[dimension].min())
        upper = 15 - int(parent.basis_codes[dimension].max())
        for shift in sorted({-1, 1, lower, upper}):
            if shift == 0 or shift < lower or shift > upper:
                continue
            shifts = np.zeros(pk2.DIM, dtype=np.int32)
            shifts[dimension] = shift
            row = materialize_unique("E2_exact_dc_shift", parent, dc_shifted_state(parent, shifts))
            if row is not None:
                dc_rows.append(row)
                singleton_dc_rows.append((dimension, shift, row))
    best_dc_by_dimension: list[tuple[int, int, dict[str, Any]]] = []
    for dimension in range(pk2.DIM):
        candidates = [item for item in singleton_dc_rows if item[0] == dimension]
        if candidates:
            best_dc_by_dimension.append(
                min(candidates, key=lambda item: (item[2]["archive_bytes"], item[2]["candidate_id"]))
            )
    top_dc_operations = sorted(
        best_dc_by_dimension,
        key=lambda item: (item[2]["archive_bytes"], item[2]["candidate_id"]),
    )[:4]
    for mask in range(1, 1 << len(top_dc_operations)):
        shifts = np.zeros(pk2.DIM, dtype=np.int32)
        for bit, (dimension, shift, _) in enumerate(top_dc_operations):
            if mask & (1 << bit):
                shifts[dimension] = shift
        row = materialize_unique("E2_exact_dc_shift", parent, dc_shifted_state(parent, shifts))
        if row is not None:
            dc_rows.append(row)
    current, current_row = advance_stage("E2_exact_dc_shift", parent, current_row, dc_rows)

    # E3: every pairwise transposition.  Inner coder totals are invariant; only
    # the outer XZ/ZIP ordering may change.
    parent = current
    permutation_rows: list[dict[str, Any]] = []
    for first in range(pk2.DIM):
        for second in range(first + 1, pk2.DIM):
            permutation = np.arange(pk2.DIM, dtype=np.int32)
            permutation[first], permutation[second] = permutation[second], permutation[first]
            row = materialize_unique("E3_exact_permutation", parent, permuted_state(parent, permutation))
            if row is not None:
                if (
                    row["basis_bit_count"] != current_row["basis_bit_count"]
                    or row["coefficient_bit_count"] != current_row["coefficient_bit_count"]
                ):
                    raise RuntimeError("dimension permutation changed an inner coder bit total")
                permutation_rows.append(row)
    current, current_row = advance_stage("E3_exact_permutation", parent, current_row, permutation_rows)

    # E4: positive raw-basis radial gauges.  Coefficients are deliberately
    # unchanged because receiver normalization eliminates basis magnitude.
    parent = current
    radial_rows: list[dict[str, Any]] = []

    # materialize only the declared top eight directions at eight amplitudes.

                    np.mean((delta[:, source] / delta_std[source]) * (delta[:, destination] / delta_std[destination]))

    parent = current
    delta = np.diff(parent.coefficients.astype(np.float64), axis=0)

        "ip,jp->ij", normalized_parent64, normalized_parent64, optimize=False

    eigenvalues = np.maximum(eigenvalues, 1e-8)
    root = np.einsum(
        "ik,jk->ij",
        eigenvectors_r * np.sqrt(eigenvalues)[None],
        eigenvectors_r,
        optimize=False,
    )

        "ik,jk->ij",
        eigenvectors_r * (1.0 / np.sqrt(eigenvalues))[None],
        eigenvectors_r,
        optimize=False,
    )

        "ij,jk,kl->il", root, klt, inverse_root, optimize=False
    )

        raise RuntimeError("structured receiver-orthogonal transform is non-finite")

                global_rows.append(row)

def build_parser() -> argparse.ArgumentParser:

    parser.add_argument("--archive", type=Path, default=DEFAULT_ARCHIVE)
    parser.add_argument("--out-dir", type=Path, default=DEFAULT_OUT)

        "--resume-from",
        type=Path,
        default=DEFAULT_OUT / "progress.json",
    )
    parser.add_argument("--required-free-bytes", type=int, default=5_000_000_000)
    parser.add_argument("--max-candidates", type=int, default=452)
    parser.set_defaults(func=run)
    return parser

    args = build_parser().parse_args()
    if args.max_candidates < 452:
        raise ValueError("max-candidates must admit the preregistered <=452-row bank")
    return int(args.func(args))
