"""Per-tensor pruning ablation on the EXACT scorer (no grad plumbing needed).

For each big weight tensor, prune it to keep=K (magnitude) while leaving all
others intact, and measure the exact d_seg/d_pose impact + the blob byte saving.
This reveals which tensors are score-CHEAP to prune (low distortion impact per
byte saved) vs score-CRITICAL. The per-tensor score-cost-per-byte is the
allocation prior for a mixed-rate pruning schedule (compose with #69 requant).
"""
import sys, time
sys.path.insert(0,'experiments/results/pr110pp_r2_nonmps_candidate_20260609/analysis')
sys.path.insert(0,'src')
import render_and_score_lib as L
import numpy as np, torch, copy
from model import HNeRVDecoder
from tac.optimization import structural_q_compression as SQ

N_EVAL = int(sys.argv[1]) if len(sys.argv)>1 else 10
KEEP = float(sys.argv[2]) if len(sys.argv)>2 else 0.5
pairs = list(range(N_EVAL))
scorer = L.ExactScorer()
p = L.parse_frontier()
sd = p['decoder_sd']
probe = HNeRVDecoder(latent_dim=28, base_channels=36, eval_size=(384,512))
order = [name for name,_ in probe.state_dict().items()]
items = [(name, sd[name].numpy().astype(np.float64)) for name in order]
base_codes = SQ.recover_all_codes(items)
base_blob = SQ.encode_decoder_blob_bytes(base_codes)
# big weights (>1000 elems) by name
BIG = {idx:tc for idx,tc in base_codes.items() if tc.codes_i8.size>1000}

gt = L.decode_gt_pairs(pairs)
gt_bthwc = {pi: L.comp_pair_to_bthwc(gt[pi].float().permute(0,3,1,2)) for pi in pairs}

def score_sd(decoder_sd):
    parsed = copy.copy(p); parsed['decoder_sd'] = decoder_sd
    r = L.FrontierRenderer(parsed=parsed)
    comp = r.render_baseline_pairs(pairs)
    segs=[]; poses=[]
    for pi in pairs:
        po,se = scorer.score_batch(gt_bthwc[pi], L.comp_pair_to_bthwc(comp[pi]))
        segs.append(float(se[0])); poses.append(float(po[0]))
    return float(np.mean(segs)), float(np.mean(poses))

base_seg, base_pose = score_sd(sd)
print(f'BASELINE ({N_EVAL} pairs): d_seg={base_seg:.6f} d_pose={base_pose:.6e} blob={base_blob}')
print(f'per-tensor prune to keep={KEEP} (magnitude), measure exact distortion + byte saving:')
print(f'{"tensor":18s} {"d_seg_delta":>12s} {"d_pose_delta":>13s} {"byte_save":>10s} {"score_cost":>11s} {"cost/kB":>9s}')
RATE_DENOM=37545489
rows=[]
for idx,tc in sorted(BIG.items(), key=lambda kv:-kv[1].codes_i8.size):
    t0=time.time()
    sens = SQ.magnitude_sensitivity(tc.codes_i8)
    pc = SQ.score_aware_prune_codes(tc.codes_i8, sens, KEEP)
    sd2 = {k:v.clone() for k,v in sd.items()}
    sd2[tc.name] = torch.from_numpy(pc.astype(np.float32)*tc.scale)
    s,po = score_sd(sd2)
    codes2 = dict(base_codes); codes2[idx]=SQ.TensorCode(tc.name,idx,tc.shape,pc,tc.scale)
    blob = SQ.encode_decoder_blob_bytes(codes2)
    byte_save = base_blob - blob
    # score cost = distortion increase (in S units); rate gain = 25*byte_save/DENOM
    dscore_distort = 100*(s-base_seg) + (np.sqrt(10*po)-np.sqrt(10*base_pose))
    cost_per_kb = dscore_distort/(byte_save/1000.0) if byte_save>0 else float('inf')
    print(f'{tc.name:18s} {s-base_seg:+12.6f} {po-base_pose:+13.3e} {byte_save:+10d} {dscore_distort:+11.5f} {cost_per_kb:+9.4f}  [{time.time()-t0:.0f}s]', flush=True)
    rows.append((tc.name, s-base_seg, po-base_pose, byte_save, dscore_distort))
np.save('/Volumes/VertigoDataTier/pact/task71_per_tensor_ablation_rows.npy', np.array(rows,dtype=object), allow_pickle=True)
