def _sha(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


# --------------------------------------------------------------------------- #
# Per-frame-role quantization: frame0 (EVEN) vs frame1 (ODD)
# --------------------------------------------------------------------------- #
def _quant_planes(planes: np.ndarray, *, bits: int, per: str) -> np.ndarray:
    """Quantize a (K,3,H,W) block to 2**bits levels, dequant float64."""
    if bits >= 53:  # identity float64
        return planes.astype(np.float64, copy=True)
    if bits == 16:  # fp16 passthrough
        return planes.astype(np.float16).astype(np.float64)
    levels = (1 << bits) - 1
    a = planes.astype(np.float64, copy=False)
    if per == "global":
        lo = float(a.min()); hi = float(a.max())
        span = max(hi - lo, 1e-12)
        codes = np.rint((a - lo) / span * levels)
        return (codes / levels * span + lo).astype(np.float64)
    if per == "frame":
        lo = a.min(axis=(1, 2, 3), keepdims=True)
        hi = a.max(axis=(1, 2, 3), keepdims=True)
    else:
        raise ValueError(per)
    span = np.maximum(hi - lo, 1e-12)
    codes = np.rint((a - lo) / span * levels)
    return (codes / levels * span + lo).astype(np.float64)


def frame_role_quant(skip_high, *, frame0_bits, frame0_per, frame1_bits, frame1_per):
    """skip_high (1200,3,192,256); EVEN=frame0 ODD=frame1. Returns dequant float64."""
    out = skip_high.astype(np.float64, copy=True)
    even = np.arange(0, skip_high.shape[0], 2)  # frame0
    odd = np.arange(1, skip_high.shape[0], 2)   # frame1
    out[even] = _quant_planes(skip_high[even], bits=frame0_bits, per=frame0_per)
    out[odd] = _quant_planes(skip_high[odd], bits=frame1_bits, per=frame1_per)
    return out

def reencode_decoder_payload(decoder_payload, new_skip_high):
    from tac.substrates.snerv_inverse_steg_carrier import archive as A

    header, comp = A._unpack_subpacket(
        decoder_payload, magic=A.SNERV_DECODER_MAGIC,
        schema=A.DECODER_PAYLOAD_OFFICIAL_MFU_HFR_TUB_SCHEMA,
    )
    raw = lzma.decompress(comp)
    tensors = A._unpack_tensor_manifest(raw, header.get("tensor_manifest") or [])
    tensors[SKIP_HIGH_NAME] = np.ascontiguousarray(new_skip_high, dtype=np.float64)
    manifest, new_raw = A._pack_tensor_manifest(tensors)
    new_comp = lzma.compress(
        new_raw, format=lzma.FORMAT_XZ,
        preset=A.DECODER_PAYLOAD_OFFICIAL_MFU_HFR_TUB_LZMA_PRESET,
    )
    new_header = dict(header)
    new_header["tensor_manifest"] = manifest
    new_header["tensor_count"] = len(manifest)
    new_header["raw_tensor_bytes"] = len(new_raw)
    new_header["raw_tensor_sha256"] = A._sha256(new_raw)
    new_header["compressed_bytes"] = len(new_comp)
    new_header["compressed_sha256"] = A._sha256(new_comp)
    return A._pack_subpacket(A.SNERV_DECODER_MAGIC, new_header, new_comp)


def rebuild_packet(orig_packet, new_decoder_payload):
    from tac.substrates.snerv_inverse_steg_carrier import archive as A

    decoded = A.unpack_snerv_archive(orig_packet)
    sections = dict(decoded.sections)
    sections["decoder_payload"] = new_decoder_payload
    return A.pack_snerv_archive_snar2(
        metadata_payload=sections["metadata_payload"],
        lf_payload=sections["lf_payload"],
        decoder_payload=sections["decoder_payload"],
        step_map_packet=sections["step_map_packet"],
        metadata=decoded.metadata,
    ).packet


# --------------------------------------------------------------------------- #
# Per-role entropy-true storage floor (split streams by frame role)
# --------------------------------------------------------------------------- #
def carrier_lf_role_storage_bytes(skip_high, *, frame0_bits, frame0_per, frame1_bits, frame1_per):
    """Carrier lossless intN codec on EACH role's integer codes, split streams.

    Returns (frame0_code_bytes, frame1_code_bytes, scale_sidecar_bytes).
    Split-by-role is the measured-axis storage (frame0 codes are far lower entropy
    after hard quant). NOT receiver-decodable yet (receiver wants float64) — a
    storage-floor estimate apples-to-apples with the fidelity rows.
    """
    from tac.substrates.snerv_inverse_steg_carrier.lf_payload_codec import (
        encode_lf_quant_payload_v2_with_report,
    )

    def _codes(planes, bits, per):
        levels = (1 << bits) - 1
        a = planes.astype(np.float64, copy=False)
        if per == "global":
            lo = np.array(a.min()).reshape(1, 1, 1, 1)
            hi = np.array(a.max()).reshape(1, 1, 1, 1)
        elif per == "frame":
            lo = a.min(axis=(1, 2, 3), keepdims=True)
            hi = a.max(axis=(1, 2, 3), keepdims=True)
        else:
            raise ValueError(per)
        span = np.maximum(hi - lo, 1e-12)
        codes = np.rint((a - lo) / span * levels).astype(np.int64)
        return codes, lo, span

    even = np.arange(0, skip_high.shape[0], 2)
    odd = np.arange(1, skip_high.shape[0], 2)
    c0, lo0, sp0 = _codes(skip_high[even], frame0_bits, frame0_per)
    c1, lo1, sp1 = _codes(skip_high[odd], frame1_bits, frame1_per)

    def _enc(codes):
        planes = [codes[i].reshape(codes.shape[1], -1) for i in range(codes.shape[0])]
        pkt, _rep = encode_lf_quant_payload_v2_with_report(
            planes, mode="portfolio_auto", wrapper="portfolio_auto"
        )
        return len(pkt)

    b0 = _enc(c0)
    b1 = _enc(c1)
    scale_raw = np.concatenate([
        lo0.astype(np.float32).reshape(-1), sp0.astype(np.float32).reshape(-1),
        lo1.astype(np.float32).reshape(-1), sp1.astype(np.float32).reshape(-1),
    ]).tobytes()
    scale_bytes = len(lzma.compress(scale_raw, preset=6))
    return b0, b1, scale_bytes


def _weights_lzma_bytes(tensors):
    from tac.substrates.snerv_inverse_steg_carrier import archive as A

    sub = {k: v for k, v in tensors.items() if k != SKIP_HIGH_NAME}
    _manifest, raw = A._pack_tensor_manifest(sub)
    return len(lzma.compress(raw, format=lzma.FORMAT_XZ, preset=6))


def _zip_stored_bytes(packet):
    import io

    buf = io.BytesIO()
    with zipfile.ZipFile(buf, "w", zipfile.ZIP_STORED) as zf:
        zf.writestr("x", packet)
    return len(buf.getvalue())


def score_packet(packet, n_pairs, source_pairs, posenet, segnet):
    from tac.substrates.snerv_inverse_steg_carrier.archive import decode_snerv_archive_frames
    from tac.substrates.snerv_inverse_steg_carrier.inflate import _resize_nchw_bilinear
    from tac.analysis.inverse_steganalysis_linf_vs_l2_gate import measure_pairs_d_seg_d_pose_batched
    import torch

    t0 = time.time()
    frames = decode_snerv_archive_frames(packet)
    decode_s = time.time() - t0
    arr = np.asarray(frames, dtype=np.float32)
    if arr.ndim == 5 and arr.shape[1] == 2:
        pairs_arr = arr
    elif arr.ndim == 4 and arr.shape[0] % 2 == 0:
        pairs_arr = arr.reshape(arr.shape[0] // 2, 2, *arr.shape[1:])
    else:
        raise RuntimeError(f"unexpected decoded frame shape {arr.shape}")
    cam_h, cam_w = int(source_pairs.shape[-2]), int(source_pairs.shape[-1])
    n = min(n_pairs, int(pairs_arr.shape[0]))
    sel = pairs_arr[:n]
    flat = sel.reshape(-1, 3, sel.shape[-2], sel.shape[-1])
    resized = _resize_nchw_bilinear(flat, out_hw=(cam_h, cam_w))
    u8 = np.rint(np.clip(resized, 0.0, 255.0)).astype(np.uint8)
    recon_np = u8.astype(np.float32).reshape(n, 2, 3, cam_h, cam_w)
    recon = torch.from_numpy(recon_np).to(source_pairs)
    t1 = time.time()
    dsegs, dposes = measure_pairs_d_seg_d_pose_batched(
        posenet, segnet, source_pairs[:n], recon, batch_pairs=8
    )
    d_seg = float(np.mean(dsegs))
    d_pose = float(np.mean(dposes))
    return {
        "decode_seconds": decode_s,
        "score_seconds": time.time() - t1,
        "n_pairs_scored": n,
        "d_seg": d_seg,
        "d_pose": d_pose,
        "pose_score_term": float(np.sqrt(10.0 * max(d_pose, 0.0))),
        "per_pair_d_seg_first8": [float(v) for v in dsegs[:8]],
        "per_pair_d_pose_first8": [float(v) for v in dposes[:8]],
    }

def _verdict(out):
    rungs = out["rungs"]
    if not rungs:
        return {"note": "no rungs"}
    ctrl = next((r for r in rungs if r["rung_id"] == "S5a_ctrl_both_uint8_global"), None)
    # The frame0 floor: the hardest frame0 quant whose d_seg ~ ctrl AND pose guard holds.
    f0_rungs = [r for r in rungs if r["frame0_bits"] < 8 and r["guard_pass"]]
    best_floor = min(rungs, key=lambda r: r["advisory_total_storage_floor"]) if rungs else None
    return {
        "ctrl_d_seg": ctrl["d_seg"] if ctrl else None,
        "frame0_segnet_blind_confirmed": (
            all(abs(r.get("delta_d_seg_vs_ctrl") or 0.0) < 5e-4 for r in f0_rungs)
            if f0_rungs else None
        ),
        "best_floor_total": best_floor["advisory_total_storage_floor"] if best_floor else None,
        "best_floor_rung": best_floor["rung_id"] if best_floor else None,
        "best_floor_bytes": best_floor["est_packet_floor_bytes"] if best_floor else None,
        "note": (
            "frame0 is SegNet-blind: harder frame0 quant should leave d_seg ~ ctrl "
            "(delta_d_seg_vs_ctrl ~ 0) while d_pose prices the frame0 pose floor. "
            "The frame0-asymmetry win = the byte saving from quantizing 50% of the "
            "planes harder at near-zero d_seg cost."
        ),
    }
