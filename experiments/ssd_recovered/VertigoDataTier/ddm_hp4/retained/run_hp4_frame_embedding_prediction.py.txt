def sha256_bytes(value: bytes) -> str:
    return hashlib.sha256(value).hexdigest()


def sha256_file(path: Path) -> str:
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def atomic_bytes(path: Path, value: bytes, *, executable: bool = False) -> None:
    """Write once, or verify an identical resume artifact without overwriting it."""

    path.parent.mkdir(parents=True, exist_ok=True)
    if path.exists():
        if path.read_bytes() != value:
            raise RuntimeError(f"resume artifact differs: {path}")
        return
    temporary = path.with_name(f".{path.name}.tmp.{os.getpid()}")
    with temporary.open("xb") as stream:
        stream.write(value)
        stream.flush()
        os.fsync(stream.fileno())
    if executable:
        temporary.chmod(0o755)
    os.replace(temporary, path)


def atomic_json(path: Path, value: Any) -> None:
    atomic_bytes(path, (json.dumps(value, indent=2, sort_keys=True) + "\n").encode())


def atomic_npy(path: Path, value: np.ndarray) -> None:
    buffer = io.BytesIO()
    np.save(buffer, value, allow_pickle=False)
    atomic_bytes(path, buffer.getvalue())


def file_record(path: Path) -> dict[str, Any]:
    return {
        "path": str(path.resolve()),
        "bytes": path.stat().st_size,
        "sha256": sha256_file(path),
    }


def verify_file(path: Path, expected: tuple[int, str], label: str) -> dict[str, Any]:
    record = file_record(path)
    if (record["bytes"], record["sha256"]) != expected:
        raise RuntimeError(f"{label} pin differs: {record}")
    return record


def deterministic_zip(member: bytes) -> bytes:
    output = io.BytesIO()
    with zipfile.ZipFile(output, "w", allowZip64=False) as archive:
        info = zipfile.ZipInfo("p", date_time=(1980, 1, 1, 0, 0, 0))
        info.compress_type = zipfile.ZIP_STORED
        info.create_system = 3
        info.external_attr = 0o100644 << 16
        archive.writestr(info, member)
    return output.getvalue()


def read_member(path: Path) -> bytes:
    with zipfile.ZipFile(path) as archive:
        infos = archive.infolist()
        if len(infos) != 1 or infos[0].filename != "p" or infos[0].compress_type != zipfile.ZIP_STORED:
            raise RuntimeError("expected one stored ZIP member named p")
        value = archive.read(infos[0])
        if archive.testzip() is not None:
            raise RuntimeError("ZIP CRC validation failed")
    return value


def brotli(value: bytes, quality: int, binary: str) -> bytes:
    completed = subprocess.run(
        [binary, "-q", str(quality), "-c"],
        input=value,
        check=False,
        capture_output=True,
    )
    if completed.returncode:
        raise RuntimeError(completed.stderr.decode(errors="replace"))
    return completed.stdout


def unbrotli(value: bytes, binary: str) -> bytes:
    completed = subprocess.run(
        [binary, "-d", "-c"],
        input=value,
        check=False,
        capture_output=True,
    )
    if completed.returncode:
        raise RuntimeError(completed.stderr.decode(errors="replace"))
    return completed.stdout


def split_models(models: bytes) -> tuple[bytes, bytes, bytes]:
    if len(models) < SPLIT_HEADER.size:
        raise RuntimeError("current split-model payload is truncated")
    a, b, c = SPLIT_HEADER.unpack_from(models)
    if SPLIT_HEADER.size + a + b + c != len(models):
        raise RuntimeError("current split-model lengths do not reconcile")
    offset = SPLIT_HEADER.size
    return (
        models[offset : offset + a],
        models[offset + a : offset + a + b],
        models[offset + a + b :],
    )


def unpack_nibbles(packed: bytes) -> np.ndarray:
    source = np.frombuffer(packed, dtype=np.uint8)
    values = np.empty(source.size * 2, dtype=np.int16)
    values[0::2] = source & 0x0F
    values[1::2] = source >> 4
    return values


def pack_nibbles(values: np.ndarray) -> bytes:
    source = np.asarray(values, dtype=np.int16).reshape(-1)
    if source.size % 2 or np.any(source < 0) or np.any(source > 15):
        raise RuntimeError("nibble stream is invalid")
    unsigned = source.astype(np.uint8)
    return (unsigned[0::2] | (unsigned[1::2] << 4)).tobytes()


def signed_mod(values: np.ndarray, bits: int) -> np.ndarray:
    modulus = 1 << bits
    half = modulus >> 1
    return ((np.asarray(values, dtype=np.int64) + half) % modulus - half).astype(np.int16)


def trunc_div(values: np.ndarray, divisor: int) -> np.ndarray:
    source = np.asarray(values, dtype=np.int64)
    return (np.sign(source) * (np.abs(source) // divisor)).astype(np.int64)


def class_histograms(path: Path) -> np.ndarray:
    tokens = np.memmap(path, mode="r", dtype=np.uint8, shape=(FRAME_COUNT, 384, 512))
    result = np.empty((FRAME_COUNT, 5), dtype=np.int64)
    for frame in range(FRAME_COUNT):
        result[frame] = np.bincount(tokens[frame].reshape(-1), minlength=5)[:5]
    return result


def state_features(coefficients: np.ndarray, histograms: np.ndarray) -> dict[str, np.ndarray]:
    pose = np.zeros((FRAME_COUNT, 12), dtype=np.int16)
    coefficient_delta = np.zeros_like(coefficients, dtype=np.int32)
    coefficient_delta[1:] = signed_mod(coefficients[1:] - coefficients[:-1], 12)
    pose[1:] = signed_mod(trunc_div(coefficient_delta[1:], 64), 4)

    partition = np.zeros((FRAME_COUNT, 10), dtype=np.int16)
    for frame in range(1, FRAME_COUNT):
        previous = histograms[frame - 1]
        older = histograms[frame - 2] if frame >= 2 else np.zeros(5, dtype=np.int64)
        partition[frame, :5] = signed_mod(previous // 4096, 4)
        partition[frame, 5:] = signed_mod(trunc_div(previous - older, 1024), 4)

    ones = np.ones((FRAME_COUNT, 1), dtype=np.int16)
    ones[0] = 0
    return {
        "state_pose": np.concatenate((pose, ones), axis=1),
        "state_partition": np.concatenate((partition, ones), axis=1),
        "state_joint": np.concatenate((pose, partition, ones), axis=1),
    }


def fit_integer_weights(values: np.ndarray, features: np.ndarray) -> np.ndarray:
    """Deterministic charged int4 coordinate fit for temporal embedding deltas."""

    target = signed_mod(values[1:] - values[:-1], 4)
    design = np.asarray(features[1:], dtype=np.int16)
    weights = np.zeros((FRAME_DIM, design.shape[1]), dtype=np.int8)
    for dimension in range(FRAME_DIM):
        current = np.zeros(design.shape[1], dtype=np.int16)
        for _ in range(3):
            for feature in range(design.shape[1]):
                best: tuple[tuple[int, int, int, int, int], int] | None = None
                for candidate in range(-8, 8):
                    trial = current.copy()
                    trial[feature] = candidate
                    residual = signed_mod(
                        target[:, dimension] - design.astype(np.int64) @ trial.astype(np.int64),
                        4,
                    )
                    score = (
                        int(np.count_nonzero(residual)),
                        int(np.abs(residual).sum()),
                        int(np.square(residual.astype(np.int32)).sum()),
                        abs(candidate),
                        candidate,
                    )
                    if best is None or score < best[0]:
                        best = (score, candidate)
                assert best is not None
                current[feature] = best[1]
        weights[dimension] = current.astype(np.int8)
    return weights


def predictor_residuals(
    values: np.ndarray,
    predictor: str,
    features: np.ndarray | None,
    weights: np.ndarray | None,
) -> np.ndarray:
    residuals = np.empty_like(values, dtype=np.uint8)
    for frame in range(FRAME_COUNT):
        if predictor == "order0" or frame == 0:
            prediction = np.zeros(FRAME_DIM, dtype=np.int16)
        elif predictor == "ar1":
            prediction = values[frame - 1]
        elif predictor == "ar2":
            prediction = values[frame - 1] if frame == 1 else signed_mod(2 * values[frame - 1] - values[frame - 2], 4)
        else:
            assert features is not None and weights is not None
            correction = features[frame].astype(np.int64) @ weights.astype(np.int64).T
            prediction = signed_mod(values[frame - 1] + correction, 4)
        residuals[frame] = ((values[frame] - prediction) & 0x0F).astype(np.uint8)
    return residuals


def restore_values(
    residuals: np.ndarray,
    predictor: str,
    features: np.ndarray | None,
    weights: np.ndarray | None,
) -> np.ndarray:
    restored = np.empty((FRAME_COUNT, FRAME_DIM), dtype=np.int16)
    source = np.asarray(residuals, dtype=np.int16).reshape(FRAME_COUNT, FRAME_DIM)
    for frame in range(FRAME_COUNT):
        if predictor == "order0" or frame == 0:
            prediction = np.zeros(FRAME_DIM, dtype=np.int16)
        elif predictor == "ar1":
            prediction = restored[frame - 1]
        elif predictor == "ar2":
            prediction = (
                restored[frame - 1] if frame == 1 else signed_mod(2 * restored[frame - 1] - restored[frame - 2], 4)
            )
        else:
            assert features is not None and weights is not None
            correction = features[frame].astype(np.int64) @ weights.astype(np.int64).T
            prediction = signed_mod(restored[frame - 1] + correction, 4)
        restored[frame] = signed_mod(prediction + source[frame], 4)
    return restored


def load_rc64(output: Path):
    spec = importlib.util.spec_from_file_location("ddm_hp4_real_rc64", RC64_PY)
    if spec is None or spec.loader is None:
        raise RuntimeError("cannot import the retained RC64 implementation")
    module = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = module
    spec.loader.exec_module(module)
    library = output / "runtime" / "rc64_backend.so"
    if not library.exists():
        library.parent.mkdir(parents=True, exist_ok=True)
        module.compile_backend(library)
    return module, library


def adaptive_frequency_rows(symbols: np.ndarray, total: int) -> tuple[np.ndarray, np.ndarray]:
    source = np.asarray(symbols, dtype=np.int32).reshape(FRAME_COUNT, FRAME_DIM, 2)
    flattened = source.reshape(-1)
    contexts = np.broadcast_to(
        np.arange(FRAME_DIM * 2, dtype=np.int16).reshape(1, FRAME_DIM, 2),
        source.shape,
    ).reshape(-1)
    counts = np.ones((FRAME_DIM * 2, 5), dtype=np.uint64)
    rows = np.empty((flattened.size, 5), dtype=np.uint32)
    for index, (symbol, context) in enumerate(zip(flattened, contexts, strict=True)):
        current = counts[int(context)]
        frequencies = np.maximum(1, current * total // int(current.sum())).astype(np.int64)
        winner = int(np.argmax(current))
        frequencies[winner] += total - int(frequencies.sum())
        if np.any(frequencies <= 0) or int(frequencies.sum()) != total:
            raise RuntimeError("adaptive RC64 frequency normalization failed")
        rows[index] = frequencies.astype(np.uint32)
        counts[int(context), int(symbol)] += 1
    return flattened, rows


def rc64_encode(raw_nibbles: bytes, module, library: Path) -> bytes:
    nibbles = unpack_nibbles(raw_nibbles).reshape(FRAME_COUNT, FRAME_DIM)
    digits = np.stack((nibbles & 3, nibbles >> 2), axis=2).astype(np.int32)
    symbols, rows = adaptive_frequency_rows(digits, module.TOTAL)
    encoder = module.NativeEncoder(library)
    try:
        encoder.encode_frequencies(symbols, rows)
        return encoder.finish()
    finally:
        encoder.close()


def rc64_decode(payload: bytes, module, library: Path) -> bytes:
    counts = np.ones((FRAME_DIM * 2, 5), dtype=np.uint64)
    decoded = np.empty(FRAME_VALUES * 2, dtype=np.int16)
    decoder = module.NativeDecoder(library, payload)
    try:
        for index in range(decoded.size):
            dimension_plane = index % (FRAME_DIM * 2)
            current = counts[dimension_plane]
            frequencies = np.maximum(
                1,
                current * module.TOTAL // int(current.sum()),
            ).astype(np.int64)
            winner = int(np.argmax(current))
            frequencies[winner] += module.TOTAL - int(frequencies.sum())
            symbol = int(decoder.decode_frequencies(frequencies.astype(np.uint32).reshape(1, 5))[0])
            if symbol >= 4:
                raise RuntimeError("RC64 decoded the reserved fifth symbol")
            decoded[index] = symbol
            counts[dimension_plane, symbol] += 1
    finally:
        decoder.close()
    digits = decoded.reshape(FRAME_COUNT, FRAME_DIM, 2)
    nibbles = digits[:, :, 0] | (digits[:, :, 1] << 2)
    return pack_nibbles(nibbles.reshape(-1))


def encode_codec(raw: bytes, codec: str, brotli_binary: str, rc64_module, library: Path) -> bytes:
    if codec == "rc64_adaptive_base4":
        return rc64_encode(raw, rc64_module, library)
    if codec == "brotli_q11":
        return brotli(raw, 11, brotli_binary)
    return lzma.compress(raw, format=lzma.FORMAT_RAW, filters=LZMA_FILTERS[codec])


def decode_codec(
    payload: bytes,
    codec: str,
    brotli_binary: str,
    rc64_module,
    library: Path,
) -> bytes:
    if codec == "rc64_adaptive_base4":
        return rc64_decode(payload, rc64_module, library)
    if codec == "brotli_q11":
        return unbrotli(payload, brotli_binary)
    return lzma.decompress(payload, format=lzma.FORMAT_RAW, filters=LZMA_FILTERS[codec])


def pack_hp4_model(
    predictor: str,
    codec: str,
    remainder_stream: bytes,
    parameters: bytes,
    embedding_stream: bytes,
    semantic_stream: bytes,
    carrier_stream: bytes,
) -> bytes:
    fields = (
        remainder_stream,
        parameters,
        embedding_stream,
        semantic_stream,
        carrier_stream,
    )
    if any(len(value) >= 1 << 16 for value in fields):
        raise RuntimeError("HP4 model field exceeds its u16 length")
    header = HP4_HEADER.pack(
        HP4_MAGIC,
        HP4_VERSION,
        PREDICTOR_IDS[predictor],
        CODEC_IDS[codec],
        0,
        *(len(value) for value in fields),
    )
    return header + b"".join(fields)


def unpack_hp4_model(model: bytes) -> tuple[str, str, tuple[bytes, bytes, bytes, bytes, bytes]]:
    if len(model) < HP4_HEADER.size:
        raise RuntimeError("HP4 model is truncated")
    magic, version, predictor_id, codec_id, flags, *lengths = HP4_HEADER.unpack_from(model)
    if magic != HP4_MAGIC or version != HP4_VERSION or flags:
        raise RuntimeError("unsupported HP4 model header")
    if HP4_HEADER.size + sum(lengths) != len(model):
        raise RuntimeError("HP4 model lengths do not consume the payload")
    fields = []
    offset = HP4_HEADER.size
    for length in lengths:
        fields.append(model[offset : offset + length])
        offset += length
    return ID_TO_PREDICTOR[predictor_id], ID_TO_CODEC[codec_id], tuple(fields)  # type: ignore[return-value]


def entropy_bits(values: np.ndarray) -> float:
    counts = np.bincount(np.asarray(values, dtype=np.uint8).reshape(-1), minlength=16)
    nonzero = counts[counts > 0].astype(np.float64)
    return float((-nonzero * np.log2(nonzero / nonzero.sum())).sum())


def persist_candidate(
    output: Path,
    predictor: str,
    codec: str,
    raw_residual: bytes,
    parameters: bytes,
    remainder_stream: bytes,
    semantic_stream: bytes,
    carrier_stream: bytes,
    residual_payload: bytes,
    residual_repeat: bytes,
    compact_residual: bytes,
    tokens: bytes,
    brotli_binary: str,
    rc64_module,
    library: Path,
    features: np.ndarray | None,
    weights: np.ndarray | None,
    source_values: np.ndarray,
    source_body: bytes,
) -> dict[str, Any]:
    candidate_root = output / "candidates" / predictor / codec
    paths = {
        "raw": candidate_root / "embedding_residual.nibbles.bin",
        "raw_repeat": candidate_root / "embedding_residual.repeat.nibbles.bin",
        "parameters": candidate_root / "predictor_parameters.bin",
        "payload": candidate_root / "embedding.payload.bin",
        "payload_repeat": candidate_root / "embedding.repeat.payload.bin",
        "model": candidate_root / "models.hp4m.bin",
        "model_repeat": candidate_root / "models.repeat.hp4m.bin",
        "member": candidate_root / "p",
        "member_repeat": candidate_root / "p.repeat",
        "archive": candidate_root / "archive.zip",
        "archive_repeat": candidate_root / "archive.repeat.zip",
    }
    atomic_bytes(paths["raw"], raw_residual)
    atomic_bytes(paths["raw_repeat"], raw_residual)
    atomic_bytes(paths["parameters"], parameters)
    atomic_bytes(paths["payload"], residual_payload)
    atomic_bytes(paths["payload_repeat"], residual_repeat)

    model = pack_hp4_model(
        predictor,
        codec,
        remainder_stream,
        parameters,
        residual_payload,
        semantic_stream,
        carrier_stream,
    )
    model_repeat = pack_hp4_model(
        predictor,
        codec,
        remainder_stream,
        parameters,
        residual_repeat,
        semantic_stream,
        carrier_stream,
    )
    atomic_bytes(paths["model"], model)
    atomic_bytes(paths["model_repeat"], model_repeat)
    member = model + compact_residual + tokens
    member_repeat = model_repeat + compact_residual + tokens
    atomic_bytes(paths["member"], member)
    atomic_bytes(paths["member_repeat"], member_repeat)
    archive = deterministic_zip(member)
    archive_repeat = deterministic_zip(member_repeat)
    atomic_bytes(paths["archive"], archive)
    atomic_bytes(paths["archive_repeat"], archive_repeat)

    parsed_predictor, parsed_codec, fields = unpack_hp4_model(paths["model"].read_bytes())
    parsed_remainder, parsed_parameters, parsed_embedding, parsed_semantic, parsed_carrier = fields
    if parsed_predictor != predictor or parsed_codec != codec or parsed_parameters != parameters:
        raise RuntimeError("HP4 header/parameter parse-back differs")
    restored_remainder = unbrotli(parsed_remainder, brotli_binary)
    decoded_raw = decode_codec(parsed_embedding, codec, brotli_binary, rc64_module, library)
    if decoded_raw != raw_residual:
        raise RuntimeError("embedding residual coder parse-back differs")
    decoded_unsigned = unpack_nibbles(decoded_raw).reshape(FRAME_COUNT, FRAME_DIM)
    restored_values = restore_values(decoded_unsigned, predictor, features, weights)
    if not np.array_equal(restored_values, source_values):
        raise RuntimeError("causal predictor did not restore the exact embedding")
    restored_packed = pack_nibbles((restored_values & 0x0F).reshape(-1))
    restored_body = restored_remainder[:FRAME_OFFSET_BODY] + restored_packed + restored_remainder[FRAME_OFFSET_BODY:]
    if (
        restored_body != source_body
        or parsed_semantic != semantic_stream
        or parsed_carrier != carrier_stream
        or read_member(paths["archive"]) != member
    ):
        raise RuntimeError("complete candidate parse-back differs")
    if archive != archive_repeat or model != model_repeat or residual_payload != residual_repeat:
        raise RuntimeError("determinism repeat differs")

    row = {
        "candidate": f"{predictor}__{codec}",
        "predictor": predictor,
        "codec": codec,
        "axis": AXIS,
        "score_claim": False,
        "whole_container_recount": True,
        "causal_lossless_decode": True,
        "embedding_decode_byte_identical": True,
        "hpac_body_decode_byte_identical": True,
        "semantic_stream_byte_identical": True,
        "carrier_stream_byte_identical": True,
        "repeat_byte_identical": True,
        "raw_residual_h0_bits": entropy_bits(decoded_unsigned),
        "delta_bytes_vs_cp135": paths["archive"].stat().st_size - EXPECTED["archive"][0],
        "derived_rate_score_delta": 25.0
        * (paths["archive"].stat().st_size - EXPECTED["archive"][0])
        / RATE_DENOMINATOR,
        "artifacts": {name: file_record(path) for name, path in paths.items()},
    }
    atomic_json(candidate_root / "RESULT.json", row)
    return row


def parser() -> argparse.ArgumentParser:
    value = argparse.ArgumentParser(description=__doc__)
    value.add_argument("--output", type=Path, default=ROOT)
    value.add_argument("--resume-from", choices=("auto",), default="auto")
    value.add_argument("--brotli", default=shutil.which("brotli") or "brotli")
    return value


def main() -> None:
    args = parser().parse_args()
    output = args.output.resolve()
    output.mkdir(parents=True, exist_ok=True)
    started_path = output / "RUN_CONTEXT.json"
    if started_path.exists():
        run_context = json.loads(started_path.read_text())
    else:
        run_context = {
            "schema": "ddm_hp4_run_context.v1",
            "started_utc": datetime.now(UTC).isoformat(),
            "argv": sys.argv,
            "cwd": str(Path.cwd().resolve()),
            "output": str(output),
            "resume_from": args.resume_from,
        }
        atomic_json(started_path, run_context)

    free_bytes = shutil.disk_usage(output).free
    if free_bytes < 512 * 1024 * 1024:
        raise RuntimeError("HP4 storage preflight requires at least 512 MiB free")
    sources = {
        "archive": verify_file(BASE_ARCHIVE, EXPECTED["archive"], "archive"),
        "models": verify_file(BASE_MODELS, EXPECTED["models"], "models"),
        "residual": verify_file(BASE_RESIDUAL, EXPECTED["residual"], "residual"),
        "tokens": verify_file(BASE_TOKENS, EXPECTED["tokens"], "tokens"),
        "decoded_tokens": verify_file(
            DECODED_TOKENS,
            EXPECTED["decoded_tokens"],
            "decoded tokens",
        ),
        "coefficients": verify_file(COEFFICIENTS, EXPECTED["coefficients"], "coefficients"),
        "rc64_python": file_record(RC64_PY),
        "rc64_c": file_record(RC64_C),
        "runner": file_record(Path(__file__)),
    }
    preflight_checkpoint = output / "checkpoint_00_preflight.json"
    preflight_record = {
        "schema": "ddm_hp4_checkpoint.v1",
        "stage": "preflight",
        "free_bytes": free_bytes,
        "required_bytes": 512 * 1024 * 1024,
        "sources": sources,
    }
    if preflight_checkpoint.exists():
        previous_preflight = json.loads(preflight_checkpoint.read_text())
        if previous_preflight.get("sources") != sources:
            raise RuntimeError("resume source pins differ from the preflight checkpoint")
    else:
        atomic_json(preflight_checkpoint, preflight_record)

    models = BASE_MODELS.read_bytes()
    compact_residual = BASE_RESIDUAL.read_bytes()
    tokens = BASE_TOKENS.read_bytes()
    hpac_stream, semantic_stream, carrier_stream = split_models(models)
    hpac_body = unbrotli(hpac_stream, args.brotli)
    if (len(hpac_body), sha256_bytes(hpac_body)) != EXPECTED["hpac_body"]:
        raise RuntimeError("current HPAC body pin differs")
    rebuilt_hpac_stream = brotli(hpac_body, 10, args.brotli)
    if rebuilt_hpac_stream != hpac_stream:
        raise RuntimeError("Brotli 1.2.0 q10 does not reproduce the incumbent HPAC stream")
    incumbent_root = output / "incumbent"
    atomic_bytes(incumbent_root / "hpac_body.bin", hpac_body)
    atomic_bytes(incumbent_root / "hpac_q10.br", hpac_stream)
    atomic_bytes(incumbent_root / "semantic_q11.br", semantic_stream)
    atomic_bytes(incumbent_root / "carrier_q11.br", carrier_stream)
    incumbent_member = models + compact_residual + tokens
    atomic_bytes(incumbent_root / "p", incumbent_member)
    incumbent_archive = deterministic_zip(incumbent_member)
    atomic_bytes(incumbent_root / "archive.zip", incumbent_archive)
    atomic_bytes(incumbent_root / "archive.repeat.zip", deterministic_zip(incumbent_member))
    if incumbent_archive != BASE_ARCHIVE.read_bytes() or read_member(BASE_ARCHIVE) != incumbent_member:
        raise RuntimeError("incumbent complete-container rebuild differs")

    full_ihs2 = IHS2_HEADER + hpac_body
    packed_embedding = full_ihs2[FRAME_OFFSET_IHS2 : FRAME_OFFSET_IHS2 + FRAME_BYTES]
    if len(packed_embedding) != FRAME_BYTES:
        raise RuntimeError("IHS2 frame field is truncated")
    unsigned_values = unpack_nibbles(packed_embedding).reshape(FRAME_COUNT, FRAME_DIM)
    source_values = signed_mod(unsigned_values, 4)
    if int(source_values.min()) != -6 or int(source_values.max()) != 6:
        raise RuntimeError("current HP3-step2 frame embedding range differs")
    remainder = hpac_body[:FRAME_OFFSET_BODY] + hpac_body[FRAME_END_BODY:]
    remainder_stream = brotli(remainder, 10, args.brotli)
    atomic_bytes(incumbent_root / "embedding.int4.bin", packed_embedding)
    atomic_npy(incumbent_root / "embedding.int8.npy", source_values.astype(np.int8))
    atomic_bytes(incumbent_root / "hpac_remainder.bin", remainder)
    atomic_bytes(incumbent_root / "hpac_remainder_q10.br", remainder_stream)

    zero_body = hpac_body[:FRAME_OFFSET_BODY] + bytes(FRAME_BYTES) + hpac_body[FRAME_END_BODY:]
    zero_stream = brotli(zero_body, 10, args.brotli)
    zero_repeat = brotli(zero_body, 10, args.brotli)
    zero_root = output / "current_size_counterfactual_zero_field"
    atomic_bytes(zero_root / "hpac_body.bin", zero_body)
    atomic_bytes(zero_root / "hpac_q10.br", zero_stream)
    atomic_bytes(zero_root / "hpac.repeat_q10.br", zero_repeat)
    zero_models = (
        SPLIT_HEADER.pack(len(zero_stream), len(semantic_stream), len(carrier_stream))
        + zero_stream
        + semantic_stream
        + carrier_stream
    )
    atomic_bytes(zero_root / "models.bin", zero_models)
    zero_member = zero_models + compact_residual + tokens
    atomic_bytes(zero_root / "p", zero_member)
    zero_archive = deterministic_zip(zero_member)
    atomic_bytes(zero_root / "archive.zip", zero_archive)
    atomic_bytes(zero_root / "archive.repeat.zip", deterministic_zip(zero_member))
    if zero_stream != zero_repeat:
        raise RuntimeError("zero-field counterfactual repeat differs")

    histograms = class_histograms(DECODED_TOKENS)
    coefficients = np.load(COEFFICIENTS, allow_pickle=False)
    if coefficients.shape != (FRAME_COUNT, 12) or coefficients.dtype != np.int16:
        raise RuntimeError("counted carrier coefficient geometry differs")
    atomic_npy(output / "features" / "decoded_partition_histograms.int64.npy", histograms)
    feature_bank = state_features(coefficients, histograms)
    for name, features in feature_bank.items():
        atomic_npy(output / "features" / f"{name}.int16.npy", features)
    atomic_json(
        output / "checkpoint_01_sources_and_features.json",
        {
            "schema": "ddm_hp4_checkpoint.v1",
            "stage": "sources_and_features",
            "incumbent_archive": file_record(incumbent_root / "archive.zip"),
            "incumbent_hpac_stream": file_record(incumbent_root / "hpac_q10.br"),
            "embedding": file_record(incumbent_root / "embedding.int4.bin"),
            "hpac_remainder_q10": file_record(incumbent_root / "hpac_remainder_q10.br"),
            "zero_field_archive": file_record(zero_root / "archive.zip"),
            "partition_histograms": file_record(output / "features" / "decoded_partition_histograms.int64.npy"),
            "feature_files": {name: file_record(output / "features" / f"{name}.int16.npy") for name in feature_bank},
        },
    )

    rc64_module, library = load_rc64(output)
    atomic_json(
        output / "checkpoint_02_runtime.json",
        {
            "schema": "ddm_hp4_checkpoint.v1",
            "stage": "runtime",
            "rc64_library": file_record(library),
            "compiler": subprocess.run(
                ["cc", "--version"], check=False, capture_output=True, text=True
            ).stdout.splitlines()[0],
            "brotli_version": subprocess.run(
                [args.brotli, "--version"], check=False, capture_output=True, text=True
            ).stdout.strip(),
        },
    )

    predictor_state: dict[str, tuple[np.ndarray | None, np.ndarray | None, bytes]] = {}
    for predictor in PREDICTOR_IDS:
        features = feature_bank.get(predictor)
        weights = fit_integer_weights(source_values, features) if features is not None else None
        parameters = weights.tobytes() if weights is not None else b""
        predictor_root = output / "predictors" / predictor
        atomic_bytes(predictor_root / "parameters.bin", parameters)
        if weights is not None:
            atomic_npy(predictor_root / "weights.int8.npy", weights)
        residuals = predictor_residuals(source_values, predictor, features, weights)
        restored = restore_values(residuals, predictor, features, weights)
        if not np.array_equal(restored, source_values):
            raise RuntimeError(f"{predictor} exact teacher-forcing reconstruction failed")
        raw = pack_nibbles(residuals.reshape(-1))
        atomic_bytes(predictor_root / "embedding_residual.nibbles.bin", raw)
        atomic_bytes(predictor_root / "embedding_residual.repeat.nibbles.bin", raw)
        predictor_state[predictor] = (features, weights, raw)

    rows = []
    for predictor, (features, weights, raw_residual) in predictor_state.items():
        parameters = weights.tobytes() if weights is not None else b""
        for codec in CODEC_IDS:
            candidate_root = output / "candidates" / predictor / codec
            payload = encode_codec(
                raw_residual,
                codec,
                args.brotli,
                rc64_module,
                library,
            )
            atomic_bytes(candidate_root / "embedding.payload.bin", payload)
            payload_repeat = encode_codec(
                raw_residual,
                codec,
                args.brotli,
                rc64_module,
                library,
            )
            atomic_bytes(candidate_root / "embedding.repeat.payload.bin", payload_repeat)
            row = persist_candidate(
                output,
                predictor,
                codec,
                raw_residual,
                parameters,
                remainder_stream,
                semantic_stream,
                carrier_stream,
                payload,
                payload_repeat,
                compact_residual,
                tokens,
                args.brotli,
                rc64_module,
                library,
                features,
                weights,
                source_values,
                hpac_body,
            )
            rows.append(row)

    winner = min(rows, key=lambda row: (row["artifacts"]["archive"]["bytes"], row["candidate"]))
    current_size = {
        "physical_embedding_bytes": FRAME_BYTES,
        "incumbent_hpac_stream_bytes": len(hpac_stream),
        "incumbent_complete_archive_bytes": EXPECTED["archive"][0],
        "zero_field_counterfactual_complete_archive_bytes": (zero_root / "archive.zip").stat().st_size,
        "conditional_marginal_vs_zero_field_bytes": EXPECTED["archive"][0] - (zero_root / "archive.zip").stat().st_size,
        "boundary": (
            "The incumbent outer Brotli stream jointly codes all HPAC fields, so the 600x8 field "
            "has no separable literal compressed span. The conditional marginal is a retained real "
            "complete-container zero-field counterfactual, not a valid output candidate."
        ),
    }
    result = {
        "schema": "ddm_hp4_frame_embedding_prediction_result.v1",
        "axis": AXIS,
        "score_claim": False,
        "scorer_ran": False,
        "modal_ran": False,
        "mps_ran": False,
        "population": FRAME_COUNT,
        "frame_embedding_shape": [FRAME_COUNT, FRAME_DIM],
        "source_archive": sources["archive"],
        "source_hpac_body": file_record(incumbent_root / "hpac_body.bin"),
        "source_embedding": file_record(incumbent_root / "embedding.int4.bin"),
        "source_decoded_partition_state": sources["decoded_tokens"],
        "source_counted_carrier_coefficients": sources["coefficients"],
        "current_coded_size": current_size,
        "candidate_denominator": len(rows),
        "predictor_denominator": len(PREDICTOR_IDS),
        "codec_denominator": len(CODEC_IDS),
        "candidate_rows": rows,
        "winner": winner,
        "verdict_scope": "FORMULATION: tested causal integer predictors and exact listed coder forms on the pinned CP135 HP3-step2 embedding",
        "falsifier_fired": winner["delta_bytes_vs_cp135"] >= 0,
        "formulation_verdict": (
            "CLOSED_COMPLETE_CONTAINER_DELTA_NONNEGATIVE"
            if winner["delta_bytes_vs_cp135"] >= 0
            else "SURVIVES_COMPLETE_CONTAINER_BYTE_WIN"
        ),
        "all_payloads_retained": True,
        "deterministic_repeats_passed": all(row["repeat_byte_identical"] for row in rows),
        "causality": {
            "previous_embedding": "only reconstructed rows < t",
            "pose_feature": "counted carrier coefficient row t, decoded from the model section before token decode",
            "partition_feature": "class histograms from decoded token rows <= t-1 only",
            "future_state_used": False,
            "teacher_forcing_identity": "retained decoded token state supplies the exact prior partitions; every embedding row is reconstructed byte-identically",
        },
        "reproducibility": {
            "run_context": file_record(started_path),
            "runner": file_record(Path(__file__)),
            "rc64_python": file_record(RC64_PY),
            "rc64_c": file_record(RC64_C),
            "rc64_library": file_record(library),
            "brotli": subprocess.run(
                [args.brotli, "--version"], check=False, capture_output=True, text=True
            ).stdout.strip(),
            "numpy": np.__version__,
            "python": sys.version,
        },
        "boundaries": [
            "No scorer, renderer, evaluate.py, Modal, MPS, CUDA, or network operation ran.",
            "The exact contest pointer cannot move from this byte-only row.",
            "The zero-field counterfactual prices a conditional marginal but is not a semantically valid candidate.",
            "The HP4 receiver seam reconstructs the exact HPAC bytes; production F26 integration was not needed for the scorer-free lossless verdict.",
        ],
    }
    final_path = output / "FINAL_RESULT.json"
    atomic_json(final_path, result)
    atomic_json(
        output / "checkpoint_03_candidates_complete.json",
        {
            "schema": "ddm_hp4_checkpoint.v1",
            "stage": "candidates_complete",
            "candidate_denominator": len(rows),
            "winner": winner,
            "final_result": file_record(final_path),
        },
    )
    manifest_rows = []
    for path in sorted(candidate for candidate in output.rglob("*") if candidate.is_file()):
        if path.name == "ARTIFACT_MANIFEST.json":
            continue
        manifest_rows.append({"relative_path": path.relative_to(output).as_posix(), **file_record(path)})
    manifest = {
        "schema": "ddm_hp4_artifact_manifest.v1",
        "root": str(output),
        "file_denominator": len(manifest_rows),
        "files": manifest_rows,
    }
    atomic_json(output / "ARTIFACT_MANIFEST.json", manifest)
    elapsed = time.monotonic()
    print(
        json.dumps(
            {
                "winner": winner["candidate"],
                "archive_bytes": winner["artifacts"]["archive"]["bytes"],
                "delta_bytes": winner["delta_bytes_vs_cp135"],
                "final_result": file_record(final_path),
                "manifest": file_record(output / "ARTIFACT_MANIFEST.json"),
                "monotonic_end": elapsed,
            },
            sort_keys=True,
        )
    )


if __name__ == "__main__":
    main()
