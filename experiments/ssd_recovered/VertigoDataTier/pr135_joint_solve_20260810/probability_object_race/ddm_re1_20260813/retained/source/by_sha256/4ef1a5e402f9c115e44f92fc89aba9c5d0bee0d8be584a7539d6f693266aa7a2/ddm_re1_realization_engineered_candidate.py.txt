This runner does not call SegNet, PoseNet, Modal, or ``upstream/evaluate.py``.
It consumes the retained VD1 full-n600 singleton measurements, starts from the
strongest exact singleton projection, then adds the best negative projected event
from each distinct affected pair.  The projection is explicitly tied to an
alternate component-only surface and is not comparable to the canonical pointer.  Both proposal
rounds are independently HP3/RC64 re-encoded, parsed, and decoded by the shipped
RC64 receiver.  They remain proposals until the sole full-n600 scorer owner
returns a whole-candidate receipt.

class RE1Error(RuntimeError):
    """An RE1 custody, selection, resume, or receiver invariant failed."""


def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat().replace("+00:00", "Z")

def _require_record(record: dict[str, Any]) -> None:
    path = Path(record["path"])
    if not path.is_file() or jo1.file_record(path) != record:
        raise RE1Error(f"retained artifact failed custody: {path}")


def retain_content_addressed_bytes(
    root: Path,
    name: str,
    payload: bytes,
    *,
    executable: bool = False,
) -> dict[str, Any]:
    """Write once by content hash and fail if an existing version was altered."""

    digest = jo1.sha256_bytes(payload)
    path = root / "by_sha256" / digest / name
    if path.exists():
        if not path.is_file() or path.read_bytes() != payload:
            raise RE1Error(f"content-addressed artifact drifted: {path}")
    else:
        jo1.atomic_bytes(path, payload, executable=executable)
    return jo1.file_record(path)


def source_custody(output: Path) -> dict[str, Any]:
    """Bind immutable producing bytes separately from the current receipt writer."""

    source_root = output / "retained/source"
    producing_path = (
        source_root
        / "by_sha256"
        / PRODUCING_SOURCE_SHA256
        / "ddm_re1_realization_engineered_candidate.py"
    )
    jo1.require(
        producing_path,
        size=PRODUCING_SOURCE_BYTES,
        digest=PRODUCING_SOURCE_SHA256,
    )
    producing = jo1.file_record(producing_path)
    recovery_path = source_root / "recovery/PRODUCING_SOURCE_RECOVERY.json"
    jo1.require(recovery_path, digest=PRODUCING_SOURCE_RECOVERY_SHA256)
    recovery = jo1.file_record(recovery_path)
    current = retain_content_addressed_bytes(
        source_root,
        Path(__file__).name,
        Path(__file__).read_bytes(),
        executable=True,
    )
    return {
        "payload_producing_runner_source": producing,
        "producing_source_recovery_receipt": recovery,
        "receipt_refresh_runner_source": current,
        "legacy_mutable_source_path": str((source_root / Path(__file__).name).resolve()),
        "legacy_mutable_source_is_not_producer_authority": True,
    }


def runtime_tree_record(runtime_root: Path, source_root: Path = jo1.CP135_RUNTIME) -> dict[str, Any]:
    """Reject extra/missing runtime files, then hash the exact candidate tree."""

    def relative_files(root: Path) -> set[str]:
        return {
            path.relative_to(root).as_posix()
            for path in root.rglob("*")
            if path.is_file() and "__pycache__" not in path.parts and path.suffix != ".pyc"
        }

    expected = relative_files(source_root)
    actual = relative_files(runtime_root)
    if actual != expected:
        raise RE1Error(
            "candidate runtime file set drifted: "
            f"missing={sorted(expected - actual)} extra={sorted(actual - expected)}"
        )
    tree = jo1.cp135.tree_record(runtime_root)
    return {
        "schema": "ddm_re1_runtime_tree.v1",
        "tree": tree,
        "inflate_sh": jo1.file_record(runtime_root / "inflate.sh"),
        "inflate_py": jo1.file_record(runtime_root / "inflate.py"),
        "archive": jo1.file_record(runtime_root / "archive.zip"),
        "source_runtime": str(source_root.resolve()),
    }


@contextmanager
def exclusive_arm_lock(output: Path) -> Iterator[dict[str, Any]]:
    """Hold one nonblocking lock over every write to the RE1 target store."""

    output.mkdir(parents=True, exist_ok=True)
    lock_path = output / ".ddm_re1.lock"
    stream = lock_path.open("a+")
    try:
        try:
            fcntl.flock(stream.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
        except BlockingIOError as exc:
            raise RE1Error(f"RE1 target store is locked: {lock_path}") from exc
        receipt = {
            "schema": "ddm_re1_exclusive_lock_receipt.v1",
            "lock_path": str(lock_path.resolve()),
            "pid": os.getpid(),
            "acquired_utc": utc_now(),
            "scope": str(output.resolve()),
            "exclusive_nonblocking_flock": True,
        }
        jo1.atomic_json(output / "00_LOCK_RECEIPT.json", receipt)
        yield receipt
    finally:
        try:
            fcntl.flock(stream.fileno(), fcntl.LOCK_UN)
        finally:
            stream.close()


def singleton_component_projection_delta(
    row: dict[str, Any],
    *,
    base_d_pose: float = PROJECTION_BASE_D_POSE,
) -> float:
    """Project a singleton Seg+Pose delta on PO1's alternate component surface."""

    delta_flips = int(row["delta_flips_candidate_minus_base"])
    delta_pose = float(row["delta_d_pose_global_n600"])
    if base_d_pose <= 0.0 or base_d_pose + delta_pose < 0.0:
        raise RE1Error("singleton pose delta leaves the score domain")
    seg_term = 100.0 * delta_flips / TOTAL_SEG_SITES
    pose_term = math.sqrt(10.0 * (base_d_pose + delta_pose)) - math.sqrt(10.0 * base_d_pose)
    return seg_term + pose_term


def combined_additive_projection(rows: list[dict[str, Any]]) -> dict[str, float | int]:
    """Assume component additivity, then apply the nonlinear Pose score once."""

    delta_flips = sum(int(row["delta_flips_candidate_minus_base"]) for row in rows)
    delta_pose = sum(float(row["delta_d_pose_global_n600"]) for row in rows)
    if PROJECTION_BASE_D_POSE + delta_pose < 0.0:
        raise RE1Error("combined pose delta leaves the score domain")
    delta_s = 100.0 * delta_flips / TOTAL_SEG_SITES + (
        math.sqrt(10.0 * (PROJECTION_BASE_D_POSE + delta_pose))
        - math.sqrt(10.0 * PROJECTION_BASE_D_POSE)
    )
    errors = PROJECTION_BASE_SEG_ERRORS + delta_flips
    return {
        "delta_flips_candidate_minus_base": delta_flips,
        "candidate_seg_errors": errors,
        "candidate_d_seg_exact_from_errors": errors / TOTAL_SEG_SITES,
        "delta_d_pose_global_n600": delta_pose,
        "candidate_d_pose": PROJECTION_BASE_D_POSE + delta_pose,
        "delta_s": delta_s,
        "additivity_assumed": True,
        "projection_surface_only": True,
    }


def load_census_rows() -> list[dict[str, Any]]:
    jo1.require(CENSUS_FINAL, digest=CENSUS_FINAL_SHA256)
    jo1.require(CENSUS_EVENTS, size=646_298, digest=CENSUS_EVENTS_SHA256)
    rows: list[dict[str, Any]] = []
    seen: set[str] = set()
    with CENSUS_EVENTS.open(encoding="utf-8") as stream:
        for line in stream:
            row = json.loads(line)
            proposal_id = str(row.get("proposal_id", ""))
            if proposal_id in seen:
                raise RE1Error(f"duplicate VD1 singleton row: {proposal_id}")
            if (
                row.get("schema") != "ddm_vd1_singleton_result.v1"
                or row.get("n600_denominator_exact") is not True
                or row.get("score_claim") is not False
                or int(row.get("site_count", 0)) < 1
            ):
                raise RE1Error(f"inadmissible VD1 singleton row: {proposal_id}")
            seen.add(proposal_id)
            rows.append(row)
    if len(rows) != 200:
        raise RE1Error(f"VD1 singleton census is incomplete: {len(rows)}/200")
    return rows


def derive_round_plan(rows: list[dict[str, Any]]) -> dict[str, Any]:
    """Derive two deterministic proposals without claiming cross-surface score authority."""

    by_id = {str(row["proposal_id"]): row for row in rows}
    if set(CP5V_EVENT_IDS) - set(by_id):
        raise RE1Error("CP5V members are absent from the pinned VD1 census")
    scored = []
    for row in rows:
        projected_delta = singleton_component_projection_delta(row)
        scored.append(
            {
                "proposal_id": str(row["proposal_id"]),
                "ordinal": int(row["ordinal"]),
                "pair": int(row["pair"]),
                "delta_flips_candidate_minus_base": int(row["delta_flips_candidate_minus_base"]),
                "delta_d_pose_global_n600": float(row["delta_d_pose_global_n600"]),
                "site_count": int(row["site_count"]),
                "singleton_seg_pose_projection_delta_s": projected_delta,
                "downstream_selection_eligible": bool(row["downstream_selection_eligible"]),
            }
        )
    negative_rows = sorted(
        (
            row
            for row in scored
            if row["site_count"] == 1
            and row["singleton_seg_pose_projection_delta_s"] < 0.0
        ),
        key=lambda row: (
            row["singleton_seg_pose_projection_delta_s"],
            row["ordinal"],
            row["proposal_id"],
        ),
    )
    negative_ids = [str(row["proposal_id"]) for row in negative_rows]
    expected_negative = [
        CP5V_EVENT_IDS[0],
        CP5V_EVENT_IDS[1],
        CP5V_EVENT_IDS[2],
        "ec1_0120_463b0cb756b2",
    ]
    if negative_ids != expected_negative:
        raise RE1Error(f"current singleton projection set drifted: {negative_ids}")

    cp5v_singleton_sum = sum(
        singleton_component_projection_delta(by_id[value]) for value in CP5V_EVENT_IDS
    )
    cp5v_report_point_delta = CP5V_REPORT_POINT_SCORE - CANONICAL_POINTER_SCORE
    cp5v_residual_point = cp5v_report_point_delta - cp5v_singleton_sum
    report_bound = CP5V_REPORT_SCORE_ERROR_BOUND + CANONICAL_REPORT_SCORE_ERROR_BOUND
    round_1_rows = [negative_rows[0]]
    best_by_pair: dict[int, dict[str, Any]] = {}
    for row in negative_rows:
        best_by_pair.setdefault(int(row["pair"]), row)
    round_2_rows = sorted(
        best_by_pair.values(),
        key=lambda row: (
            row["singleton_seg_pose_projection_delta_s"],
            row["ordinal"],
            row["proposal_id"],
        ),
    )
    round_1_ids = [str(row["proposal_id"]) for row in round_1_rows]
    round_2_ids = [str(row["proposal_id"]) for row in round_2_rows]
    if round_1_ids != [CP5V_EVENT_IDS[0]] or round_2_ids != [
        CP5V_EVENT_IDS[0],
        CP5V_EVENT_IDS[2],
        "ec1_0120_463b0cb756b2",
    ]:
        raise RE1Error(f"distinct-pair round set drifted: {round_2_ids}")
    round_1_components = combined_additive_projection(round_1_rows)
    round_2_components = combined_additive_projection(round_2_rows)
    round_1_projected_delta = float(round_1_components["delta_s"])
    round_2_projected_delta = float(round_2_components["delta_s"])
    if round_2_projected_delta >= round_1_projected_delta:
        raise RE1Error("second proposal round does not improve its triage projection")

    def max_break_even_bytes(delta_s: float) -> int:
        crossing = abs(delta_s) * RATE_DENOMINATOR / 25.0
        return max(0, math.ceil(crossing) - 1)

    return {
        "schema": "ddm_re1_round_plan.v2",
        "canonical_pointer_score": CANONICAL_POINTER_SCORE,
        "projection_surface": {
            "base_d_pose": PROJECTION_BASE_D_POSE,
            "base_seg_errors": PROJECTION_BASE_SEG_ERRORS,
            "base_score_reconstructed": (
                100.0 * PROJECTION_BASE_SEG_ERRORS / TOTAL_SEG_SITES
                + math.sqrt(10.0 * PROJECTION_BASE_D_POSE)
                + 25.0 * jo1.BASE_BYTES / RATE_DENOMINATOR
            ),
            "raw_sha256": PROJECTION_RAW_SHA256,
            "canonical_raw_sha256": CANONICAL_RAW_SHA256,
            "raw_matches_canonical": False,
            "axis_status": "BLOCKED_AXIS_MISMATCH",
            "allowed_use": "within-surface proposal triage only",
        },
        "seg_denominator": TOTAL_SEG_SITES,
        "incumbent": {
            "name": "cp5v_validated_five",
            "proposal_ids": list(CP5V_EVENT_IDS),
            "canonical_report_component_point_score": CP5V_REPORT_POINT_SCORE,
            "point_delta_vs_cp135": cp5v_report_point_delta,
            "report_score_error_bound": CP5V_REPORT_SCORE_ERROR_BOUND,
            "status": "MEASURED_REPORT_POINT_ABOVE_POINTER_NOT_PROMOTED",
        },
        "cp5v_composition_residual": {
            "sum_of_singleton_component_projections": cp5v_singleton_sum,
            "whole_candidate_report_point_delta": cp5v_report_point_delta,
            "residual_point_estimate": cp5v_residual_point,
            "report_rounding_bound_at_least": report_bound,
            "rounding_interval": [cp5v_residual_point - report_bound, cp5v_residual_point + report_bound],
            "projection_and_report_surfaces_match": False,
            "sign_resolved": False,
            "load_bearing": False,
        },
        "rounds": [
            {
                "name": ROUND_NAMES[0],
                "proposal_ids": round_1_ids,
                "selection": "strongest VD1 exact-input singleton projection on the PO1 surface",
                "singleton_rows": round_1_rows,
                "combined_additive_projection": round_1_components,
                "projected_seg_pose_delta_s": round_1_projected_delta,
                "max_delta_bytes_on_projection_surface": max_break_even_bytes(
                    round_1_projected_delta
                ),
                "acceptance_status": "UNACCEPTED_PENDING_WHOLE_CANDIDATE_SCORE",
            },
            {
                "name": ROUND_NAMES[1],
                "proposal_ids": round_2_ids,
                "selection": (
                    "best negative singleton projection per distinct affected pair; one-best-per-pair "
                    "is the predeclared structural deduplication rule"
                ),
                "singleton_rows": round_2_rows,
                "combined_additive_projection": round_2_components,
                "projected_seg_pose_delta_s": round_2_projected_delta,
                "max_delta_bytes_on_projection_surface": max_break_even_bytes(
                    round_2_projected_delta
                ),
                "acceptance_status": "UNACCEPTED_PENDING_WHOLE_CANDIDATE_SCORE",
            },
        ],
        "projection_is_not_pointer_comparable": True,
        "projection_is_not_acceptance": True,
        "scorer_run": False,
        "score_claim": False,
    }

            "live_queue_ingested": False,
            "scorer_run": False,
            "score_claim": False,
        },
    )


def validate_evidence_bindings() -> dict[str, Any]:
    """Pin and cross-check every score/component claim consumed by proposal triage."""

    pinned = (
        (CP135_AUTH_RECEIPT, CP135_AUTH_RECEIPT_SHA256),
        (CANONICAL_POINTER, CANONICAL_POINTER_SHA256),
        (PO1_COMPONENT_RECEIPT, PO1_COMPONENT_RECEIPT_SHA256),
        (JS1B_AXIS_MISMATCH_RECEIPT, JS1B_AXIS_MISMATCH_RECEIPT_SHA256),
        (CP5V_COMPOSE, CP5V_COMPOSE_SHA256),
        (CP5V_FINAL, CP5V_FINAL_SHA256),
        (CP5V_EXACT, CP5V_EXACT_SHA256),
    )
    for path, digest in pinned:
        jo1.require(path, digest=digest)

    cp135 = json.loads(CP135_AUTH_RECEIPT.read_text())
    pointer = json.loads(CANONICAL_POINTER.read_text())
    po1 = json.loads(PO1_COMPONENT_RECEIPT.read_text())
    js1b = json.loads(JS1B_AXIS_MISMATCH_RECEIPT.read_text())
    cp5v_compose = json.loads(CP5V_COMPOSE.read_text())
    cp5v_final = json.loads(CP5V_FINAL.read_text())
    cp5v_exact = json.loads(CP5V_EXACT.read_text())

    if (
        float(cp135["canonical_score"]) != CANONICAL_POINTER_SCORE
        or cp135["provenance"]["archive_sha256"] != jo1.BASE_SHA256
        or int(cp135["n_samples"]) != 600
        or cp135["score_claim_valid"] is not True
    ):
        raise RE1Error("CP135 authority receipt drifted")
    effective = pointer["effective_frontier"]
    if (
        float(effective["score"]) != CANONICAL_POINTER_SCORE
        or effective["archive_sha256"] != jo1.BASE_SHA256
        or effective["axis"] != "contest_cuda"
    ):
        raise RE1Error("canonical frontier pointer drifted")
    if (
        float(po1["pose_feedback"]["d_pose_decoded_first"]) != PROJECTION_BASE_D_POSE
        or int(po1["seg_feedback"]["flips"]) != PROJECTION_BASE_SEG_ERRORS
        or po1["receiver"]["raw"]["sha256"] != PROJECTION_RAW_SHA256
        or js1b["axis_adjudication"]["status"] != "BLOCKED_AXIS_MISMATCH"
        or js1b["axis_adjudication"]["admitted_for_js1_stage0"] is not False
    ):
        raise RE1Error("PO1 projection surface or JS1B axis verdict drifted")
    cp5v_archive_sha = cp5v_compose["primary"]["sha256"]
    if (
        cp5v_compose["event_ids"] != list(CP5V_EVENT_IDS)
        or cp5v_final["event_ids"] != list(CP5V_EVENT_IDS)
        or cp5v_final["archive"]["sha256"] != cp5v_archive_sha
        or cp5v_exact["provenance"]["archive_sha256"] != cp5v_archive_sha
        or float(cp5v_exact["canonical_score"]) != CP5V_REPORT_POINT_SCORE
    ):
        raise RE1Error("CP5V composition-to-score binding drifted")
    return {
        "cp135_authority": jo1.file_record(CP135_AUTH_RECEIPT),
        "canonical_pointer_snapshot": jo1.file_record(CANONICAL_POINTER),
        "po1_component_surface": jo1.file_record(PO1_COMPONENT_RECEIPT),
        "js1b_axis_mismatch": jo1.file_record(JS1B_AXIS_MISMATCH_RECEIPT),
        "cp5v_compose": jo1.file_record(CP5V_COMPOSE),
        "cp5v_final": jo1.file_record(CP5V_FINAL),
        "cp5v_exact": jo1.file_record(CP5V_EXACT),
    }


def preflight(output: Path) -> dict[str, Any]:
    output.mkdir(parents=True, exist_ok=True)
    if output.resolve() != OUTPUT.resolve():
        raise RE1Error("RE1 resumes only from its exact chartered target store")
    usage = shutil.disk_usage(output)
    if usage.free < MIN_FREE_BYTES:
        raise RE1Error("SSD storage preflight failed closed")
    jo1.require(JO1_SOURCE, digest=JO1_SOURCE_SHA256)
    jo1.require(CP135_SOURCE, digest=CP135_SOURCE_SHA256)
    jo1.require(jo1.BASE_ARCHIVE, size=jo1.BASE_BYTES, digest=jo1.BASE_SHA256)
    jo1.require(jo1.BASE_SPATIAL, size=jo1.TOTAL_EVENTS, digest=jo1.BASE_SPATIAL_SHA256)
    evidence = validate_evidence_bindings()
    rows = load_census_rows()
    plan = derive_round_plan(rows)
    sources = source_custody(output)
    plan_payload = (json.dumps(plan, indent=2, sort_keys=True) + "\n").encode()
    versioned_plan = retain_content_addressed_bytes(
        output / "retained/plans",
        "ROUND_PLAN.json",
        plan_payload,
    )
    plan_record = jo1.atomic_json(output / "10_ROUND_PLAN.json", plan)
    result = {
        "schema": "ddm_re1_preflight.v2",
        "axis": AXIS,
        "git_head": git_head(),
        "free_bytes": usage.free,
        "required_free_bytes": MIN_FREE_BYTES,
        "target_store": str(output.resolve()),
        "base_archive": jo1.file_record(jo1.BASE_ARCHIVE),
        "base_spatial_tokens": jo1.file_record(jo1.BASE_SPATIAL),
        "jo1_source": jo1.file_record(JO1_SOURCE),
        "cp135_source": jo1.file_record(CP135_SOURCE),
        "vd1_final": jo1.file_record(CENSUS_FINAL),
        "vd1_event_results": jo1.file_record(CENSUS_EVENTS),
        "evidence_bindings": evidence,
        "round_plan": plan_record,
        "content_addressed_round_plan": versioned_plan,
        **sources,
        "resume_policy": (
            "candidate payloads are reused only after record and proposal-ID validation; current "
            "source and plans are append-only content-addressed versions; top-level receipts refresh"
        ),
        "scorer_slot_owned": False,
        "scorer_run": False,
        "score_claim": False,
        "all_materialized_payloads_retained": True,
    }
    jo1.atomic_json(output / "05_PREFLIGHT.json", result)
    return result


def _retained_reclose(output: Path, name: str, *, repeat: bool) -> dict[str, Any] | None:
    root = jo1.candidate_root(output, name, repeat=repeat)
    path = root / "50_RECLOSE_RESULT.json"
    if not path.is_file():
        return None
    result = json.loads(path.read_text())
    for key in (
        "archive",
        "token",
        "member",
        "models",
        "residual",
        "probability_export",
        "rc64_result",
        "encoder_verifier_backend",
        "shipped_receiver_backend",
        "shipped_receiver_parseback",
        "shipped_decoded_spatial_tokens",
    ):
        _require_record(result[key])
    if (
        result.get("schema") != "ddm_jo1_reclose_result.v1"
        or result.get("receiver_closed") is not True
        or result.get("score_claim") is not False
    ):
        raise RE1Error(f"retained reclose receipt is inadmissible: {path}")
    return result


def _materialize_and_reclose(
    output: Path,
    name: str,
    proposal_ids: list[str],
    *,
    repeat: bool,
) -> dict[str, Any]:
    retained = _retained_reclose(output, name, repeat=repeat)
    if retained is not None:
        prepared = json.loads(
            (jo1.candidate_root(output, name, repeat=repeat) / "20_MATERIALIZE_RESULT.json").read_text()
        )
        for key in ("spatial_tokens", "event_order", "source_manifest", "event_applications"):
            _require_record(prepared[key])
        if (
            prepared.get("schema") != "ddm_jo1_materialized_candidate.v1"
            or prepared.get("proposal_ids") != proposal_ids
            or prepared.get("repeat") is not repeat
        ):
            raise RE1Error(f"retained round proposal set drifted: {name}")
        return retained
    jo1.materialize_candidate(output, name, proposal_ids, repeat=repeat)
    return jo1.reclose_candidate(output, name, repeat=repeat)


def _actual_token_diffs(base: np.ndarray, candidate: np.ndarray) -> list[dict[str, int]]:
    rows: list[dict[str, int]] = []
    for frame in range(jo1.FRAMES):
        before = np.asarray(base[frame]).reshape(-1)
        after = np.asarray(candidate[frame]).reshape(-1)
        for index in np.flatnonzero(before != after).tolist():
            rows.append(
                {
                    "frame": frame,
                    "index": int(index),
                    "source_class": int(before[index]),
                    "target_class": int(after[index]),
                }
            )
    return rows


def _expected_token_diffs(applications: list[dict[str, Any]]) -> list[dict[str, int]]:
    rows: list[dict[str, int]] = []
    for application in applications:
        for index in application["indices"]:
            rows.append(
                {
                    "frame": int(application["frame"]),
                    "index": int(index),
                    "source_class": int(application["source_class"]),
                    "target_class": int(application["target_class"]),
                }
            )
    return sorted(rows, key=lambda row: (row["frame"], row["index"]))


def build_round(output: Path, round_spec: dict[str, Any]) -> dict[str, Any]:
    name = str(round_spec["name"])
    proposal_ids = [str(value) for value in round_spec["proposal_ids"]]
    primary = _materialize_and_reclose(output, name, proposal_ids, repeat=False)
    primary_root = jo1.candidate_root(output, name)
    archive_path = Path(primary["archive"]["path"])
    runtime_root = primary_root / "adapted_runtime"
    runtime = jo1.copy_runtime(runtime_root, archive_path.read_bytes())
    runtime_tree = runtime_tree_record(runtime_root)
    runtime_tree_receipt = jo1.atomic_json(primary_root / "70_RUNTIME_TREE.json", runtime_tree)
    repeated = _materialize_and_reclose(output, name, proposal_ids, repeat=True)
    same_archive = primary["archive"] == {
        **repeated["archive"],
        "path": primary["archive"]["path"],
    }
    same_token = primary["token"]["sha256"] == repeated["token"]["sha256"]
    same_spatial = (
        primary["shipped_decoded_spatial_tokens"]["sha256"]
        == repeated["shipped_decoded_spatial_tokens"]["sha256"]
    )
    applications = json.loads((primary_root / "EVENT_APPLICATIONS.json").read_text())["rows"]
    expected = _expected_token_diffs(applications)
    base = np.memmap(
        jo1.BASE_SPATIAL,
        mode="r",
        dtype=np.uint8,
        shape=(jo1.FRAMES, jo1.HEIGHT, jo1.WIDTH),
    )
    candidate = np.memmap(
        Path(primary["shipped_decoded_spatial_tokens"]["path"]),
        mode="r",
        dtype=np.uint8,
        shape=(jo1.FRAMES, jo1.HEIGHT, jo1.WIDTH),
    )
    actual = _actual_token_diffs(base, candidate)
    exact_diff = actual == expected and len(actual) == len(expected)
    delta_bytes = int(primary["delta_bytes_vs_cp135"])
    projected_delta = float(round_spec["projected_seg_pose_delta_s"])
    projection_with_rate = projected_delta + 25.0 * delta_bytes / RATE_DENOMINATOR
    rate_screen = delta_bytes <= int(round_spec["max_delta_bytes_on_projection_surface"])
    diff_result = {
        "schema": "ddm_re1_exact_token_diff.v1",
        "round": name,
        "base": jo1.file_record(jo1.BASE_SPATIAL),
        "candidate": primary["shipped_decoded_spatial_tokens"],
        "expected": expected,
        "actual": actual,
        "exact_requested_cells_and_nowhere_else": exact_diff,
    }
    diff_record = jo1.atomic_json(primary_root / "60_EXACT_TOKEN_DIFF.json", diff_result)
    preflight_receipt = json.loads((output / "05_PREFLIGHT.json").read_text())
    result = {
        "schema": "ddm_re1_round_result.v2",
        "round": name,
        "proposal_ids": proposal_ids,
        "selection_evidence": round_spec,
        "archive": primary["archive"],
        "independent_repeat": repeated["archive"],
        "archive_repeat_byte_identical": same_archive,
        "rc64_repeat_byte_identical": same_token,
        "spatial_repeat_byte_identical": same_spatial,
        "runtime": str(runtime_root.resolve()),
        "runtime_archive": runtime["archive"],
        "runtime_tree": runtime_tree,
        "runtime_tree_receipt": runtime_tree_receipt,
        "payload_producing_runner_source": preflight_receipt[
            "payload_producing_runner_source"
        ],
        "receipt_refresh_runner_source": preflight_receipt["receipt_refresh_runner_source"],
        "entropy_receiver_closed": bool(primary["receiver_closed"] and repeated["receiver_closed"]),
        "receiver_closed": False,
        "full_public_runtime_receiver_closed": False,
        "full_receiver_boundary": (
            "the shipped RC64 receiver decoded every token exactly, but the CUDA-locked public "
            "inflate.sh renderer has not produced 0.raw on this lane"
        ),
        "exact_token_diff": diff_record,
        "delta_bytes_vs_cp135": delta_bytes,
        "projected_delta_s_on_alternate_component_surface_plus_real_rate": projection_with_rate,
        "max_delta_bytes_on_projection_surface": round_spec[
            "max_delta_bytes_on_projection_surface"
        ],
        "projection_surface_rate_screen_passed": rate_screen,
        "canonical_pointer_comparison": "UNAVAILABLE_CROSS_SURFACE",
        "projection_authority": (
            "triage only on the PO1 component surface, whose raw differs from the canonical CP135 "
            "raw; it is not a pointer-comparable candidate verdict"
        ),
        "complete_s_acceptance": False,
        "acceptance_status": "UNACCEPTED_PENDING_WHOLE_CANDIDATE_SCORE",
        "decoded_raw_video_retained": False,
        "decoded_raw_video_boundary": (
            "the F26 renderer is CUDA-locked and this charter owns no scorer slot; raw and scorer "
            "fields are required returns of the store-local fire order pending MAIN ingestion"
        ),
        "scorer_run": False,
        "score_claim": False,
        "all_materialized_payloads_retained": True,
    }
    jo1.atomic_json(output / f"{name}_RESULT.json", result)
    if not all((same_archive, same_token, same_spatial, exact_diff, result["entropy_receiver_closed"])):
        jo1.atomic_json(
            output / f"{name}_FALSIFIER.json",
            {"schema": "ddm_re1_round_falsifier.v1", "status": "FIRED", **result},
        )
        raise RE1Error(f"receiver/determinism falsifier fired for {name}")
    return result

def all_stages(output: Path) -> dict[str, Any]:
    preflight(output)
    plan = json.loads((output / "10_ROUND_PLAN.json").read_text())
    rounds: list[dict[str, Any]] = []
    completed = ["preflight"]
    for round_spec in plan["rounds"]:
        name = str(round_spec["name"])
        save_state(output, name, completed)
        rounds.append(build_round(output, round_spec))
        completed.append(name)
    save_state(output, "finalize", completed)
    return finalize(output, rounds)


def parser() -> argparse.ArgumentParser:
    value = argparse.ArgumentParser(description=__doc__)
    value.add_argument("stage", choices=("preflight", "all"))
    value.add_argument("--output", type=Path, default=OUTPUT)
    value.add_argument(
        "--resume-from",
        type=Path,
        required=True,
        help="must equal --output; the same path is used for first launch and resume",
    )
    return value


def main() -> None:
    args = parser().parse_args()
    if args.resume_from.resolve() != args.output.resolve():
        raise SystemExit("FATAL: --resume-from must equal --output")
    with exclusive_arm_lock(args.output):
        result = preflight(args.output) if args.stage == "preflight" else all_stages(args.output)
    print(json.dumps(result, indent=2, sort_keys=True), flush=True)


if __name__ == "__main__":
    main()
