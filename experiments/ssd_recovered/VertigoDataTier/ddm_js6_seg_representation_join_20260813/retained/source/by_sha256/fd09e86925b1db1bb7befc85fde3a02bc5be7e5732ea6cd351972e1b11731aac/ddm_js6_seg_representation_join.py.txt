the SSD.  This program does not import a scorer, invoke Modal, or claim a score.

class JS6Error(RuntimeError):
    """A pin, representation, receiver, retention, or admission invariant failed."""


@dataclass(frozen=True, slots=True)
class ComponentSpec:
    pair: int
    source_class: int
    target_class: int
    component_id: int
    component_pixels: int
    token_sites: tuple[int, ...]


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        while block := stream.read(8 * 1024 * 1024):
            digest.update(block)
    return digest.hexdigest()


def file_record(path: Path) -> dict[str, Any]:
    return {
        "path": str(path.resolve()),
        "bytes": path.stat().st_size,
        "sha256": sha256_file(path),
    }


def atomic_bytes(path: Path, payload: bytes) -> dict[str, Any]:
    path.parent.mkdir(parents=True, exist_ok=True)
    partial = path.with_name(f".{path.name}.{os.getpid()}.partial")
    try:
        with partial.open("wb") as stream:
            stream.write(payload)
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(partial, path)
    finally:
        partial.unlink(missing_ok=True)
    return file_record(path)


def retain_bytes(path: Path, payload: bytes) -> dict[str, Any]:
    """Write once, or verify that the already-retained bytes are identical."""

    if path.is_file():
        record = file_record(path)
        if record["bytes"] != len(payload) or record["sha256"] != hashlib.sha256(payload).hexdigest():
            raise JS6Error(f"refusing to overwrite a different retained payload: {path}")
        return record
    return atomic_bytes(path, payload)


def atomic_json(path: Path, value: Any) -> dict[str, Any]:
    payload = (json.dumps(value, indent=2, sort_keys=True, allow_nan=False) + "\n").encode()
    return atomic_bytes(path, payload)


def atomic_npy(path: Path, value: np.ndarray) -> dict[str, Any]:
    path.parent.mkdir(parents=True, exist_ok=True)
    partial = path.with_name(f".{path.name}.{os.getpid()}.partial")
    try:
        with partial.open("wb") as stream:
            np.save(stream, np.asarray(value), allow_pickle=False)
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(partial, path)
    finally:
        partial.unlink(missing_ok=True)
    return file_record(path)


def retain_npy(path: Path, value: np.ndarray) -> dict[str, Any]:
    """Persist an array without ever replacing different retained signal."""

    expected = np.asarray(value)
    if path.is_file():
        retained = np.load(path, mmap_mode="r", allow_pickle=False)
        if (
            retained.shape != expected.shape
            or retained.dtype != expected.dtype
            or not np.array_equal(retained, expected, equal_nan=True)
        ):
            raise JS6Error(f"refusing to overwrite a different retained array: {path}")
        return file_record(path)
    return atomic_npy(path, expected)


def require_record(record: dict[str, Any], *, beneath: Path | None = None) -> Path:
    path = Path(str(record.get("path", ""))).resolve()
    if beneath is not None:
        try:
            path.relative_to(beneath.resolve())
        except ValueError as exc:
            raise JS6Error(f"retained path escapes {beneath}: {path}") from exc
    if not path.is_file() or file_record(path) != record:
        raise JS6Error(f"retained artifact differs: {path}")
    return path


def event_type(source: int, target: int) -> int:
    if (source, target) in ((0, 1), (1, 0)):
        return ec1.EVENT_TYPE["lane_program_delta"]
    if target == 3:
        return ec1.EVENT_TYPE["island_birth"]
    if source == 3:
        return ec1.EVENT_TYPE["island_death"]
    return ec1.EVENT_TYPE["boundary_offset"]


def encode_join_payload(spec: ComponentSpec, component: np.ndarray) -> tuple[bytes, dict[str, Any]]:
    labels = np.zeros((1, H, W), dtype=np.uint8)
    labels[0, component] = 3
    worldsheet, metadata = encode_g1_movable_worldsheet(labels)
    decoded, decoded_metadata = decode_g1_movable_worldsheet(worldsheet, expected_pairs=1)
    if decoded_metadata.payload_sha256 != metadata.payload_sha256:
        raise JS6Error("G1 byte-canonical parse-back differs")
    indices = np.asarray(spec.token_sites, dtype=np.int64)
    event = ec1.proposal_payload(
        spec.pair,
        spec.source_class,
        spec.target_class,
        indices,
        event_type(spec.source_class, spec.target_class),
    )
    header = JOIN_HEADER.pack(
        JOIN_MAGIC,
        spec.pair,
        spec.source_class,
        spec.target_class,
        FAMILY_WORLDSHEET,
        event_type(spec.source_class, spec.target_class),
        len(worldsheet),
        len(event),
    )
    payload = header + worldsheet + event
    parsed = decode_join_payload(payload)
    if parsed["worldsheet"] != worldsheet or parsed["event"] != event:
        raise JS6Error("joined proposal parse-back differs")
    return payload, {
        "worldsheet_bytes": len(worldsheet),
        "worldsheet_sha256": hashlib.sha256(worldsheet).hexdigest(),
        "worldsheet_metadata": asdict(metadata),
        "worldsheet_component_symmetric_difference_pixels": int(
            np.count_nonzero(decoded[0] != component)
        ),
        "event_bytes": len(event),
        "event_sha256": hashlib.sha256(event).hexdigest(),
    }


def decode_join_payload(payload: bytes) -> dict[str, Any]:
    if len(payload) < JOIN_HEADER.size:
        raise JS6Error("joined proposal is truncated")
    magic, pair, source, target, family, encoded_event_type, g1_bytes, event_bytes = JOIN_HEADER.unpack_from(payload)
    if magic != JOIN_MAGIC or family != FAMILY_WORLDSHEET:
        raise JS6Error("joined proposal header differs")
    expected = JOIN_HEADER.size + g1_bytes + event_bytes
    if expected != len(payload):
        raise JS6Error("joined proposal length/trailing bytes differ")
    worldsheet = payload[JOIN_HEADER.size : JOIN_HEADER.size + g1_bytes]
    event = payload[JOIN_HEADER.size + g1_bytes :]
    decoded_mask, _ = decode_g1_movable_worldsheet(worldsheet, expected_pairs=1)
    event_row = ec1.decode_proposal(event)
    if event_row[:4] != (pair, source, target, encoded_event_type):
        raise JS6Error("joined EC1 address differs from its outer header")
    if int(np.count_nonzero(decoded_mask)) == 0 or len(event_row[4]) < 2:
        raise JS6Error("joined proposal is not a nonempty coupled object")
    return {
        "pair": pair,
        "source_class": source,
        "target_class": target,
        "family": "g1_worldsheet_event_edit",
        "event_type": ec1.EVENT_NAME[encoded_event_type],
        "worldsheet": worldsheet,
        "event": event,
        "component": decoded_mask[0],
        "token_sites": event_row[4],
    }


def edge_census(base: np.ndarray, gt: np.ndarray) -> dict[tuple[int, int], int]:
    if base.shape != (N, H, W) or gt.shape != base.shape:
        raise JS6Error("argmax field geometry differs")
    matrix = np.zeros((len(CLASSES), len(CLASSES)), dtype=np.int64)
    for source in range(len(CLASSES)):
        for target in range(len(CLASSES)):
            if source != target:
                matrix[source, target] = np.count_nonzero((base == source) & (gt == target))
    return {
        (source, target): int(matrix[source, target])
        for source in range(len(CLASSES))
        for target in range(len(CLASSES))
        if source != target and matrix[source, target]
    }


def component_specs(
    base: np.ndarray,
    gt: np.ndarray,
    tokens: np.ndarray,
) -> list[ComponentSpec]:
    """Derive coupled components without a radius, template, or sample constant.

    Each target pixel selects the nearest current-base token of the directed
    source class.  Ties follow SciPy's deterministic EDT index order.  This is
    the current receiver lattice itself, not an inherited spatial basis.
    """

    rows: list[ComponentSpec] = []
    structure = np.ones((3, 3), dtype=np.uint8)
    for pair in range(N):
        base_pair = np.asarray(base[pair])
        gt_pair = np.asarray(gt[pair])
        token_pair = np.asarray(tokens[pair])
        mismatch = base_pair != gt_pair
        if not np.any(mismatch):
            continue
        nearest: dict[int, np.ndarray] = {}
        for source in np.unique(base_pair[mismatch]).tolist():
            source = int(source)
            if np.any(token_pair == source):
                nearest[source] = ndimage.distance_transform_edt(
                    token_pair != source,
                    return_distances=False,
                    return_indices=True,
                )
        codes = np.unique(base_pair[mismatch].astype(np.int16) * len(CLASSES) + gt_pair[mismatch])
        for code in codes.tolist():
            source, target = divmod(int(code), len(CLASSES))
            if source not in nearest:
                continue
            component_labels, count = ndimage.label(
                (base_pair == source) & (gt_pair == target),
                structure=structure,
            )
            component_sizes = np.bincount(component_labels.reshape(-1), minlength=count + 1)
            nearest_y, nearest_x = nearest[source]
            for component_id in range(1, count + 1):
                ys, xs = np.nonzero(component_labels == component_id)
                token_sites = np.unique(nearest_y[ys, xs] * W + nearest_x[ys, xs])
                if token_sites.size < 2:
                    continue
                rows.append(
                    ComponentSpec(
                        pair=pair,
                        source_class=source,
                        target_class=target,
                        component_id=component_id,
                        component_pixels=int(component_sizes[component_id]),
                        token_sites=tuple(int(value) for value in token_sites.tolist()),
                    )
                )
    rows.sort(
        key=lambda row: (
            -row.component_pixels,
            -len(row.token_sites),
            row.pair,
            row.source_class,
            row.target_class,
            row.component_id,
        )
    )
    return rows


def reconstruct_component(spec: ComponentSpec, base: np.ndarray, gt: np.ndarray) -> np.ndarray:
    mask = (np.asarray(base[spec.pair]) == spec.source_class) & (
        np.asarray(gt[spec.pair]) == spec.target_class
    )
    labels, count = ndimage.label(mask, structure=np.ones((3, 3), dtype=np.uint8))
    if not 1 <= spec.component_id <= count:
        raise JS6Error("component ID escaped its re-derived edge field")
    component = labels == spec.component_id
    if int(np.count_nonzero(component)) != spec.component_pixels:
        raise JS6Error("component mass differs on re-derivation")
    return component


def render_candidate(semantic: Any, tokens: np.ndarray, pair: int) -> tuple[np.ndarray, ...]:
    import torch
    from torch.nn import functional

    with torch.inference_mode():
        pre_r = semantic(
            torch.from_numpy(np.asarray(tokens).copy())[None].long(),
            torch.tensor([pair], dtype=torch.long),
        )
        camera = (
            functional.interpolate(pre_r, size=(CAM_H, CAM_W), mode="bilinear", align_corners=False)
            .clamp(0.0, 255.0)
            .round()
            .to(torch.uint8)
        )
        scorer = functional.interpolate(
            camera.float(), size=(H, W), mode="bilinear", align_corners=False
        )
    return (
        pre_r[0].cpu().numpy().astype(np.float32, copy=False),
        camera[0].permute(1, 2, 0).cpu().numpy(),
        scorer[0].cpu().numpy().astype(np.float32, copy=False),
    )


def base_scorer_input(base_camera: np.ndarray) -> np.ndarray:
    import torch
    from torch.nn import functional

    with torch.inference_mode():
        camera = torch.from_numpy(np.asarray(base_camera).copy()).permute(2, 0, 1)[None].float()
        scorer = functional.interpolate(camera, size=(H, W), mode="bilinear", align_corners=False)
    return scorer[0].cpu().numpy().astype(np.float32, copy=False)


def preflight(run_root: Path, store: Path) -> dict[str, Any]:
    import torch

    run_root.mkdir(parents=True, exist_ok=True)
    store.mkdir(parents=True, exist_ok=True)
    usage = shutil.disk_usage(run_root)
    if usage.free < MIN_FREE_BYTES:
        raise JS6Error(f"storage preflight failed: {usage.free} < {MIN_FREE_BYTES}")
    inputs = {}
    for path, digest in PINNED.items():
        if not path.is_file() or sha256_file(path) != digest:
            raise JS6Error(f"pinned input differs: {path}")
        inputs[str(path.resolve())] = file_record(path)
    for path in (BASE_ARGMAX, GT_ARGMAX):
        field = np.load(path, mmap_mode="r", allow_pickle=False)
        if field.shape != (N, H, W) or field.dtype != np.uint8:
            raise JS6Error(f"argmax field geometry differs: {path}")
    with zipfile.ZipFile(ec1.CP135_ARCHIVE) as archive:
        cp135_members = archive.namelist()
    if cp135_members != ["p"]:
        raise JS6Error("CP135 archive member surface differs")
    result = {
        "schema": "ddm_js6_seg_representation_join_preflight.v1",
        "axis": AXIS,
        "score_claim": False,
        "scorer_invoked": False,
        "modal_invoked": False,
        "mps_invoked": False,
        "selection_mode": "full n600 census; no sample",
        "proposal_limit": PROPOSAL_LIMIT,
        "storage": {"free_bytes": usage.free, "required_bytes": MIN_FREE_BYTES},
        "runtime": {
            "platform": platform.platform(),
            "python": sys.version,
            "numpy": np.__version__,
            "scipy": scipy.__version__,
            "torch": torch.__version__,
            "rng": "none; deterministic full-population enumeration",
        },
        "inputs": inputs,
        "cp135_archive_members": cp135_members,
        "current_receiver_grammar_compatibility": {
            "g1_worldsheet_member": False,
            "shared_template_member": False,
            "lane_program_member": False,
            "meaning": (
                "v19 grammar objects are proposal-source objects only; the CP135 public receiver "
                "cannot consume them until a byte-closed adapter is built"
            ),
        },
    }
    atomic_json(run_root / "00_PREFLIGHT.json", result)
    source_payload = Path(__file__).read_bytes()
    source_sha = hashlib.sha256(source_payload).hexdigest()
    retain_bytes(
        run_root / f"retained/source/by_sha256/{source_sha}/ddm_js6_seg_representation_join.py",
        source_payload,
    )
    return result


def load_prior(
    receipt_path: Path,
    store: Path,
    *,
    expected_payload_sha256: str,
    producer_source_sha256: str,
) -> dict[str, Any] | None:
    if not receipt_path.is_file():
        return None
    row = json.loads(receipt_path.read_text())
    if (
        row.get("schema") != "ddm_js6_seg_representation_proposal.v3"
        or row.get("producer_source_sha256") != producer_source_sha256
    ):
        return None
    retained_payloads = row.get("retained_payloads")
    if not isinstance(retained_payloads, dict):
        raise JS6Error("current-source proposal receipt lacks its retained payload map")
    for record in retained_payloads.values():
        require_record(record, beneath=store)
    raw_record = retained_payloads.get("proposal.js6p")
    if not isinstance(raw_record, dict) or raw_record.get("sha256") != expected_payload_sha256:
        raise JS6Error("retained proposal payload differs from its freshly derived address")
    return row


def process_proposal(
    ordinal: int,
    spec: ComponentSpec,
    *,
    store: Path,
    semantic: Any,
    tokens: np.ndarray,
    base_raw: np.ndarray,
    base: np.ndarray,
    gt: np.ndarray,
    producer_source_sha256: str,
) -> dict[str, Any]:
    component = reconstruct_component(spec, base, gt)
    payload, representation = encode_join_payload(spec, component)
    proposal_id = f"js6_{ordinal:04d}_{hashlib.sha256(payload).hexdigest()[:12]}"
    root = store / "proposals" / proposal_id
    receipt_path = root / "proposal.json"
    prior = load_prior(
        receipt_path,
        store,
        expected_payload_sha256=hashlib.sha256(payload).hexdigest(),
        producer_source_sha256=producer_source_sha256,
    )
    if prior is not None:
        return prior
    parsed = decode_join_payload(payload)
    base_tokens = np.asarray(tokens[spec.pair])
    indices = np.asarray(parsed["token_sites"], dtype=np.int64)
    if not np.all(base_tokens.reshape(-1)[indices] == spec.source_class):
        raise JS6Error("proposal token source precondition differs")
    candidate_tokens = base_tokens.copy()
    candidate_tokens.reshape(-1)[indices] = spec.target_class
    candidate_pre_r, candidate_camera, candidate_scorer = render_candidate(
        semantic, candidate_tokens, spec.pair
    )
    expected_base_camera = np.asarray(base_raw[2 * spec.pair + 1])
    base_scorer = base_scorer_input(expected_base_camera)
    scorer_delta = candidate_scorer - base_scorer
    strong_support = np.max(np.abs(scorer_delta), axis=0) >= 1.0
    if not np.any(strong_support) or np.array_equal(candidate_camera, expected_base_camera):
        raise JS6Error("coupled proposal is not receiver-effective through R and uint8")
    exact_edge = (np.asarray(base[spec.pair]) == spec.source_class) & (
        np.asarray(gt[spec.pair]) == spec.target_class
    )
    all_errors = np.asarray(base[spec.pair]) != np.asarray(gt[spec.pair])
    target_mass = int(np.count_nonzero(strong_support & exact_edge))
    other_error_mass = int(np.count_nonzero(strong_support & all_errors & ~exact_edge))
    correct_field_mass = int(np.count_nonzero(strong_support & ~all_errors))
    raw = payload
    br = brotli.compress(payload, quality=11)
    xz = lzma.compress(payload, preset=9 | lzma.PRESET_EXTREME)
    coded_name, coded = min(
        (("raw", raw), ("brotli_q11", br), ("lzma9_extreme", xz)),
        key=lambda item: (len(item[1]), item[0]),
    )
    retained = {
        "proposal.js6p": retain_bytes(root / "proposal.js6p", raw),

def build_edge_table(
    census: dict[tuple[int, int], int], rows: list[dict[str, Any]]
) -> list[dict[str, Any]]:
    table = []
    for (source, target), mass in sorted(census.items(), key=lambda item: (-item[1], item[0])):
        edge_rows = [
            row
            for row in rows
            if row["source_class_id"] == source and row["target_class_id"] == target
        ]
        coded = [
            int(row["complete_s"]["delta_bytes"]["actual_selected_section_bytes"])
            for row in edge_rows
        ]
        median_coded = float(np.median(coded)) if coded else None
        table.append(
            {
                "source_class": CLASSES[source],
                "target_class": CLASSES[target],
                "directed_edge": f"{CLASSES[source]}->{CLASSES[target]}",
                "measured_flip_mass": mass,
                "measured_flip_mass_share": mass / sum(census.values()),
                "retained_proposals": len(edge_rows),
                "measured_total_selected_section_bytes": sum(coded),
                "measured_median_selected_section_bytes": median_coded,
                "derived_edge_priority_flip_mass_times_inverse_median_byte_price": (
                    mass / median_coded if median_coded else None
                ),
                "score_claim": False,
            }
        )
    table.sort(
        key=lambda row: (
            -(
                row["derived_edge_priority_flip_mass_times_inverse_median_byte_price"]
                or -1.0
            ),
            -row["measured_flip_mass"],
            row["directed_edge"],
        )
    )
    return table


def zero_pose_break_even_max_integer_bytes(target_mass: int) -> int:
    if isinstance(target_mass, bool) or not isinstance(target_mass, int) or target_mass < 0:
        raise JS6Error("target mass must be a nonnegative integer")
    return max(
        -1,
        math.ceil(4.0 * UNCOMPRESSED_BYTES * target_mass / PIXELS) - 1,
    )


def audit_store(
    store: Path,
    rows: list[dict[str, Any]],
    *,
    producer_source_sha256: str,
) -> dict[str, Any]:
    if len(rows) != PROPOSAL_LIMIT or len({row["proposal_id"] for row in rows}) != len(rows):
        raise JS6Error("proposal audit cardinality/uniqueness differs")
    exact_g1 = 0
    nonzero_target = 0
    for row in rows:
        if row.get("score_claim") is not False or row["complete_s"]["recomputed_score"] is not None:
            raise JS6Error("proposal audit found a score claim")
        if row.get("producer_source_sha256") != producer_source_sha256:
            raise JS6Error("proposal audit found a stale producer source")
        root = store / "proposals" / row["proposal_id"]
        for record in row["retained_payloads"].values():
            require_record(record, beneath=store)
        decoded = decode_join_payload((root / "proposal.js6p").read_bytes())
        if decoded["pair"] != row["pair"] or len(decoded["token_sites"]) != row["token_site_count"]:
            raise JS6Error("proposal audit decoded address differs")
        expected_arrays = {
            "target_component.bool.npy": ((H, W), np.dtype(np.bool_)),
            "g1_decoded_component.bool.npy": ((H, W), np.dtype(np.bool_)),
            "candidate_tokens.uint8.npy": ((H, W), np.dtype(np.uint8)),
            "candidate_pre_r.float32.npy": ((3, H, W), np.dtype(np.float32)),
            "candidate_pre_r.float16.npy": ((3, H, W), np.dtype(np.float16)),
            "base_camera.uint8.npy": ((CAM_H, CAM_W, 3), np.dtype(np.uint8)),
            "candidate_camera.uint8.npy": ((CAM_H, CAM_W, 3), np.dtype(np.uint8)),
            "base_scorer_input.float32.npy": ((3, H, W), np.dtype(np.float32)),
            "base_scorer_input.float16.npy": ((3, H, W), np.dtype(np.float16)),
            "candidate_scorer_input.float32.npy": ((3, H, W), np.dtype(np.float32)),
            "candidate_scorer_input.float16.npy": ((3, H, W), np.dtype(np.float16)),
            "scorer_delta.float32.npy": ((3, H, W), np.dtype(np.float32)),
            "scorer_delta.float16.npy": ((3, H, W), np.dtype(np.float16)),
            "strong_support.bool.npy": ((H, W), np.dtype(np.bool_)),
        }
        for name, (shape, dtype) in expected_arrays.items():
            value = np.load(root / name, mmap_mode="r", allow_pickle=False)
            if value.shape != shape or value.dtype != dtype:
                raise JS6Error(f"proposal audit array geometry differs: {root / name}")
        for stem in (
            "candidate_pre_r",
            "base_scorer_input",
            "candidate_scorer_input",
            "scorer_delta",
        ):
            exact = np.load(root / f"{stem}.float32.npy", mmap_mode="r", allow_pickle=False)
            compact = np.load(root / f"{stem}.float16.npy", mmap_mode="r", allow_pickle=False)
            if not np.array_equal(np.asarray(exact).astype(np.float16), compact):
                raise JS6Error(f"proposal audit float32/float16 parity differs: {root / stem}")
        base_scorer = np.load(root / "base_scorer_input.float32.npy", mmap_mode="r", allow_pickle=False)
        candidate_scorer = np.load(
            root / "candidate_scorer_input.float32.npy", mmap_mode="r", allow_pickle=False
        )
        delta = np.load(root / "scorer_delta.float32.npy", mmap_mode="r", allow_pickle=False)
        support = np.load(root / "strong_support.bool.npy", mmap_mode="r", allow_pickle=False)
        if not np.array_equal(np.asarray(candidate_scorer) - np.asarray(base_scorer), delta):
            raise JS6Error(f"proposal audit scorer delta differs: {root}")
        if not np.array_equal(np.max(np.abs(delta), axis=0) >= 1.0, support):
            raise JS6Error(f"proposal audit strong support differs: {root}")
        exact_g1 += int(row["g1_target_component_exact"])
        nonzero_target += int(
            row["receiver_surface"]["exact_field_target_edge_mass_on_support"] > 0
        )
    files = [path for path in store.rglob("*") if path.is_file()]
    return {
        "schema": "ddm_js6_seg_representation_join_store_audit.v1",
        "axis": "[local byte/hash/shape/parse-back audit; scorer-free]",
        "score_claim": False,
        "proposal_count": len(rows),
        "unique_proposal_count": len({row["proposal_id"] for row in rows}),
        "all_retained_payload_records_verified": True,
        "all_join_payloads_parse_back": True,
        "all_array_shapes_and_dtypes_verified": True,
        "all_float32_to_float16_parity_verified": True,
        "all_scorer_deltas_and_supports_recomputed": True,
        "all_score_claims_false": True,
        "all_producer_source_pins_current": True,
        "producer_source_sha256": producer_source_sha256,
        "g1_exact_target_components": exact_g1,
        "g1_approximate_target_components": len(rows) - exact_g1,
        "nonzero_exact_field_target_support_proposals": nonzero_target,
        "store_files": len(files),
        "store_bytes": sum(path.stat().st_size for path in files),
    }
