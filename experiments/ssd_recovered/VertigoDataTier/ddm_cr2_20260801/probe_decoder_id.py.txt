def packet_sections(pk):
    hdr=rt.PACKET_HEADER; ml=int(hdr.unpack_from(pk,0)[-1]); me=hdr.size+ml; out={}
    for i,n in enumerate(rt.SECTION_NAMES):
        _,off,ln,_=rt.SECTION_ENTRY.unpack_from(pk, me+i*rt.SECTION_ENTRY.size)
        out[n]=pk[off:off+ln]
    return out
