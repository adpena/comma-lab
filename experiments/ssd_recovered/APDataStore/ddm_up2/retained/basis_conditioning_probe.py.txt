import sys, numpy as np, torch
torch.set_num_threads(4)
sys.path.insert(0,"/Users/adpena/Projects/pact/experiments"); sys.path.insert(0,"/Users/adpena/Projects/pact/src")
import ddm_up2_shipping_pose_solve as up2
from torch.nn import functional
up2.enable_posenet_gradients()
st=up2.load_carrier_state(up2.DEFAULT_RUNTIME); raw=up2.open_raw(up2.DEFAULT_RAW,verify_sha=False)
tgt,_=up2.load_gt_poses(up2.DEFAULT_DALI_GT); posenet=up2.load_posenet()
rng=np.random.default_rng(1234); pairs=np.sort(rng.choice(600,6,replace=False)).astype(np.int64)
AMP=64.0/np.sqrt(12)
print(f"{'pair':>5} {'|r|':>9} {'12dof_LSBrms':>13} {'12dof_codes':>12} {'2304dof_LSBrms':>15} {'penalty':>9}")
tot=[]
for p in pairs:
    idx=np.array([p],dtype=np.int64); f1=up2.frames_to_bchw(raw[2*idx+1]); tb=torch.from_numpy(tgt[idx]).float()
    J12,r,_=up2.jacobian_and_residual(posenet,st,st.coefficients[idx].clone(),f1,tb,idx)
    rd=r[0].double()
    # 12-dof min-norm step -> image delta in LSB
    dc=torch.linalg.lstsq(J12[0].double(), -rd.unsqueeze(-1)).solution.squeeze(-1)
    dimg12 = AMP*torch.einsum("k,kchw->chw", dc, st.basis_norm.double())
    lsb12 = dimg12.square().mean().sqrt().item()
    codes12 = (dc.abs()/st.coefficient_scales.double()).max().item()
    # 2304-dof (relax the 12-dim basis, KEEP the 24x32 band-limit)
    carrier=torch.einsum("bk,kchw->bchw",st.coefficients[idx],st.basis_norm)/np.sqrt(12)
    low0=(127.5+64.0*carrier).clamp(0,255)
    lowr=functional.interpolate(low0,size=(24,32),mode="area").detach().requires_grad_(True)
    rows=[]
    for c in range(6):
        up_=functional.interpolate(lowr,size=(384,512),mode="bicubic",align_corners=False)
        sl=up2._round_ste(functional.interpolate(up_,size=(874,1164),mode="bicubic",align_corners=False).clamp(0,255))
        sl=up2.apply_selector_float(sl, st.selector_modes, st.selector_choices[idx])
        pose=up2.pose_from_frames(posenet, sl, f1)
        rows.append(torch.autograd.grad(pose[:,c].sum(), lowr)[0].flatten())
    J23=torch.stack(rows).double()
    dl=torch.linalg.lstsq(J23, -rd.unsqueeze(-1)).solution.squeeze(-1)
    dimg23=functional.interpolate(dl.reshape(1,3,24,32),size=(384,512),mode="bicubic",align_corners=False)
    lsb23=dimg23.square().mean().sqrt().item()
    print(f"{int(p):5d} {rd.norm():9.6f} {lsb12:13.4f} {codes12:12.1f} {lsb23:15.4f} {lsb12/max(lsb23,1e-12):9.1f}")
    tot.append((lsb12,lsb23))
a=np.array(tot); print(f"\nmedian 12-dof step {np.median(a[:,0]):.3f} LSB rms  vs  2304-dof {np.median(a[:,1]):.4f} LSB rms  -> basis penalty {np.median(a[:,0]/a[:,1]):.1f}x")
