def brotli11(v):
    c=subprocess.run(["brotli","-q","11","-c"],input=v,capture_output=True,check=True).stdout
    assert subprocess.run(["brotli","-d","-c"],input=c,capture_output=True,check=True).stdout==v
    return c
def build_from_records(recs):
    blob,_=encode_wans1(recs,strategy="per_tensor")
    body=encode_f12_wans_body(blob,ORDER)
    s=brotli11(body)
    model=rx1.pack_rx1_model(hpac_s,s,car_s,codec_id=codec,table_mode=tmode)
    z=rx1.deterministic_zip(model+tail)
    return z,len(s),len(body),blob

def prune(recs, targets, keep_pct):
    out=[]; changed=0; mse_num=0.0; n=0; worst=0.0
    for r in recs:
        if r.schema.name in targets and r.codes is not None:
            codes=np.array(r.codes,dtype=np.int8)
            rows=codes.shape[0]
            flat=codes.reshape(rows,-1).astype(np.float64)
            sc=np.asarray(r.scales,dtype=np.float64).reshape(-1)
            # rank by restored-row energy (codes*scale)
            restored=flat*sc[:,None] if sc.size==rows else flat*sc.reshape(1,-1)[:, :flat.shape[1]]
            norms=(restored**2).sum(axis=1)
            keep=max(1,round(rows*keep_pct/100.0))
            order=np.argsort(-norms, kind="stable")
            drop=order[keep:]
            newcodes=codes.copy().reshape(rows,-1); newcodes[drop]=0
            newcodes=newcodes.reshape(codes.shape)
            oldv=np.asarray(r.values,dtype=np.float32)
            newv=(newcodes.astype(np.float32).reshape(rows,-1)*(sc[:,None] if sc.size==rows else 1.0)).reshape(oldv.shape).astype(np.float32)
            d=np.abs(newv-oldv); worst=max(worst,float(d.max()))
            mse_num+=float(((newv-oldv)**2).sum()); n+=oldv.size
            changed+=int((newcodes!=codes).sum())
            out.append(dataclasses.replace(r, codes=newcodes, values=newv))
        else:
            out.append(r); n+=np.asarray(r.values).size
    return tuple(out), changed, (mse_num/n if n else 0.0), worst
