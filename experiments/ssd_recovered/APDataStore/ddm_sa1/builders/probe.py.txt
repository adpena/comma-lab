P=Path(RT)/"archive.zip"
outer=zipfile.ZipFile(P).read("p")
H=rx1.RX1_HEADER
magic,ver,codec,tmode,res,hb,sb,cb = H.unpack_from(outer)
o=H.size
hpac_s=outer[o:o+hb]; o+=hb
sem_s =outer[o:o+sb]; o+=sb
car_s =outer[o:o+cb]; o+=cb
tail  =outer[o:]
BASE_BYTES=181161

def brotli11(v):
    c=subprocess.run(["brotli","-q","11","-c"],input=v,capture_output=True,check=True).stdout
    back=subprocess.run(["brotli","-d","-c"],input=c,capture_output=True,check=True).stdout
    assert back==v, "brotli parse-back differs"
    return c

def build(sem_body):
    s=brotli11(sem_body)
    model=rx1.pack_rx1_model(hpac_s, s, car_s, codec_id=codec, table_mode=tmode)
    member=model+tail
    z=rx1.deterministic_zip(member)
    return z, len(s), len(sem_body)

# rr4 decoded state
parts=ra.read_residual_archive(P)
recs=ra.decode_wans1(parts.semantic_blob)
state=OrderedDict((r.schema.name, torch.from_numpy(np.ascontiguousarray(r.values,dtype=np.float32))) for r in recs)
print("state tensors:",len(state))

def err(expected):
    tot=0.0; n=0; worst=("",0.0)
    for k,v in state.items():
        e=expected[k].float()
        d=float((e-v).abs().max())
        mse=float(((e-v)**2).mean())
        tot+=mse*v.numel(); n+=v.numel()
        if d>worst[1]: worst=(k,d)
    return tot/n, worst

rows=[]
qn=sd1.quantized_names(state)
print("quantized names:",len(qn))

# C0 all-q4 SD1M control
alloc={n:4 for n in qn}
body,exp = sd1.pack_semantic_state(state, alloc, legacy_int4=False)
z,ps,rb = build(body)
m,w = err(exp)
rows.append(dict(id="C0_sd1m_all_q4", archive=len(z), delta=len(z)-BASE_BYTES, packed_sem=ps, raw_sem=rb, mse=m, worst_name=w[0], worst_abs=w[1], sha=hashlib.sha256(z).hexdigest()))

# C1 mixed q3/q4 (mz2 selected set)

rows.append(dict(id="C1_sd1m_mixed_q3q4", archive=len(z), delta=len(z)-BASE_BYTES, packed_sem=ps, raw_sem=rb, mse=m, worst_name=w[0], worst_abs=w[1], sha=hashlib.sha256(z).hexdigest()))

# C2..C7 SM3R prune ladder

    rows.append(dict(id=f"C_sm3r_keep{pct}", archive=len(z), delta=len(z)-BASE_BYTES, packed_sem=ps, raw_sem=rb, mse=m, worst_name=w[0], worst_abs=w[1], sha=hashlib.sha256(z).hexdigest()))
