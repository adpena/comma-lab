def brotli11(v):
    c=subprocess.run(["brotli","-q","11","-c"],input=v,capture_output=True,check=True).stdout
    assert subprocess.run(["brotli","-d","-c"],input=c,capture_output=True,check=True).stdout==v
    return c
def build(body):
    s=brotli11(body)
    model=rx1.pack_rx1_model(hpac_s,s,car_s,codec_id=codec,table_mode=tmode)
    member=model+tail
    return rx1.deterministic_zip(member), len(s), member
