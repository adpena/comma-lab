def race(row_id):
    return json.loads((ROOT / "race" / f"{row_id}.json").read_text())


def project(code_bytes):
    token = math.ceil(code_bytes)
    archive = RR4_BYTES - (RR4_TOKEN - token)
    return token, archive, INC_S - 25.0 * (INC_BYTES - archive) / DEN


def candidate(label, race_id, timing_id, members, context, role, note):
    row = race(race_id)
    token, archive, projected_s = project(row["code_bytes"])
    timing = race(timing_id) if timing_id else None
    body = {
        "id": label,
        "role": role,
        "members": members,
        "member_count": len(members),
        "mixer_context": context,
        "count_buckets": 1,
        "sse_context": "off  (measured NEGATIVE in 6 of 6 formulations; see the memo section 3)",
        "raced_code_bytes": row["code_bytes"],
        "delta_code_bytes_vs_rr4": row["delta_bytes_vs_live"],
        "delta_code_bytes_vs_fx1": row["delta_bytes_vs_fx1_winner"],
        "projected_token_stream_bytes": token,
        "projected_archive_bytes": archive,
        "projected_S": projected_s,
        "projected_delta_S_vs_fx1": projected_s - INC_S,
        "naming_bar_multiple": abs(projected_s - INC_S) / 1e-5,
        "payload_sha256": row["payload_sha256"],
        "status": "CODE LENGTH MEASURED, NOT BYTE-CLOSED",
        "note": note,
    }
    if timing:
        base = race("T0_serial_shipped_rr4")["elapsed_s"]
        marginal = timing["elapsed_s"] - base
        fx1_marginal = race("T1_serial_fx1_11x")["elapsed_s"] - base
        scale = FX1_PARSEBACK_MARGINAL / fx1_marginal
        projected_parseback = FX1_PARSEBACK_BASE + scale * marginal
        body["decode_cost"] = {
            "serial_harness_seconds": timing["elapsed_s"],
            "serial_harness_marginal_vs_shipped_law": marginal,
            "projected_parseback_seconds": projected_parseback,
            "projected_margin_vs_1800s_budget": BUDGET - projected_parseback,
            "calibration": (
                "ONE point: fx1's 11-member candidate measured +137.49 s on the real parse-back for "
                f"a serial harness marginal of {fx1_marginal:.0f} s (scale {scale:.3f}). The MARGINAL "
                "carries because it is additive host-CPU numpy work; the 1,502.29 s ABSOLUTE base is a "
                "[macOS-CPU advisory] local replay and does NOT carry to T4."
            ),
        }
    return body
