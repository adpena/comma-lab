"""DDM-CK1: build the composed keep01 x S2 candidate body, archive, generation tree.

Composition (both legs ADMITTED upstream, measured on separate rows):
  * keep01  -- SM3R MODE_ROW_PRUNE keep_percent=1 over {blocks.1/2/3.film.weight}
  * S2      -- the two SD1M legs that do NOT overlap the prune:
               frame_embed.weight q4->q3, blocks.0.film.weight q4->q3

Realized as SM3R MODE_ROW_PRUNE_MIXED (mode 6): keep01's exact row geometry
plus an SD1M-style per-tensor depth table.

Controls run before the composed body is trusted:
  C1  mode 5 @ keep_percent=1 reproduces keep01's retained semantic body bytes
  C2  the mode-5 body rebuilds keep01's exact archive (bytes + sha256)
  C3  the composed archive is deterministic (two independent builds agree)
  C4  parse-back through the STAGED shipping runtime returns the packer state
  C5  every non-semantic archive section is byte-unchanged vs keep01

No score claim.  d_seg / d_pose are UNMEASURED here; this run produces bytes
and a decode proof only.
"""

def sha256_bytes(payload: bytes) -> str:
    return hashlib.sha256(payload).hexdigest()


def sha256_file(path: Path) -> str:
    return sha256_bytes(path.read_bytes())


def utc_now() -> str:
    return dt.datetime.now(dt.UTC).strftime("%Y-%m-%dT%H:%M:%SZ")


def atomic_write_bytes(path: Path, payload: bytes) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_name(f".{path.name}.{os.getpid()}.tmp")
    with tmp.open("wb") as handle:
        handle.write(payload)
        handle.flush()
        os.fsync(handle.fileno())
    os.replace(tmp, path)


def artifact(path: Path) -> dict:
    return {"path": str(path), "bytes": path.stat().st_size, "sha256": sha256_file(path)}


def brotli11(value: bytes) -> bytes:
    compressed = subprocess.run(
        ["brotli", "-q", "11", "-c"], input=value, capture_output=True, check=True
    ).stdout
    restored = subprocess.run(
        ["brotli", "-d", "-c"], input=compressed, capture_output=True, check=True
    ).stdout
    if restored != value:
        raise ValueError("brotli round trip differs")
    return compressed


# --------------------------------------------------------------------------
# Base archive sections (identical to the sa1 builders' container path)
# --------------------------------------------------------------------------
base_archive = RT / "archive.zip"
outer = zipfile.ZipFile(base_archive).read("p")
header = rx1.RX1_HEADER
magic, version, codec, table_mode, reserved, hb, sb, cb = header.unpack_from(outer)
offset = header.size
hpac_stream = outer[offset : offset + hb]
offset += hb
offset += sb
carrier_stream = outer[offset : offset + cb]
offset += cb
tail = outer[offset:]


def build_archive(body: bytes) -> tuple[bytes, int]:
    stream = brotli11(body)
    packed = rx1.pack_rx1_model(
        hpac_stream, stream, carrier_stream, codec_id=codec, table_mode=table_mode
    )
    return rx1.deterministic_zip(packed + tail), len(stream)

# The semantic state keep01 was built from

base_parts = ra.read_residual_archive(base_archive)

state: OrderedDict[str, torch.Tensor] = OrderedDict(
    (r.schema.name, torch.from_numpy(np.ascontiguousarray(r.values, dtype=np.float32)))
    for r in records
)
qnames = sd1.quantized_names(state)

checks: dict[str, object] = {}

# C1 -- mode 5 positive control on the real state
keep01_body, keep01_expected, keep01_details = sm3.pack_prune_candidate(state, KEEP_PERCENT)

    keep01_body == retained_body and sha256_bytes(keep01_body) == KEEP01_BODY_SHA256
)

# C2 -- and rebuilds keep01's exact archive
keep01_archive, keep01_stream = build_archive(keep01_body)
checks["C2_mode5_reproduces_keep01_archive"] = (
    len(keep01_archive) == KEEP01_ARCHIVE_BYTES
    and sha256_bytes(keep01_archive) == KEEP01_ARCHIVE_SHA256
)

if not (checks["C1_mode5_reproduces_keep01_body"] and checks["C2_mode5_reproduces_keep01_archive"]):
    raise SystemExit(f"positive controls failed: {json.dumps(checks, indent=2)}")

# --------------------------------------------------------------------------
# The composed body
# --------------------------------------------------------------------------
depths = {name: 3 for name in sorted(sm3.PRUNE_MIXED_Q3_NAMES)}
body, expected, details = sm3.pack_prune_mixed_candidate(state, KEEP_PERCENT, depths)
archive, stream_bytes = build_archive(body)
repeat_archive, repeat_stream = build_archive(body)
checks["C3_archive_is_deterministic"] = archive == repeat_archive and stream_bytes == repeat_stream

# Packer-side round trip before anything is staged.
sm3.assert_state_equal(expected, sm3.unpack_candidate(body, state))
checks["C_packer_roundtrip_exact"] = True

# Row geometry must be identical to keep01's.
checks["C_row_geometry_matches_keep01"] = details["kept_rows"] == keep01_details["kept_rows"]

# --------------------------------------------------------------------------
# Stage the generation tree
# --------------------------------------------------------------------------
if GEN.exists():
    shutil.rmtree(GEN)

def ignore(_directory, names):
    return {n for n in names if n == "__pycache__" or n.startswith("._")}


for name in ("runtime", "cpr1"):
    shutil.copytree(KEEP01_GEN / name, GEN / name, ignore=ignore)
shutil.copy2(KEEP01_GEN / "inflate.sh", GEN / "inflate.sh")

# The receiver: the repo module, which now decodes mode 6.  Its only non-mode-6
# delta against keep01's shipped copy is the module docstring (recorded below).
shipped_receiver = (KEEP01_GEN / "cpr1/ddm_mp2_semantic_receiver.py").read_bytes()
repo_receiver = (REPO / "experiments/ddm_mp2_semantic_receiver.py").read_bytes()
atomic_write_bytes(GEN / "cpr1/ddm_mp2_semantic_receiver.py", repo_receiver)
receiver_diff = "".join(
    difflib.unified_diff(

# inflate.py: re-pin to the composed archive.

if (

):
    raise SystemExit("keep01 inflate.py does not carry the expected pins")
inflate_source = inflate_source.replace(KEEP01_ARCHIVE_SHA256, sha256_bytes(archive))
inflate_source = inflate_source.replace(f"{KEEP01_ARCHIVE_BYTES:_}", f"{len(archive):_}")
atomic_write_bytes(GEN / "inflate.py", inflate_source.encode())
atomic_write_bytes(GEN / "archive.zip", archive)

stripped = strip_import_residue(GEN)
checks["C6_staged_tree_free_of_import_residue"] = True
if sha256_bytes((GEN / "archive.zip").read_bytes()) != sha256_bytes(archive):

# --------------------------------------------------------------------------
# C4/C5 -- parse-back through the STAGED tree
# --------------------------------------------------------------------------
staged = {}

parts = ra.read_residual_archive(GEN / "archive.zip")

    for name in ("hpac_blob", "carrier_blob", "token_stream", "residual_payload")

checks["C4b_staged_receiver_matches_packer_exactly"] = max_dev == 0.0 and list(decoded) == list(
